\section{Homotopy} 

	\begin{definition}[Homotopy]
		Let $f, g: X \to Y$ be continuous maps. $f$ is said to be \textbf{homotopic} to $g$, denoted $f \simeq g$, if there exists a continuous map $H: X \times [0, 1] \to Y$---called a \textit{homotopy}---satisfying 
		\begin{align}
			H(x, 0) & = f(x) \\ 
			H(x, 1) & = g(x) 
		\end{align}
		We claim that homotopy is an equivalence relation. 
		\begin{enumerate}
		  \item $f$ and $G$ are said to be \textbf{homotopic relative to a subset} $S \subset X$ if $H$ also satisfies 
				\begin{align}
					H(s, t) = f(s) = g(s)
				\end{align}
				for all $s \in S$. 

			\item $f$ is said to be \textbf{null-homotopic} if it is homotopic to a constant function. 
		\end{enumerate}

	\begin{proof}
		This is clearly an equivalence relation defined on $C(X, Y)$, the set of all continuous functions from $X$ to $Y$.
		\begin{enumerate}
			\item \textit{Reflexive}. Clearly, $f$ is homotopic to itself by setting $H(x, t) = f(x)$. 

			\item \textit{Symmetric}. Given homotopy $H(x, t)$ from $f$ to $g$, the homotopy $\overline{H}(x, t) \coloneqq H(x, 1-t)$ maps from $g$ to $f$. 

			\item \textit{Transitive}. Given homotopy $H_1$ from $f$ to $g$, and homotopy $H_2$ from $g$ to $h$, the homotopy defined
				\begin{equation}
					H (x, t) \coloneqq \begin{cases}
						H_1 (x, 2t) & 0 \leq t \leq \frac{1}{2} \\
						H_2 (x, 2t - 1) & \frac{1}{2} \leq t \leq 1
					\end{cases}
				\end{equation}
				is indeed a homotopy from $F_1$ to $F_3$. To prove continuity, note that $H_1 (x, 2t)$ is continuous over the closed set $X \times [0, \frac{1}{2}]$ and $H_2 (x, 2t - 1)$ is continuous over the closed set $X \times [\frac{1}{2}, 1]$, so by the gluing lemma $H$ is continuous. 
		\end{enumerate}
	\end{proof}

	\begin{definition}[Homotopy Class]
    The space of homotopy classes from topological space $X$ to $Y$ is denoted
    \begin{equation}
      [X, Y] \coloneqq \frac{C^0 (X, Y)}{\sim}
    \end{equation}
    where $\sim$ is the homotopy relation. 
  \end{definition}

	\begin{example}[Simple Homotopies]
		We look at two homotopies. 
		\begin{enumerate}
		  \item If $f, g: \mathbb{R} \longrightarrow \mathbb{R}^2$ is defined as $f(x) \equiv (x, x^3), \; g(x) \equiv (x, e^x)$, then the map 
				\begin{equation}
					H: \mathbb{R} \times [0,1] \longrightarrow \mathbb{R}^2, \; H(x, t) \equiv \big( x, (1-t) x^3 + t e^x \big)
				\end{equation}
				is a homotopy between them. 

			\item Let $id_B: B^n \longrightarrow B^n$ be the identity function on the unit $n$-disk, and let $c_0: B^n \longrightarrow B^n$ be the $0$-function sending every vector to $0$. Then, $id_B$ and $c_0$ are homotopic, with homotopy explicitly defined
				\begin{equation}
					H: B^n \times [0,1] \longrightarrow B^n, \; H(x, t) \equiv (1-t) x
				\end{equation}
		\end{enumerate}
  \end{example}

	The first realization is that path homotopies are trivial if the codomain is convex. 

  \begin{example}[Homotopy on Convex Spaces]
    If $Y = \mathbb{R}^n$, then for any space $X$, any two continuous maps $f_0, f_1: X \to \mathbb{R}^n$ are homotopic by defining the \textit{linear path homotopy}
    \begin{equation}
      F(x, t) = (1 - t) f_0 (x) + t f_1 (x) 
    \end{equation}
    which is well-defined as $Y$ is convex. 

		Furthermore, it is path connected, and so the value of the constant function does not matter. Any two constant functions are homotopic.\footnote{Let $f, g: \mathbb{R} \longrightarrow \mathbb{R}$ be 2 continuous functions. Then $f \simeq g$, since we can construct $F: [0,1] \times \mathbb{R} \longrightarrow \mathbb{R}$ defined $F(x, t) \equiv (1-t) f(x) + t g(x)$. (Note that the set of continuous functions from $\mathbb{R}$ to $\mathbb{R}$ is a convex set.)}
  \end{example}

  \begin{example}[Two Non-Homotopic Functions]
    Consider $X = S^1, Y = \mathbb{R}^2 \setminus \{0\}$, with $f: S^1 \to Y$ the canonical inclusion, and $g$ any constant map. Then the straight line homotopy no longer works, since the circle gets ``stuck'' on the hole. That gives some sort of rigorous definition of what a hole can be. 
  \end{example}

  However, note that if two functions are null-homotopic, it \textit{does not} mean that they are homotopic to each other. 

  \begin{example}[Null-Homotopic Paths are Not Homotopic]
    Given $X = \mathbb{R}, Y = (0, 1) \cup (2, 3)$, say that we are given the functions $f, g: X \to Y$ defined $f(x) = 0.5, f(y) = 2.5$. Then by definition they are null-homotopic, but they are not homotopic to each other. 
  \end{example}

	\begin{example}[Null-Homotopic Function]
    Take a look at a function $f: \mathbb{R}^2 \longrightarrow \mathbb{R}$, which represents an arbitrary surface in $\mathbb{R}^2 \oplus \mathbb{R}$. Now, observe the constant function $c(x, y) \equiv c$, which represents a plane parallel to the $x, y$-plane. Clearly, we can imagine a deformation of the surface of $f$ to the flat surface of $c$ with the homotopy
    \begin{equation}
      H(x, t) \coloneqq t \, f(x) + (1-t) c(t)
    \end{equation}
    which visually represents a linear deformation of $c$ to $f$. Therefore, $f$ is null-homotopic. 
  \end{example}

  Now we have seen that homotopies form an equivalence class. Now it turns out that under function compositions, homotopies are preserved, making homotopies a congruence class. 

  \begin{theorem}[Homotopies is a Congruence Relation Under Composition]
    Given continuous $f_0, f_1: X \to Y$ and $g_0, g_1: Y \to Z$, 
    \begin{equation}
      f_0 \simeq f_1, g_0 \simeq g_1 \implies g_0 \circ f_0 \simeq g_1 \circ f_1 
    \end{equation}
  \end{theorem}
  \begin{proof}
    We have $F: [0, 1] \times X \to Y$ and $G: [0, 1] \times Y \to Z$. Then, we can simply define the component-wise composition of maps
    \begin{equation}
      G \circ (F \times \mathrm{id}): [0, 1] \times X \to Z, \qquad (G \circ (F \times \mathrm{id})) (t, x) \coloneqq G \big(t, F(t, x) \big)
    \end{equation} 
    which is continuous. Also, it is the case that 
    \begin{align}
      (G \circ F) (0, x) & = G\big(0, F(0, x) \big) = G (0, f_0 (x)) = g_0 (f_0 (x)) = (g_0 \circ f_0) (x) \\ 
      (G \circ F) (1, x) & = G\big(1, F(1, x) \big) = G (1, f_1 (x)) = g_1 (f_1 (x)) = (g_1 \circ f_1) (x) 
    \end{align}
    So this is indeed a homotopy between $g_0 \circ f_0 \simeq g_1 \circ f_1$. 
  \end{proof}

	Therefore, we can compose them with arbitrary continuous functions. 

	\begin{corollary}[Continuous Compositions of Homotopic Functions are Homotopic]
	  If $f_0 \simeq f_1$, and $g$ is a contiuous function, then 
		\begin{equation}
		  g \circ f_0 \simeq g \circ f_1, \quad f_0 \circ g \simeq f_1 \circ g
		\end{equation}
	\end{corollary}
	\begin{proof}
	  Just set $g_0 = g_1 = g$ from the previous theorem. 
	\end{proof}

\subsection{Homotopy Equivalence}

	\begin{definition}[Homotopy Equivalence]
		Two topolgical spaces $X, Y$ are said to be \textbf{homotopy equivalent}, or to have the same \textbf{homotopy type}---denoted $X \simeq Y$---if there exists a pair of continuous maps $f: X \to Y, g: Y \to X$, such that 
		\begin{equation}
		  g \circ f \simeq \id_X, \quad f \circ g \simeq \id_Y
		\end{equation}
		Sometimes, we may say $f: X \to Y$ is a homotopy equivalence if there exists a $g$ satisfying the above. This is an equivalence relation. 
  \end{definition}
	\begin{proof}
	  Reflexive and symmetric properties are obvious. If $X \simeq Y$ (with $f_1: X \to Y, g_1: Y \to X$) and $Y \simeq Z$ (with $f_2: Y \to Z, g_2: Z \to Y$) with 
		\begin{align}
		  g_1 \circ f_1 \simeq \id_X, \quad f_1 \circ g_1 \simeq \id_Y \\ 
		  g_2 \circ f_2 \simeq \id_Y, \quad f_2 \circ g_2 \simeq \id_Z
		\end{align}
		Then, we can construct $f_2 \circ f_1: X \to Z, g_1 \circ g_2: Z \to X$ such that
		\begin{align}
			(g_1 \circ g_2) \circ (f_2 \circ f_1) & = g_1 \circ \id_Y \circ f_1 = \id_X \\ 
			(f_2 \circ f_1) \circ (g_1 \circ g_2) & = f_2 \circ \id_Y \circ g_2 = \id_Z 
		\end{align}
	\end{proof}

	\begin{lemma}[Homotopic to Homotopy Equivalence is Homotopy Equivalence]
		Let $f_0: X \to Y$ be a homotopy equivalence, and $f_1 \simeq f_0$. Then $f_1: X \to Y$ is also a homotopy equivalence. 
	\end{lemma}
	\begin{proof}
		Since $f_0: X \to Y$ is a homotopy equivalence, there exists a $g: Y \to X$ such that 
		\begin{equation}
			f_0 \circ g \simeq \id_Y, \quad g \circ f_0 \simeq \id_X
		\end{equation}
		Since $f_0 \simeq f_1$, we can compose with $g$ to get 
		\begin{align}
			f_1 \simeq f_0 & \implies g \circ f_1 \simeq g \circ f_0 \simeq \id_X \\ 
			f_1 \simeq f_0 & \implies f_1 \circ g \simeq f_0 \circ g \simeq \id_Y
		\end{align}
	\end{proof}

	\begin{theorem}[Homoeomorphic Implies Homotopy Equivalent]
		$X \cong Y \implies X \simeq Y$. 
  \end{theorem}
  \begin{proof}
		Given homeomorphism $f: X \to Y$, we have our desired pairs $f, f^{-1}$. 
  \end{proof}

  Recall that visually, two spaces are homeomorphic if they can be transformed into one another by bending. Homotopy equivalence can be viewed as allows shrinking and expanding transformations, and so the converse is not true.

	\begin{theorem}[$\mathbb{R}^n$]
    $\mathbb{R}^n$ is homotopically equivalent to a point $\{0\}$.
  \end{theorem}
  \begin{proof}
    We claim that the continuous maps (canonical injection and projection)
    \begin{equation}
      id_{\mathbb{R}^n}: \{0\} \longrightarrow \mathbb{R}^n , \; p_0 : \mathbb{R}^n \longrightarrow \{0\}
    \end{equation}
    have the property that 
    \begin{equation}
      id_{\mathbb{R}^n} \circ p_0 \simeq id_{\mathbb{R}^n} , \; p_0 \circ id_{\mathbb{R}^n} \simeq id_{\{0\}}
    \end{equation}
    The right-hand homotopy is trivial since $id_{\mathbb{R}^n} \circ p_0 = id_{\mathbb{R}^n}$, and as for the left-hand homotopy, we can explicitly define it as
    \begin{equation}
      H: [0,1] \times \mathbb{R}^n \longrightarrow \mathbb{R}^n
    \end{equation}
    with
    \begin{equation}
      H(t, x) \equiv (t) (id_{\mathbb{R}^n} \circ p_0 )(x) + (1-t) \, id_{\mathbb{R}^n} (x) = (1-t)\, id_{\mathbb{R}^n} (x)
    \end{equation}
  \end{proof}

	We end with a definition on contractible spaces. 

	\begin{definition}[Contractible]
		$X$ is \textbf{contractible} if is satisfies the equivalent definitions. 
		\begin{enumerate}
			\item $X$ is homotopy equivalent to a point $X \simeq \{x_0\}$. 
			\item $\id_X$ is null-homotopic. 
		\end{enumerate}
  \end{definition}
	\begin{proof}
	  We prove bidirectionally. 
		\begin{enumerate}
			\item $(\rightarrow)$. Since $X \simeq \{x_0\}$, there exists functions $f: X \to \{x_0\}, g: \{x_0\} \to X$ such that $f \circ g \simeq \id_{\{x_0\}}$ and $g \circ f \simeq \id_X$. But $g \circ f: X \to X$ is defined $c_{g(x_0)} \simeq \id_X$. 

			\item $(\leftarrow)$. Let $\id_X \simeq c_{x_0}$. Then, there exists a homotopy $F: X \times [0, 1] \to X$ such that $F(x, 0) = \id_X, F(x, 1) = c_{x_0}$. Now define functions 
				\begin{align}
					f: X \to \{x_0\}, & \quad f(x) = x_0 \\ 
					g: \{x_0\} \to X, & \quad g(x_0) = x_0
				\end{align}
				Therefore, $f \circ g = \id_{\{x_0\}}$ and $g \circ f: X \to X$ satisfies $g \circ f = c_{x_0} \simeq \id_X$ from our homotopy $F$. 
		\end{enumerate}
	\end{proof}

\subsection{Retracts}

	Recall what a \hyperref[pst-def:retraction]{retract} is. 

  \begin{definition}[Retraction]
		If $A \subset X$, a \textbf{retraction} $r: X \to A$ is a continuous function such that $r(a) = a$ for all $a \in A$.  
  \end{definition}

	\begin{example}[Retracts]
    The following are retractions. 
    \begin{equation}
      r: \mathbb{R} \to \{x\} \subset \mathbb{R}, \qquad r(x) = x
    \end{equation}
    This is also a retraction. 
    \begin{equation}
      r: \mathbb{R} \setminus \{0\} \to S^{n-1}, \qquad x \mapsto \frac{x}{\|x\|} 
    \end{equation}
  \end{example}

	\begin{theorem}[Retracts of Contractible Spaces are Contractible]
		If $X$ is contractible and $A \subset X$ is a retract, then $A$ is contractible. 
	\end{theorem}
	\begin{proof}
		Let $\iota: A \to X$ be the canonical inclusion and $r: X \to A$ be the retract. Since $X$ is contractible, $\id_X \simeq c_{x_0}$. Therefore, since continuous compositions of homotopies are homotopic, we have 
		\begin{equation}
			\id_A = r \circ \id_X \circ \iota \simeq r \circ c_{x_0} \circ \iota = c_{r(x_0)}
		\end{equation}
		Therefore, $A$ is contractible. 
	\end{proof}

  It's pretty easy to show whether a particular example is a retraction or not. However, we will mainly be interested in the existence of retractions. 

  \begin{theorem}[Unit Interval to Endpoints]
    There is no retraction from $D^1 = [-1, 1]$ to $S^0 = \{-1, 1\}$. 
  \end{theorem}
  \begin{proof}
    Let $X = [a, b]$. There is no retraction $f: X \to \{a, b\}$ since $\im(f)$ must be connected. But $f(a) = a, f(b) = b$. Alternatively, we can use IVT. 
  \end{proof}

	We would like a stronger version of a retract, where we can continuously deform a topological space $X$ onto subspace $A$, while keeping every point of $A$ fixed throughout the deformation. 

	\begin{definition}[Deformation Retract]
		A subspace $A$ is a \textbf{deformation retract} of $X$ if there exists a homotopy 
		\begin{equation}
			H: X \times [0, 1] \to X
		\end{equation}
		such that 
		\begin{enumerate}
		  \item \textit{Starts at the Identity}. $H(x, 0) = x$ for all $x \in X$. 
			\item \textit{Ends at a Retraction}. $H(x, 1) \in A$ for all $x \in X$. 
			\item \textit{Subspace is Invariant}. $H(a, t) = a$ for all $a \in A$. 
		\end{enumerate}
	\end{definition}

	Note that a retract is simply a continuous map $r: X \to A$ such that $r|_A = \id_A$. Given that $\iota: A \to X$ is the inclusion map, a deformation retracts also requires that $\iota \circ r$ be homotopic to $\id_X$ relative to $A$.

	\end{example}

	  If $A \subset X$ is a deformation retract, then

			A \text{ is contractible } \iff X \text{ is contractible}. 
		\end{equation}
	\end{theorem}

		\begin{enumerate}
		  \item We first claim $S^1 \vee S^1 \cong \frac{\partial [0, 1]^2}{\sim}$. We define the quotient map 
				\begin{equation}
					p: \partial [0, 1]^2 \to \frac{\partial [0, 1]^2}{\sim}, \quad
					p(0, x) = p(1, x), p(y, 0) \sim p(y, 1)
				\end{equation}
				and let $(x, y) \sim (x^\prime, y^\prime)$ if $p(x, y) = p(x^\prime, y^\prime)$. Now let us parameterize $S^1 \vee S^1$ be defining left and right unit circles centered at $1, -1$ on the complex plane. Then, we can define the funtion 
				\begin{equation}
					\varphi: \partial [0, 1]^2 \to S^1 \vee S^1, \quad \varphi(x, y) \coloneqq \begin{cases} 
						0 & \text{ if } (x, y) \in \{(0, 0), (0, 1), (1, 0), (1, 1)\} \\ 
						e^{2\pi i x} - 1 & \text{ if } y \in \{0, 1\} \\
						1 - e^{2 \pi i y} & \text{ if } x \in \{0, 1\}
					\end{cases}
				\end{equation}
				By the \hyperref[pst-thm:universal_quotient]{universal property of the quotient}, combined with the \hyperref[pst-thm:quotient_homeomorphism]{fact that $(x, y) \sim (x^\prime, y^\prime)$ iff $\varphi(x, y) = \varphi(x^\prime, y^\prime)$}, we know that the induced map $\Tilde{\varphi}: \frac{\partial [0, 1]^2}{\sim} \to S^1 \vee S^1$ is a homeomorphism. 

				\begin{figure}[H]
					\centering 
					\begin{tikzcd}
						\partial [0, 1]^2 \arrow{d}{p} \arrow{r}{\varphi} & S^1 \vee S^1 \\
						\partial [0, 1]^2 /{\sim} \arrow[dashed]{ru}{\Tilde{\varphi}} & 
					\end{tikzcd}
				\end{figure}

			\item Now let $p = (\frac{1}{2}, \frac{1}{2})$ and let's define 
				\begin{equation}
					f: [0, 1]^2 \setminus \{p\} \to \frac{\partial [0, 1]^2} {\sim}
				\end{equation}
				to be the radial projection of $x$ to the boundary. Define the linear-path homotopy\footnote{Technically, I need to first construct the deformation retraction in the square, before passing to the quotient since addition isn't defined, but too lazy.}
				\begin{equation}
					F: \big( [0, 1]^2 \setminus \{p\} \big) \times [0, 1] \to \frac{\partial [0, 1]^2}{\sim}, \quad F(x, t) = (1 - t)x + t f(x)
				\end{equation}
				We can see that $x \sim y \implies F(x, t) \sim F(y, t)$ for all $t \in [0, 1]$. So, by the universal property of the quotient, we can  for each $t$ construct the induced map $\Tilde{f}: \frac{[0, 1]^2 \setminus \{p\}}{\sim} \to \frac{\partial [0, 1]^2}{\sim}$.\footnote{but $\Tilde{F}$ might not be jointly continuous on $x$ and $t$?} Doing this for all $t$ gives us our deformation retraction.

				\begin{figure}[H]
					\centering 
					\begin{tikzcd}
						\partial [0, 1]^2 \setminus \{p\} \arrow{d}{p} \arrow{r}{F(\cdot, t)} & \frac{\partial [0, 1]^2}{\sim} \cong S^1 \vee S^1 \\
						\frac{\partial [0, 1]^2 \setminus \{p\}}{\sim} \arrow[dashed]{ru}{\Tilde{F}(\cdot, t)} & 
					\end{tikzcd}
				\end{figure}
		\end{enumerate}
	\end{solution}

	\begin{exercise}[Hatcher 0.5]
		Show that if a space $X$ deformation retracts to a point $x \in X$, then for each neighborhood $U$ of $x$ in $X$ there exists a neighborhood $V \subset U$ of $x$ such that the inclusion map $V \hookrightarrow U$ is nullhomotopic.
	\end{exercise}
	\begin{solution}
		Since $F(x_0, t) = x_0$ for all $t \in [0, 1]$, continuity of $F$ implies that for each $t \in [0, 1]$, there exists a $V_t \ni x_0, I_t \ni t$ s.t. 
		\begin{equation}
		  F(V_t, I_t) \subset U
		\end{equation}
		$\{I_t\}_{t}$ is an open cover of $[0, 1]$. By compactness, there exists a finite subcover $\{I_k\}_{k=1}^n$. Now let 
		\begin{equation}
			V = U \cap \bigcap_{k=1}^n V_n
		\end{equation}
		which is an open neighborhood of $x_0$ in $U$. By taking this intersection, we can claim that 
		\begin{equation}
			F(V \times [0, 1]) \subset U
		\end{equation}
		since for $(v, t)$, there exists a $n$ s.t. $t \in I_n$ and $v \in V \subset V_n$. Now we see that the restriction 
		\begin{equation}
			F|_{V \times [0, 1]} : V \times [0, 1] \to U
		\end{equation}
		is a homotopy satisfying 
		\begin{align}
			F|_{V \times [0, 1]} (v, 0) & = F(v, 0) = \id_X |_V = \iota \\ 
			F|_{V \times [0, 1]} (v, 1) & = F(v, 1) = x_0 
		\end{align}
	\end{solution}

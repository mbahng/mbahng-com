\section{Product Measures} 

\subsection{Complete Measures}

  \begin{definition}[Complete Measure Spaces]
    \label{def:complete-measure-spaces}
    A \hyperref[def:measure-space]{measure space} $(X, \mathcal{M}, \mu)$ is \textbf{complete} if $\mathcal{M}$ contains all subsets of sets of measure $0$. 
  \end{definition}

  \begin{example}[Completeness of Standard Measure Spaces]
    \label{ex:completeness-of-standard-measure-spaces}
    The two standard $\sigma$-algebras on $\mathbb{R}$ behave differently. 
    \begin{enumerate}
      \item $(\mathbb{R}, \mathcal{L}, m)$ is complete. 
      \item $(\mathbb{R}, \mathcal{B}, m)$ is \textit{not} complete, since the Cantor set (a Borel set of \hyperref[def:lebesgue-measure]{Lebesgue measure} $0$) contains a subset that is not Borel. 
    \end{enumerate}
  \end{example}

  \begin{theorem}[Every Measure Space can be Completed]
    \label{thm:every-measure-space-can-be-completed}
    Let $(X, \mathcal{M}, \mu)$ be a \hyperref[def:measure-space]{measure space}. Define $\mathcal{M}_0$ to be the collection of subsets $E$ of $X$ of the form $E = A \cup B$ where 
    \begin{enumerate}
      \item $B \in \mathcal{M}$, 
      \item $A \subset C$ for some $C \in \mathcal{M}$ for which $\mu(C) = 0$.\footnote{So we are basically splitting $E$ into a measure $0$ part $C$ and everything else $B$.}
    \end{enumerate}
    For such a set $E$, define $\mu_0 (E) = \mu(B)$. Then, $\mathcal{M}_0$ is a $\sigma$-algebra that contains $\mathcal{M}$, $\mu_0$ is a measure that extends $\mu$, and $(X, \mathcal{M}_0, \mu_0)$ is a \hyperref[def:complete-measure]{complete measure} space. 
  \end{theorem}

\subsection{Premeasures} 

  Note that given a set function $\mu$ over $\mathcal{S}$, its Carathéodory extension $\overline{\mu}$ need not agree with $\mu$ for sets in $\mathcal{S}$. We would like to find adequate assumptions such that $\overline{\mu}$ is an extension of $\mu$. 

  \begin{definition}[Premeasure]
    \label{def:premeasure}
    Let $\mathcal{S}$ be a collection of subsets of $X$ and $\mu: \mathcal{S} \to [0, +\infty]$ a set function. Then, $\mu$ is called a \textbf{premeasure} if  
    \begin{enumerate}
      \item $\mu$ is finitely additive, 
      \item $\mu$ is countably monotone, 
      \item if $\emptyset \in \mathcal{S}$, then $\mu(\emptyset) = 0$. 
    \end{enumerate}
  \end{definition}

  \begin{theorem}[Premeasure Condition for Carathéodory Measure]
    \label{thm:premeasure-condition-for-caratheodory-measure}
    Let $\mathcal{S} \subset 2^X$ and $\mu: \mathcal{S} \to [0, +\infty]$ a set function. In order for the \hyperref[def:caratheodory-measure]{Carathéodory measure} induced by $\mu$ be an extension of $\mu$, it is necessary that $\mu$ is a \hyperref[def:premeasure]{premeasure}.  
  \end{theorem}

  So being a \hyperref[def:premeasure]{premeasure} is a necessary but not sufficient condition, but if we impose on $\mathcal{S}$ a finer set-theoretic structure, this necessary condition is also sufficient. 

  \begin{definition}[Closure Under Relative Complements]
    \label{def:closure-under-relative-complements}
    A collection $\mathcal{S} \subset 2^X$ is said to be closed w.r.t. the formation of relative complements provided 
    \begin{equation}
      A, B \in \mathcal{S} \implies A \setminus B \in \mathcal{S}
    \end{equation}
  \end{definition}

  However, a number of natural \hyperref[def:premeasure]{premeasures} such as the premeasure length defined on the collection of bounded intervals of reals numbers, are defined on collections of sets that are not closed w.r.t. relative complements. So we consider alternate conditions for extending measures. 

  \begin{definition}[Semiring]
    \label{def:semiring}
    A nonempty collection $\mathcal{S}$ of subsets of a set $X$ is a \textbf{semiring} if 
    \begin{enumerate}
      \item \textit{Closure under finite intersections}. 
        \begin{equation}
          A, B \in \mathcal{S} \implies A \cap B \in \mathcal{S}
        \end{equation}
      \item \textit{Disjoint decomposition of relative complements}. 
        \begin{equation}
          A, B \in \mathcal{S} \implies A \setminus B = \bigsqcup_{k=1}^n C_k
        \end{equation}
        for some collection $C_k \in \mathcal{S}$. 
    \end{enumerate}
  \end{definition}

  \begin{theorem}[Carathéodory-Hahn Theorem]
    \label{thm:caratheodory-hahn-theorem}
    Let $\mu: \mathcal{S} \to [0, +\infty]$ be a \hyperref[def:premeasure]{premeasures} on a \hyperref[def:semiring]{semiring} $\mathcal{S}$ of subsets of $X$. 
    \begin{enumerate}
      \item Then, the \hyperref[def:caratheodory-measure]{Carathéodory measure} $\overline{\mu}$ induced by $\mu$ is an extension of $\mu$. 
      \item Furthermore, if $\mu$ is $\sigma$-finite, then so is $\overline{\mu}$, and $\overline{\mu}$ is the unique measure on the $\sigma$-algebra of $\mu^\ast$-measurable sets that extends $\mu$. 
    \end{enumerate}
  \end{theorem}

\subsection{Product Measure} 

  Before, we saw how we can construct measures using Carathéodory construction. We will consider how to create product measures, which will be an extension of a set functions using the \hyperref[thm:caratheodory-hahn-theorem]{Carathéodory-Hahn theorem}. 

  \begin{definition}[Measurable Rectangle]
    \label{def:measurable-rectangle}
    Let $(X, \mathcal{A}, \mu)$, $(Y, \mathcal{B}, \nu)$ be two \hyperref[def:measure-space]{measure spaces}. Consider the product space $X \times Y$. If $A \in \mathcal{A}, B \in \mathcal{B}$, then the set $A \times B$ is called a \textbf{measurable rectangle}. 
  \end{definition}

  \begin{lemma}[Countable Additivity over Measurable Rectangles]
    \label{thm:countable-additivity-over-measurable-rectangles}
 
    Let $\{A_k \times B_k\}_{k=1}^\infty$ be a countable disjoint collection of \hyperref[def:measurable-rectangle]{measurable rectangles} whose union is also a measurable rectangle $A \times B$. Then, 
    \begin{equation}
      \mu(A) \times \nu(B) = \sum_{k=1}^\infty \mu(A_k) \times \nu(B_k)
    \end{equation}
  \end{lemma}

      \lambda(A \times B) = \mu(A) \cdot \nu(B)
    \end{equation}
    Then, $\mathcal{R}$ is a \hyperref[def:semiring]{semiring} and $\lambda: \mathcal{R} \to [0, +\infty]$ is a \hyperref[def:premeasure]{premeasure}. 
  \end{theorem}

  \begin{definition}[Product Measure]
    \label{def:product-measure}
    Let $(X, \mathcal{A}, \mu)$, $(Y, \mathcal{B}, \nu)$ be two \hyperref[def:measure-space]{measure spaces}, $\mathcal{R}$ the collection of \hyperref[def:measurable-rectangle]{measurable rectangles} contained in $X \times Y$, and $\lambda$ the \hyperref[def:premeasure]{premeasure} defined on $\mathcal{R}$ by 
    \begin{equation}
      \lambda(A \times B) = \mu(A) \cdot \nu(B)
    \end{equation}
    for all $A \times B \in \mathcal{R}$. Then, the \textbf{product measure} $\lambda = \mu \times \nu$ is the Carathéodory extension of $\lambda: \mathcal{R} \to [0, +\infty]$ defined on the $\sigma$-algebra of $(\mu \times \nu)^\ast$-measurable subsets of $X \times Y$. 
  \end{definition}

\subsection{Fubini's and Tonelli's Theorem}

  \begin{theorem}[Fubini's Theorem]
    \label{thm:fubinis-theorem}
    Let $(X, \mathcal{A}, \mu)$, $(Y, \mathcal{B}, \nu)$ be two \hyperref[def:measure-space]{measure spaces} and $\nu$ be complete. Let $f$ be integrable over $X \times Y$ w.r.t. to the \hyperref[def:product-measure]{product measure} $\mu \times \nu$. Then for almost all $x \in X$, the $x$-section of $f$, $f(x, \cdot)$, is integrable over $Y$ with respect to $\nu$, and 
    \begin{equation}
      \int_{X \times Y} f \, d(\mu \times \nu) = \int_X \bigg[ \int_Y f(x, y) \,d\nu(y) \bigg] \, d\mu(x)
    \end{equation}
  \end{theorem}

  \begin{theorem}[Tonelli's Theorem]
    \label{thm:tonellis-theorem}
    Let $(X, \mathcal{A}, \mu)$, $(Y, \mathcal{B}, \nu)$ be two \hyperref[def:measure-space]{measure spaces} and $\nu$ be complete.Let $f$ be a nonnegative $(\mu \times \nu)$-measurable function on $X \times Y$. Then, 
    \begin{enumerate}
      \item For almost all $x \in X$, the $x$-section of $f$, $f(x, \cdot)$, is $\nu$-measurable. 
      \item The function defined \hyperref[def:almost_everywhere]{almost everywhere} on $x$ by 
        \begin{equation}
          x \mapsto \int_Y f(x, y) \,d\nu(y)
        \end{equation}
        is $\mu$-measurable. 
      \item Finally, 
        \begin{equation}
          \int_{X \times Y} f \,d(\mu \times \nu) = \int_X \bigg[ \int_Y f(x, y) \,d\nu(y) \bigg] \,d \mu(x)
        \end{equation}
    \end{enumerate}
  \end{theorem}

\subsection{Exercises} 

  \begin{exercise}[Math 631 Fall 2025, Final Exam Exercise 4]
    Let $A=[-1,1]\times[-1,1]$. Let $f(x,y)=\frac{xy}{(x^{2}+y^{2})^{2}}$ (when $x=0$ or $y=0$ we set $f(x,y)=0$). Prove that the iterated integrals exist and are equal:
    \begin{equation}
      \int_{-1}^{1}(\int_{-1}^{1}f(x,y)dx)dy=\int_{-1}^{1}(\int_{-1}^{1}f(x,y)dy)dx,
    \end{equation}
    but the double integral $\int_{A}f(x,y) dxdy$ does not exist.
  \end{exercise}
  \begin{solution}
    Note that $f$ is odd with respect to both $x$ and $y$. Therefore
    \begin{equation}
      \int_{-1}^{1}f(x,y)dx = \int_{-1}^{1}f(x,y)dy=0
    \end{equation}
    for all $y$ and $x$ respectively. Hence both iterated integrals are zero. Now passing to polar coordinates, we have that $|f(r,\theta)|=\frac{|\sin 2\theta|}{2r^{2}}$. Therefore, for every $\epsilon>0$,
    \begin{equation}
      \int_{A}|f(x,y)|dxdy\ge\int_{\epsilon}^{1}\int_{0}^{\pi/2}\frac{\sin 2\theta}{r^{2}}rd\theta dr\ge c\int_{\epsilon}^{1}r^{-1}dr.
    \end{equation}
    Since $\int_{\epsilon}^{1}r^{-1}dr$ diverges as $\epsilon\rightarrow0$, $f$ is not integrable over $A$.
  \end{solution}

\section{Higher Order Derivatives} 

  So far, we have talked about derivatives as \hyperref[lin-def:linear_map]{linear maps}. However, this language is a bit too restrictive because it doesn't allow us to represent the derivatives of higher order using matrices. It turns out that these higher order derivatives are tensors. 

\subsection{Second Order Derivatives}

	\begin{definition}[Second Derivative]
	  \label{def:second-derivative}
		Let $f: E \subset \mathbb{R}^n \to \mathbb{R}$ be a function that is differentiable at $a$. Then, $Df_a: \mathbb{R}^n \to \mathbb{R}$. 

		Then, the \textbf{second total derivative} is a \hyperref[lin-def:bilinear-map]{bilinear map} 
	\end{definition}

  \begin{definition}[Hessian]
    \label{def:hessian}
    Given $f: E \subset V \to \mathbb{R}$ twice-differentiable at $x$, the \textbf{Hessian} of $f$ at $x$ is the matrix realization of the total derivative, denoted $Hf_x$. The entries of the Hessian matrix are precisely the partials. 
    \begin{equation}
      H f_{x} \coloneqq \begin{pmatrix} 
        \partial_{x_1 x_1} (a) & \ldots & \partial_{x_1 x_n} (a) \\
        \vdots & \ddots & \vdots \\
        \partial_{x_n x_1} (a) & \ldots & \partial_{x_n x_n} (a) \\
        \end{pmatrix} \text{ and } H f \coloneqq \begin{pmatrix} 
        \partial_{x_1 x_1} & \ldots & \partial_{x_1 x_n} \\
        \vdots & \ddots & \vdots \\
        \partial_{x_n x_1} & \ldots & \partial_{x_n x_n} \\
      \end{pmatrix}
    \end{equation}
  \end{definition}

	So, we can see that the \hyperref[def:hessian]{Hessian} matrix must be symmetric for a twice-differentiable function. 

  \begin{theorem}[Clairut's Theorem]
    \label{thm:clairuts-theorem}
    Given $f \in C^2$ at point $a$, its second iterated partials are equal. 
    \begin{equation}
      \partial_{x_i x_j} f (a)= \partial_{x_j x_i} f (a)\text{ for } i, j = 1, 2, \ldots, n
    \end{equation}
  \end{theorem}
  \begin{proof}
    For clarity, denote $x_i, x_j$ as $x, y$ and ignore the rest of the variables. Then, the \hyperref[def:partial-derivative]{partial derivatives} $\partial_{x y} f$ and $\partial_{y x} f$ at a point $(x_0, y_0)$ can be expressed as double limits: 
    \begin{equation}
      \partial_{x y} f (x_0, y_0) = \lim_{y \rightarrow y_0} \frac{\partial_x f (x_0, y) - \partial_x f (x_0, y_0)}{y - y_0}
    \end{equation}
    where $\partial_x f: D \subset \mathbb{R}^n \to \mathbb{R}$. We can use the two limit definitions of partial derivatives
    \begin{equation}
      \partial_x f (x_0, y) = \lim_{x \rightarrow x_0} \frac{f(x, y) - f(x_0, y)}{x-x_0} \text{ and } \partial_x f (x_0, y_0) = \lim_{x \rightarrow x_0} \frac{f(x, y_0) - f(x_0, y_0)}{x-x_0}
    \end{equation}
    and substitute them to get the two partials
    \begin{align}
      \partial_{xy} f (x_0, y_0) & = \lim_{y \rightarrow y_0} \frac{ \lim_{x \rightarrow x_0} \frac{f(x, y) - f(x_0, y)}{x-x_0} - \lim_{y \rightarrow y_0} \frac{f(x, y_0) - f(x_0, y_0)}{x-x_0}}{y - y_0} \\
      & =  \lim_{y \rightarrow y_0} \lim_{x \rightarrow x_0} \bigg( \frac{f(x, y) - f(x_0, y) - f(x, y_0) + f(x_0, y_0)}{(x - x_0) (y - y_0)} \bigg) \\
      \partial_{yx} f (x_0, y_0) & = \lim_{x \rightarrow x_0} \frac{ \lim_{y \rightarrow y_0} \frac{f(x, y) - f(x, y_0)}{y-y_0} - \lim_{y \rightarrow y_0} \frac{f(x_0, y) - f(x_0, y_0)}{y-y_0}}{x - x_0} \\
      & = \lim_{x \rightarrow x_0} \lim_{y \rightarrow y_0} \bigg( \frac{f(x, y) - f(x, y_0) - f(x_0, y) + f(x_0, y_0)}{(y-y_0) (x-x_0)} \bigg)
    \end{align}
    Now invoking our assumption that $f$ is $C^2$, the two limits, which approach $(x_0, y_0)$ along different paths, both exist and are equal to 
    \begin{equation}
      \lim_{(x, y) \rightarrow (x_0, y_0)} \frac{f(x, y) - f(x_0, y) - f(x, y_0) + f(x_0, y_0)}{(x - x_0) (y - y_0)}
    \end{equation}
    and therefore $\partial_{x y} f = \partial_{y x} f$. 
  \end{proof}

	Once we have computed the \hyperref[def:hessian]{Hessian}, let's take the eigendecomposition of it. Since $H f_a$ is a real symmetric matrix, by the spectral theorem, it will have $n$ real eigenvalues $\lambda_1, \ldots, \lambda_n$ (in descending values) and corresponding orthonormal eigenvectors $\mathbf{v}_1, \ldots, \mathbf{v}_n$. Now given the gradient $\nabla f(a)$ at $a$, we can approximate $\nabla f(a + \mathbf{h})$ at $a + \mathbf{h}$ using its total derivative as 
	\begin{equation}
  	\nabla f(a + \mathbf{h}) \approx \nabla f(a) + D \nabla f_a \mathbf{h} = \nabla f(a) + H f(a) \mathbf{h}
	\end{equation}
  The eigenvalues of $H f(a)$ will tell us how "fast" the gradient changes at $a$. That is, given a small displacement vector $\mathbf{h}$, we can take an orthonormal decomposition of it in the form 
	\begin{equation}
  	\mathbf{h} = \sum_{i} h_i \mathbf{v}_i
	\end{equation}
  and now the approximate gradient can be written as 
	\begin{equation}
  	\nabla f (a + \mathbf{h}) = \nabla f(a) + \sum_i h_i \lambda_i \mathbf{v}_i
	\end{equation}

  Therefore, bigger $\lambda_i$'s will contribute to a greater change in $f(a)$, and smaller ones will contribute less. We can use this information to speed up convergence by scaling along different axes of $\mathbf{h}$ when sampling. 

\subsection{Convexity}

	Recall what a \hyperref[real-def:convex_functions]{convex function} is in univariate analysis. We can extend this to the multivariate case. 

	\begin{definition}[Convex Set]
	  \label{def:convex-set}
	  A set $D \subset \mathbb{R}^n$ is \textbf{convex} if for all $x, y \in D$, 
		\begin{equation}
			tx + (1 - t) y \in D \quad \forall t \in [0, 1]
		\end{equation}
		That is, $D$ contains the straight line connecting $x$ and $y$. 
	\end{definition}

	\begin{definition}[Convex, Concave Function]
		\label{def:convex_function}
		Let $D \subset \mathbb{R}^n$ be convex.\footnote{Note that we require the domain to be convex.} $f: D \to \mathbb{R}$ is a 
		\begin{enumerate}
		  \item \textbf{convex function} if 
				\begin{equation}
					f\big( t x + (1 - t) y \big) \leq t f(x) + (1 - t) f(y) \quad \forall t \in [0, 1]
				\end{equation}

		  \item \textbf{concave function} if 
				\begin{equation}
					f\big( t x + (1 - t) y \big) \geq t f(x) + (1 - t) f(y) \quad \forall t \in [0, 1]
				\end{equation}
		\end{enumerate}
	\end{definition}

	For smooth functions $f: \mathbb{R}^n \to \mathbb{R}$, we have an equivalent formulation of convexity. 

  \begin{theorem}[Convexity of $C^1$ Functions]
    \label{thm:convexity-of-c1-functions}
		Let $D$ be a \hyperref[def:convex-set]{convex set} and $f: E \subset \mathbb{R}^n \to \mathbb{R}$ be $C^1$. Then, $f$ is convex over $E$ if and only if 
		\begin{equation}
			f(a) + D f_{a} (b - a) \leq f(b)
		\end{equation}
		for all $a, b \in E$. 
  \end{theorem}

	This theorem for $C^1$ functions is quite intuitive, since for a convex function, the tangent plane on its graph must never be "above" the graph. In other words, the first order approximation must be a global underestimate of $f$. 

  \begin{theorem}[Convexity of $C^2$ Functions]
    \label{thm:convexity-of-c2-functions}
		Let $f: E \subset \mathbb{R}^n \to \mathbb{R}$ be a $C^2$-function over $E$. Then, $f$ is convex over $E$ if and only if $H f$ is positive semidefinite over all interior points of $E$. 
  \end{theorem}

	In order to determine whether a \hyperref[real-def:critical-point]{critical point} $a$ is a relative maximum, minimum, or neither, we use the \hyperref[real-thm:second-der-test]{second derivative test}. 

  \begin{theorem}[2nd Derivative Test]
    \label{thm:2nd-derivative-test}
    Let $a$ be a \hyperref[real-def:critical-point]{critical point} of $C^2$ function $f: \mathbb{R}^n \to \mathbb{R}$. That is, $D f_{a} = \mathbf{0}$. Then, 
    \begin{enumerate}
      \item $a$ is a local minimum if $H f_{a}$ is positive definite.
      \item $a$ is a local maximum if $H f_{a}$ is negative definite. 
			\item $a$ is a \textit{saddle point} if $H f_{a}$ is not positive definite nor negative definite. 
    \end{enumerate}
  \end{theorem}

  Visually, this makes sense since given a \hyperref[real-def:critical-point]{critical point} $a$, the derivative matrix would be $0$, meaning that the 2nd degree Taylor expansion of $f$ near $a$ would be in form
	\begin{equation}
  	f(x) \approx f(a) + \frac{1}{2} (x - a)^T H f_{a} (x - a)
	\end{equation}

  If $H f_{a}$ is positive definite, then by definition $\frac{1}{2} (x - a)^T H f_{a} (x - a) > 0$ for all $x$ near $a$, and so $f$ would increase in every direction within the neighborhood of $a$. The logic follows similarly for negative definite matrix $H f_{a}$. If $H f_{a}$ is not positive nor negative definite, then $\frac{1}{2} (x - a)^T H f_{a} (x - a)$ could be positive or negative, depending on which direction vector $\mathbf{h} = x - a$ we choose for computing the \hyperref[def:directional-derivative]{directional derivative}. Therefore, $f$ will increase for certain $\mathbf{h}$ and decrease for other $\mathbf{h}$. 

	\begin{definition}[Coercive Function]
		\label{def:coercive}
	  A function $f: \mathbb{R}^d \to \mathbb{R}$ is \textbf{coercive} if 
		\begin{equation}
			\lim_{\|x\| \to +\infty} f(x) = +\infty
		\end{equation}
	\end{definition}

	\begin{definition}[Co-Coercive Function]
		\label{def:cocoercive}
		A function $F: \mathbb{R}^d \to \mathbb{R}^d$ is \textbf{$\beta$-co-coercive} if for some $\beta > 0$, 
		\begin{equation}
			\langle F(x) - F(w), x - w \rangle \geq \beta \|F(x) - F(w)\|^2
		\end{equation}
		holds for all $x, w \in \mathbb{R}^d$. 
	\end{definition}

	\begin{definition}[Strongly Monotone Function]
		\label{def:strongly_monotone}
		A function $F: \mathbb{R}^d \to \mathbb{R}^d$ is \textbf{$c$-strongly monotone} if for some $c > 0$, 
		\begin{equation}
			\langle F(x) - F(y), x - y \rangle \geq c \|x - y\|^2
		\end{equation}
		holds for all $x, y \in \mathbb{R}^d$. 
	\end{definition}

	\begin{definition}[$\mu$-Strongly Convex]
	  \label{def:mu-strongly-convex}
		A function $f: \mathbb{R}^d \to \mathbb{R}$ is \textbf{$\mu$-strongly convex} if it satisfies the equivalent conditions below.\footnote{$f$ is just convex if $\mu = 0$.} 
		\begin{enumerate}
		  \item \textit{Strong Convexity}. For all $w, x \in \mathbb{R}^d$, 
				\begin{equation}
					f(x) \geq f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - w\|^2
				\end{equation}
				Note that $f$ is convex when $\mu = 0$. 

			\item \textit{Strongly Monotone Gradient}. The gradient $\nabla f$ is \hyperref[mv-def:strongly_monotone]{strongly monotone}, i.e. for all $x, w \in \mathbb{R}^d$, 
				\begin{equation}
				  \langle \nabla f(x) - \nabla f(w), x - w \rangle \geq \mu \|x - w\|^2
				\end{equation}
		\end{enumerate}
	\end{definition} 
	\begin{proof}
	  We prove bidirectionally. 
		\begin{enumerate}
		  \item $(\rightarrow)$. From strong convexity, we can see that 
				\begin{align}
					\langle \nabla f(w), x - w \rangle & \leq f(x) - f(w) - \frac{\mu}{2} \|x - w\|^2 \\ 
					\langle \nabla f(x), w - x \rangle & \leq f(w) - f(x) - \frac{\mu}{2} \|x - w\|^2
				\end{align}
				Therefore, 
				\begin{align}
					\langle \nabla f(x) - \nabla f(w), x - w \rangle 
						& = \langle \nabla f(x), x - w \rangle - \langle \nabla f(w), x - w \rangle \\ 
						& \geq \big\{ -f(w) + f(x) + \frac{\mu}{2} \|x - w\|^2 \big\} + \big\{ - f(x) + f(w) + \frac{\mu}{2} \| x - w\|^2 \big\} \\ 
						& = \mu \|x - w\|^2
				\end{align}

			\item $(\leftarrow)$. We define the straight line path 
				\begin{equation}
					\varphi: [0, 1] \to \mathbb{R}, \quad \varphi(t) = f\big( w + t(x - w) \big)
				\end{equation}
				Then, by FTC
				\begin{align}
					f(x) = \varphi(1) & = \varphi(0) + \int_0^1 \varphi^\prime (t) \,dt \\ 
														& = \varphi(0) + \int_0^1 \langle \nabla f(w + t(x - w)), x - w \rangle \,dt  \\ 
														& \geq  f(w) + \int_0^1 \mu t \|x - w\|^2 + \langle \nabla f(w), x - w \rangle \,dt \\
														& = f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - w\|^2
				\end{align}
				where for the inequality, we invoke strong monotonicity and have 
				\begin{equation}
				  \langle \nabla f(w + t(x - w)) - \nabla f(w), t (x - w) \rangle \geq \mu t^2 \|x - w\|^2
				\end{equation}
				After dividing by $t$ and rearranging, we have 
				\begin{equation}
				  \langle \nabla f(w + t(x - w)), x - w \rangle \geq \mu t \|x - w\|^2 + \langle \nabla f(w), x - w \rangle
				\end{equation}
		\end{enumerate}
	\end{proof}

	This is a first order description of \hyperref[mu-strongly-convex]{$\mu$-strongly convex functions}. Here is a second order one. 

	\begin{lemma}[2nd Order Definition of Strongly Convex Functions]
	  \label{thm:2nd-order-definition-of-strongly-convex}
	  A $C^2$ function $f: \mathbb{R}^d \to \mathbb{R}$ is $\mu$-strongly convex iff $Hf(x) \succeq \mu I$.\footnote{For \hyperref[lin-def:symmetric-matrices]{symmetric matrices} $A, B$, $A \succeq B$ if $A - B$ is positive semidefinite.}
	\end{lemma}
	\begin{proof}
		We prove bidirectionally. Given $x, w \in \mathbb{R}^d$, let $g(t) = f(w + t(x - w))$, which is $C^2$. 
		\begin{enumerate}
		  \item $(\rightarrow)$. Take the second order Taylor expansion and invoke strong convexity to get the inequality 
				\begin{align}
					g(t) & = f(w) + t \langle \nabla f(w), x - w \rangle + \frac{1}{2} t^2 \langle \nabla^2 f(w) (x - w), x - w \rangle + o(\|t(x - w)\|^2) \\ 
							 & \geq f(w) + \langle \nabla f(w), t (x - w) \rangle + \frac{\mu t^2}{2} \|x - w\|^2
				\end{align}
				Cleaning this up gives 
				\begin{equation}
					\langle \nabla^2 f(w) (x - w), x - w \rangle + \frac{o(t^2 \|x - w\|^2)}{t^2} \geq \mu \|x - w\|^2
				\end{equation}
				This holds for all $t \in [0, 1]$, so by letting $t \to 0$, we have 
				\begin{equation}
				  \langle \nabla^2 f(w) (x - w), x - w \rangle \geq \mu \|x - w\|^2 \implies Hf(w) \succeq \mu I
				\end{equation}

		  \item $(\leftarrow)$. We can see that 
				\begin{align}
					g^\prime (t) & = \langle \nabla f(w + t(x - w)), x - w \rangle \\ 
					g^{\prime\prime} (t) & = (x - w)^T H f(w + t(x - w)) (x - w) \\ 
															 & \geq \mu (x - w)^T (x - w) = \mu \|x - w\|^2
				\end{align}
				So, using \hyperref[real-thm:integral_form_remainder]{Taylor's expansion with the integral form of the remainder}, we have 
				\begin{align}
					f(x) = g(1) & = g(0) + g^\prime (0) (1 - 0) + \int_0^1 g^{\prime\prime} (t) (1 - t) \,dt \\ 
											& \geq g(0) + g^\prime (0) + \mu \|x - w\|^2 \int_0^1 (1 - t) \,dt \\ 
											& = f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - w\|^2
				\end{align}
		\end{enumerate}
	\end{proof}

	\begin{example}[Strongly Convex and non-Strongly Convex Functions]
	  Here are some examples. For twice-differentiable $f$ on $\mathbb{R}^n$, $\mu$-strong convexity is equivalent to $\nabla^2 f(x) \succeq \mu I$ for all $x$, which makes most cases easy to check. 
		\begin{enumerate}
			\item \textit{Strongly convex}. $f(x) = \frac{1}{2} x^T A x - b^T x$ with $A \succ 0$ is $\lambda_{\min}$-strongly convex since it is $C^2$ and 
				\begin{equation}
					Hf(x) = A \succeq \lambda_{\min} I
				\end{equation}

		  \item \textit{Strongly convex}. $f(x) = \cosh(x) = \frac{e^x + e^{-x}}{2}$ is $1$-strongly convex since 
				\begin{equation}
				  f^{\prime\prime}(x) = \cosh(x) \geq 1
				\end{equation}

		  \item \textit{Convex but not strongly convex (flat directions)}. $f(w) = \frac{1}{2}\|Xw - y\|^2$ has Hessian $X^T X$, which is singular if $\operatorname{rank} X < d$ (e.g.\ $n < d$), so $\lambda_{\min}(X^T X) = 0$. Linear functions $f(x) = a^T x$ are the extreme case, with $\nabla^2 f = 0$. 

		  \item \textit{Convex but not strongly convex (vanishing curvature)}. $f(x) = x^4$ is strictly convex with a unique minimizer, but it is not strongly convex. Indeed, 
				\begin{equation}
					f^{\prime\prime}(x) = 12x^2 \to 0 \text{ as } x \to 0
				\end{equation}
				So strict convexity is strictly weaker than strong convexity. 


			\item \textit{Convex but not strongly convex (vanishing curvature)}. $f(x) = e^x$ is strictly convex but is not strongly convex. Indeed, 
				\begin{equation}
					f^{\prime\prime}(x) = e^x \to 0 \text{ as } x \to -\infty
				\end{equation}

			\item \textit{Convex but not strongly convex}. $f(x) = \log(1 + e^{-x})$ is strictly convex but not strongly convex. Indeed, 
				\begin{equation}
					f^{\prime\prime}(x) = \frac{e^{x}}{(1 + e^{x})^2} \to 0 \text{ as } |x| \to \infty
				\end{equation}

		  \item \textit{Not convex}. $f(x) = x^2 + 3\sin^2 x$ has $f^{\prime\prime}(x) = 2 + 6\cos 2x$, which is as low as $-4$, so it is not even convex. 
		\end{enumerate}
	\end{example}

\subsection{k-Times Continuously Differentiable Functions}

	Recall \hyperref[lin-thm:tensors_as_linear_maps]{that a linear map $f: V \to W$ is an element of the tensor product space $V^\dagger \otimes W$}, where $V^\dagger$ represents the \hyperref[lin-def:dual-space]{dual space} of $V$. That is, 
  \begin{equation}
    \mathrm{Hom}(V, W) \simeq V^\dagger \otimes W
  \end{equation}

  \begin{definition}[Frechet Derivative]
    \label{def:frechet-derivative}
    A function $f: V \to W$ is \textbf{differentiable} at $x \in V$ if there exists a unique tensor $D f_x \in V^\dagger \otimes W$---called the \textbf{total derivative} or \textbf{Frechet derivative}---such that 
    \begin{equation}
      \lim_{h \to 0} \frac{\| f(x + h) - f(x) - D f_x h \|}{\| h \|} = 0 
    \end{equation} 
  \end{definition}

  It's a simple---almost trivial---change, but extremely powerful. Note that if $f$ is differentiable over $V$, then the transformation $Df: V \to (V^\dagger \otimes W)$ that maps $x_1$ to its \hyperref[def:frechet-derivative]{Frechet derivative} $Df_{x_1}$ can also be analyzed. Supposing that it is differentiable, the derivative of it at $x$ is 
  \begin{equation}
    D^2 f_x \coloneqq D(Df)_x \in V^\dagger \otimes (V^\dagger \otimes W) \simeq (V^\dagger)^{\otimes 2} \otimes W
  \end{equation}

  by the definition above. Intuitively this also make sense: if we want to take the \hyperref[def:second-derivative]{second derivative} of $f$, we must first specify the point in the domain of $Df$ (our velocity) plus the point in the domain of $f$ itself (our position) to compute it (the acceleration). Continuing this pattern, we get the following recursive definition. 

  \begin{definition}[$k$th Frechet Derivative]
    \label{def:kth-frechet-derivative}
    For $k > 1$, a function $f: V \to W$ is \textbf{$k$-times differentiable} at $x \in V$ if 
    \begin{enumerate}
      \item $f$ is $k-1$ times differentiable at $x \in V$, and 
      \item  there exists a unique rank $(k, 1)$ tensor $D^k f_x \in (V^\dagger)^{\otimes k} \otimes W$---called the \textbf{$k$th total derivative} or \textbf{$k$th Frechet derivative}---such that 
      \begin{equation}
        \lim_{h \to 0} \frac{\| D^{k-1} f_x (x + h) - D^{k-1} f_x (x) - D^k f_x h \|}{\| h \|} = 0 
      \end{equation} 
      where $D^{k-1} f_x$ is the $(k-1)$th derivative of $f$ at $x$.
    \end{enumerate}
  \end{definition}

  Generally, multivariate calculus courses do not go into \hyperref[lin-def:tensor-algebra]{tensor algebra}, so these higher order derivatives are omitted; you only work with the \hyperref[def:hessian]{Hessian} for scalar-valued functions. But we won't be scared off and will present this in full generality, and this gives us two major benefits: 
  \begin{enumerate}
    \item We can compute higher order derivative as we have seen. 
    \item We can compute functions that also input or output tensors, such as matrix-valued functions. This knowledge becomes very useful for optimization, e.g. when you optimize a neural network with respect to matrices (in fully-connected layers) and 3-tensors (e.g. \hyperref[real-thm:convolution]{convolutions}) and is a must-know for building autograd libraries.  
  \end{enumerate} 

  \begin{definition}[$k$th Partial Derivative]
    \label{def:kth-partial-derivative}
    
  \end{definition}

  \hyperref[def:directional-derivative]{Directional derivatives} lose meaning in these higher order regimes, and we just work with iterated partials. 

  \begin{definition}[$C^k$ Functions]
		\label{def:ck_functions}
    A function $f: D \subset \mathbb{R}^n \to \mathbb{R}$ is said to be a $C^k$ function if all $k$-times iterated \hyperref[def:partial-derivative]{partial derivatives} 
    \begin{equation}
      \partial_{x_{i_1} x_{i_2} \ldots x_{i_k}} f
    \end{equation}
    exist and are continuous. The \hyperref[lin-def:vector-space]{vector space} of all $C^k$ functions is denoted $C^k (D; \mathbb{R})$, or $C^k(D)$. 
  \end{definition}

  Whenever we want to get the $k$th iterated \hyperref[def:partial-derivative]{partial derivative} of $f$, we will assume that $f \in C^k$. Again, this is overkill, but it is conventional since we don't really work with the set of functions with existing partial derivatives. Note that mathematicians throw around the word "smooth" a lot. Usually, it means one of three things: it is of class $C^1$, $C^\infty$, or $C^k$ where $k$ is however high it needs to be to satisfy our assumptions. For example, if I say let us differentiate smooth $f$ two times, then I am assuming that $f \in C^2 (\mathbb{R}^n)$. 

  Visualizing $C^k$-functions is easy for low orders. A $C^0$ function produces a graph that isn't "ripped" or "punctured," since this is exactly what a discontinuity would look like. A $C^1$ function requires the surface to be smooth in such a way that there is a well defined affine tangent subspace at every point. This means that there cannot be any sharp "points" or "edges" on the graph since a tangent subspace cannot be well defined. 

  \begin{corollary}[Symmetry in $k$th Iterated Partials]
    \label{thm:symmetry-in-kth-iterated-partials}
    Given $f \in C^k$, its $k$th iterated partials are equal. That is, given any permutation $\sigma$, 
    \begin{equation}
      \partial_{x_{i_1} x_{i_2} \ldots x_{i_k}} f = \partial_{x_{\sigma(i_1)} x_{\sigma(i_2)} \ldots x_{\sigma(i_k)}} f \text{ for } i_1, \ldots, i_k = 1, \ldots, n
    \end{equation}
  \end{corollary}
  \begin{proof}
    By induction. 
  \end{proof}

\subsection{Taylor Series}

  To talk about convergence, the \hyperref[real-def:big-o-notation]{big-O notation} is very useful. 

  \begin{definition}[Classes of Infinitesimal Functions]
    \label{def:classes-of-infinitesimal-functions}
		A function $\alpha: \mathbb{R}^n \to \mathbb{R}$ is \textbf{infinitesimal} if $\alpha \rightarrow 0$ as $x \rightarrow a$. There are multiple "levels" of infinitesimal functions, i.e. how fast they converge to $0$. We can classify them by comparing their limits to polynomials. 
		\begin{enumerate}
			\item $\alpha$ is of class $O(1)$ if 
			\begin{equation}
			  \lim_{x \rightarrow a} \frac{\alpha(x)}{1} = 0
			\end{equation}
			This means that $\alpha(x)$ tends to $0$ infinitely faster than $1$ (which just means that it tends to $0$). 
			
			\item $\alpha$ is of class $O(h)$ if 
			\begin{equation}
			  \lim_{x \rightarrow a}
			  			\frac{\alpha(x)}{\|x - a\|} = 0
			\end{equation}
			This means that $\alpha(x)$ tends to $0$ infinitely faster than the linear $\|\mathbf{h}\|$, where $\mathbf{h} = x - a$. 
			
			\item $\alpha$ is of class $O(h^2)$ if 
			\begin{equation}
			  \lim_{x \rightarrow a} \frac{\alpha(x)}{\|x - a\|^2} = 0
			\end{equation}
			This means that $\alpha(x)$ tends to $0$ infinitely faster than the quadratic $\|\mathbf{h}\|^2$, where $\mathbf{h} = x - a$. 
			
			\item $\alpha$ is of class $O(h^k)$ if 
			\begin{equation}
			  \lim_{x \rightarrow a} \frac{\alpha(x)}{\|x - a\|^k} = 0
			\end{equation}
			This means that $\alpha(x)$ tends to $0$ infinitely faster than the $k$th-order $\|\mathbf{h}\|^k$, where $\mathbf{h} = x - a$. 
		\end{enumerate}
		Clearly, $O(h^k) \supset O(h^{k+1})$. 
  \end{definition}

  Now given a $f: D \subset \mathbb{R}^n \to \mathbb{R}$, we present some polynomial approximations of $f$ at $a \in D$: 
  \begin{enumerate}
    \item If $f \in C^0$, the zeroth (constant) approximation is just 
    \begin{equation}
      P_0 (x) = f(a)
    \end{equation}
    This is not interesting at all, since it is just constant. Furthermore, the error term $\epsilon_0 (x) = f(x) - P_0 (x)$ is an infinitesimal function as $x \rightarrow a$ and is of class $O(1)$, since 
    \begin{equation}
      \lim_{x \rightarrow \mathbf{x_0}} \frac{f(x) - P_0 (x)}{1} = 0
    \end{equation}
    
    \item If $f \in C^1$, the first (linear) approximation requires us to use our total derivative: 
    \begin{equation}
      P_1 (x) = f(a) + D f_{a} (x - a)
    \end{equation}
    and we know that the error $\epsilon_1 (x) = f(x) - P_1 (x)$ is infinitesimal as $x \rightarrow a$ and is of class $O(h)$, since 
    \begin{equation}
      \lim_{x \rightarrow a} \frac{f(x) - P_1(x)}{\|x - a\|} = 0
    \end{equation}
    
    \item If $f \in C^2$, the second (quadratic) approximation requires us to use a quadratic term (i.e. a bilinear form of $\mathbf{h} = x - a$) centered at $a$. Call it $H_{a}: \mathbb{R}^n \times \mathbb{R}^n \to \mathbb{R}$, and our estimation is 
    \begin{equation}
      P_2 (x) = f(a) + D f_{a} (x - a) + \frac{1}{2} H (x - a, x - a)
    \end{equation}
    which we would like the error term $\epsilon_2 (x) = f(x) - P_2 (x)$ to be $O(h^2)$, or in limit terms, $P_2$ must satisfy 
    \begin{equation}
      \lim_{x \rightarrow a} \frac{f(x) - P_2 (x)}{\|x - a\|^2} = 0
    \end{equation}
    We show that this form $H$ is precisely the \hyperref[def:hessian]{Hessian} matrix. 
  \end{enumerate}

  \begin{theorem}[2nd Order Taylor Polynomial]
    \label{thm:2nd-order-taylor-polynomial}
		The second order approximation of a $C^2$-differentiable function $f$ about a point $a$ is 
		\begin{equation}
		  f (x) = f(a) + D f_{a} (x - a) + \frac{1}{2} (x - a)^T H f_{a} (x - a) + O(h^2)
		\end{equation}
		where $D f_{a}$ is the total derivative at $a$ and $H f_{a}$ is the \hyperref[def:hessian]{Hessian} matrix at $a$. 
  \end{theorem}

\subsection{Matrix Calculus}

  Now we will take a look at functions that have either an input or output as matrices. Essentially, matrices are also vectors, so there is nothing new here to learn, but having a concrete set of notation is useful. First, note that when we talk about a total derivative $D \mathbf{f}_a$, we can interpret this as a \hyperref[lin-def:linear_map]{linear map} that takes in some small perturbation $\mathbf{h}$ and gives us the result $D \mathbf{f}_a (\mathbf{h})$. In our column-vector setting, this just corresponded to left \hyperref[lin-def:matrix-multiplication]{matrix multiplication}: 
  \begin{equation}
    D \mathbf{f}_a (\mathbf{h}) = D \mathbf{f}_a \mathbf{h}
  \end{equation}

  This is not the case in the matrix setting. Let us compare the following: 
  \begin{enumerate}
    \item The derivative of $f: \mathbb{R} \rightarrow \mathbb{R}$ at $a$ is a linear function $D f_a: \mathbb{R} \to \mathbb{R}$ satisfying $f(a + h) \approx f(a) + D f_a (h) + O(h^2)$. But linearity reduces $D f_a$ to simply a scalar, and so our condition reduces to 
    
    \begin{equation}
      f(a + h) \approx f(a) + D f_a h + O(h^2)
    \end{equation}
    
    \item The derivative of a path function $\mathbf{f}: \mathbb{R} \rightarrow \mathbb{R}^m$ is a linear function $D \mathbf{f}_a : \mathbb{R} \to \mathbb{R}^m$ satisfying $\mathbf{f}(a + h) = \mathbf{f}(a) + D \mathbf{f}_a (h) + O(h^2)$. Linearity implies that $D \mathbf{f}_a$ is a rank-1 linear map, which reduces to it being a \hyperref[lin-def:column-vector]{column vector}, and our condition reduces to 
    \begin{equation}
      \underbrace{\mathbf{f}(a + h)}_{m \times 1} \approx \underbrace{\mathbf{f}(a)}_{m \times 1} + \underbrace{D \mathbf{f}_a}_{m \times 1} \underbrace{h}_{1 \times 1}
    \end{equation}
    
    \item The derivative of a matrix function $\mathbf{F}: \mathbb{R} \rightarrow \mathbb{R}^{m \times n}$ is a linear map $D \mathbf{F}_a: \mathbb{R} \to \mathbb{R}^{m \times n}$ satisfying $\mathbf{F}(a + h) \approx \mathbf{F}(a) + D \mathbf{F}_a (h)$. This time, linearity does not reduce it to simple left-hand matrix multiplication. We could just say that this is a left-hand scalar multiplication, but this doesn't generalize well, so we are stuck with just saying that $D \mathbf{F}_a$ is a linear map. 
    \begin{equation}
      \underbrace{\mathbf{F}(a + h)}_{m \times n} \approx \underbrace{\mathbf{F}(a)}_{m \times n} + \underbrace{D \mathbf{F}_a (h)}_{m \times n}
    \end{equation}
  \end{enumerate}
  Now let us take a look at when we have matrix inputs. 
  \begin{enumerate}
    \item The derivative of $f: \mathbb{R}^{m \times n} \to \mathbb{R}$ is a linear function $D f_{\mathbf{A}} : \mathbb{R}^{m \times n} \to \mathbb{R}$ satisfying $f(\mathbf{A} + \mathbf{H}) \approx f(\mathbf{A}) + D f_{\mathbf{A}} (\mathbf{H})$. We could let $D f_{\mathbf{A}}$ be some linear map like $\mathbf{M} \mapsto \mathbf{v}^T \mathbf{M} \mathbf{u}$, where $\mathbf{v}, \mathbf{u}$ is fixed. But in generality, we just have the condition 
    \begin{equation}
      \underbrace{f(\mathbf{A} + \mathbf{H})}_{1 \times 1} \approx \underbrace{f(\mathbf{A})}_{1 \times 1} + \underbrace{D f_{\mathbf{A}} (\mathbf{H})}_{1 \times 1}
    \end{equation}
    
    \item The derivative of $\mathbf{f}: \mathbf{R}^{m \times n} \to \mathbb{R}^d$ is some linear map $D \mathbf{f}_\mathbf{A} : \mathbf{R}^{m \times n} \to \mathbb{R}^d$ satisfying $\mathbf{f} (\mathbf{A} + \mathbf{H}) \approx \mathbf{f}(\mathbf{A}) + D \mathbf{f}_\mathbf{A} (\mathbf{H})$. Again, we could construct some form that would give us a linear map in terms of some matrix multiplication, but in generality, we have the condition 
    \begin{equation}
      \underbrace{\mathbf{f} (\mathbf{A} + \mathbf{H})}_{d \times 1} \approx \underbrace{\mathbf{f}(\mathbf{A})}_{d \times 1} + \underbrace{D \mathbf{f}_\mathbf{A} (\mathbf{H})}_{d \times 1}
    \end{equation}
  \end{enumerate}

  Now we present some theorems on basic differentiation. Proving these just requires us to expand the function and compute the derivatives component-wise. 

  \begin{theorem}[Derivative of Affine Map]
    \label{thm:derivative-of-affine-map}
  Given $f: \mathbb{R}^n \to \mathbb{R}^m$ defined $f(x) = \mathbf{A} x + \mathbf{b}$ (where $\mathbf{A}, \mathbf{b}$ is not dependent on $x$), its derivative is 
  \begin{equation}
    D \mathbf{f} = \mathbf{A}
  \end{equation}
  \end{theorem}

  \begin{theorem}[Derivative of a Bilinear Form]
    \label{thm:derivative-of-a-bilinear-form}
  Given the scalar $\alpha$ defined by 
  \begin{equation}
    \alpha = \mathbf{y}^T \mathbf{A} x
  \end{equation}
  where $\mathbf{y} \in \mathbb{R}^{m \times 1}, \mathbf{A} \in \mathbb{R}^{m \times n}$, and $x \in \mathbb{R}^{n \times 1}$, then 
  \begin{equation}
    \frac{\partial \alpha}{\partial x} = \mathbf{y}^T \mathbf{A} : \mathbb{R}^n \to \mathbb{R}
  \end{equation}
  and 
  \begin{equation}
    \frac{\partial \alpha}{\partial \mathbf{y}} = x^T \mathbf{A}^T : \mathbb{R}^m \to \mathbb{R}
  \end{equation}
  Rewritten in the total derivative notation, we can interpret $\alpha$ as a function of both $x$ and $\mathbf{y}$ and write 
  \begin{equation}
    D \alpha_{(x, \mathbf{y})} = \begin{pmatrix} \mathbf{y}^T \mathbf{A} \\ x^T \mathbf{A}^T \end{pmatrix} : \mathbb{R}^{n + m} \to \mathbb{R}
  \end{equation}
  \end{theorem}

  \begin{theorem}[Derivative of a Quadratic Form]
    \label{thm:derivative-of-a-quadratic-form}
    Given the scalar $\alpha$ defined by the quadratic form 
    \begin{equation}
      \alpha = x^T \mathbf{A}x 
    \end{equation}
    where $x \in \mathbb{R}^{n \times 1}$ and $\mathbf{A} \in \mathbb{R}^{n \times n}$, then 
    \begin{equation}
      \frac{\partial \alpha}{\partial x} = x^T \big( \mathbf{A} + \mathbf{A}^T \big)
    \end{equation}
    or in the total derivative notation, 
    \begin{equation}
      D \alpha_\mathbf{a} = \mathbf{a}^T \big( \mathbf{A} + \mathbf{A}^T \big) : \mathbb{R}^n \to \mathbb{R}
    \end{equation}
  \end{theorem}

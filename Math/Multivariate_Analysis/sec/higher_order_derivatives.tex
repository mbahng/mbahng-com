\section{Higher Order Derivatives} 

  So far, we have talked about derivatives as \hyperref[lin-def:linear_map]{linear maps}. However, this language is a bit too restrictive because it doesn't allow us to represent the derivatives of higher order using matrices. It turns out that these higher order derivatives are tensors. 

\subsection{Second Order Derivatives}

	So given differentiable $f: \mathbb{R}^n \to \mathbb{R}^m$, the first derivative is the linear map $Df(a) \in \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger$ satisfying 
	\begin{equation}
	  f(a + h) = f(a) + Df(a) (h) + o(\|h\|)
	\end{equation} 
	Now if we claim that the derivative function $Df: \mathbb{R}^n \to \mathbb{R}^m \otimes (\mathbb{R}^n)^{\dagger}, a \mapsto Df(a)$ is differentiable, then we can take its derivative to be the linear map\footnote{By the canonical identification $\mathcal{L}(\mathbb{R}^n, \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger) \cong \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger \otimes (\mathbb{R}^n)^\dagger$. } 
	\begin{equation}
		D^2 f = D(Df): \mathbb{R}^n \to \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger \otimes (\mathbb{R}^n)^\dagger, \quad a \mapsto D(Df)(a)
	\end{equation}
	satisfying\footnote{Note that all of the terms below are in $\mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger$.}
	\begin{equation}
		Df(a+h) = Df(a) + D^2 f(a) [h] + o(\|h\|) \iff \lim_{h \to 0} \frac{\|Df(a + h) - Df(a) - D^2 f(a) [h]\|_{\mathrm{op}}}{\|h\|} = 0
	\end{equation}

	\begin{definition}[Second Total/Frechet Derivative]
	  \label{def:second-derivative}
		Let $f: E \subset \mathbb{R}^n \to \mathbb{R}^m$ be differentiable on an open neighborhood of $a \in E$, so that the derivative map $Df: x \mapsto Df(x) \in \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger$ is defined near $a$. $f$ is \textbf{twice differentiable} at $a$ if $Df$ is differentiable at $a$, i.e. it satisfies either of the equivalent conditions. 

		\begin{enumerate}
			\item \textit{Linear Approximation}. There exists a tensor $D^2 f(a) \in \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger \otimes (\mathbb{R}^n)^\dagger$\footnote{Here $D^2 f(a)[h] \in \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger$ denotes the contraction of $h \in \mathbb{R}^n$ with the first $(\mathbb{R}^n)^\dagger$ factor, and $D^2 f(a)[h, k] \in \mathbb{R}^m$ denotes the contraction of $h$ and $k$ with the first and second $(\mathbb{R}^n)^\dagger$ factors, respectively.} such that 
				\begin{equation}
					\lim_{h \to 0} \frac{\| Df(a + h) - Df(a) - D^2 f(a)[h] \|_{\mathrm{op}}}{\| h \|} = 0 
				\end{equation} 

			\item \textit{Fundamental Increment Lemma}. There exists a tensor $D^2 f(a) \in \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger \otimes (\mathbb{R}^n)^\dagger$ such that 
				\begin{equation}
					Df(a + h) = Df(a) + D^2 f(a)[h] + \hyperref[real-def:little-o-notation]{o(\|h\|)}\footnote{That is, there exists a function $\Phi: \mathbb{R}^n \to \mathbb{R}^m \otimes (\mathbb{R}^n)^\dagger$ defined for sufficiently small nonzero $h \in \mathbb{R}^n$ such that $\lim_{h \to 0} \frac{\|\Phi(h)\|}{\|h\|} = 0$. }
				\end{equation}
		\end{enumerate}
		If such a tensor $D^2 f(a)$ exists, then it is unique, and we call this the \textbf{second total derivative}---or \textbf{second Frechet derivative}---of $f$ at $a$. 
	\end{definition}

	If $f: \mathbb{R}^n \to \mathbb{R}$, then the second derivative just reduces to a 2-tensor $D^2 f(a) \in (\mathbb{R}^n)^\dagger \otimes (\mathbb{R}^n)^\dagger$, which can be represented as a matrix using the canonical identification $\mathbb{R}^n \otimes (\mathbb{R}^n)^\dagger \cong (\mathbb{R}^n)^\dagger \otimes (\mathbb{R}^n)^\dagger$. 

  \begin{definition}[Hessian]
    \label{def:hessian}
    Given $f: E \subset \mathbb{R}^n \to \mathbb{R}$ twice-differentiable at $a$, the \textbf{Hessian} of $f$ at $a$ is the matrix realization of the 2nd total derivative $D^2 f(a)$, denoted $Hf(a)$. The entries of the Hessian matrix are precisely the partials. 
    \begin{equation}
      H f(a) \coloneqq \begin{pmatrix} 
				\frac{\partial^2 f}{\partial x_1^2} (a) & \ldots & \frac{\partial^2 f}{\partial x_n \partial x_1} (a) \\
        \vdots & \ddots & \vdots \\
				\frac{\partial^2 f}{\partial x_1 \partial x_n} (a) & \ldots & \frac{\partial^2 f}{\partial x_n^2} (a)
			\end{pmatrix} 
    \end{equation}
  \end{definition}
  \begin{proof}
		We can again look at the \hyperref[thm:action-of-derivative-on-basis-vectors]{action of the derivative on the basis vectors} and apply it in the same way we did to prove the Jacobian. Let us denote\footnote{though the codomain should be in $(\mathbb{R}^n)^\dagger$.} 
		\begin{equation}
			g = Df : \mathbb{R}^n \to \mathbb{R}^n, \quad g(a) = Df(a) \coloneqq \begin{pmatrix} \frac{\partial f}{\partial x_1} \\ \vdots \\ \frac{\partial f}{\partial x_n} \end{pmatrix} 
		\end{equation}
		Then, 
		\begin{equation}
			Dg = \begin{pmatrix} 
				\frac{\partial g_1}{\partial x_1} & \ldots & \frac{\partial g_1}{\partial x_n} \\ 
				\vdots & \ddots & \vdots \\ 
				\frac{\partial g_n}{\partial x_1} & \ldots & \frac{\partial g_n}{\partial x_n} 
			\end{pmatrix} = \begin{pmatrix} 
				\frac{\partial}{\partial x_1} \frac{\partial f}{\partial x_1} (a) & \ldots & \frac{\partial}{\partial x_1} \frac{\partial f}{\partial x_n} (a) \\
        \vdots & \ddots & \vdots \\
				\frac{\partial}{\partial x_n} \frac{\partial f}{\partial x_1} (a) & \ldots & \frac{\partial}{\partial x_n} \frac{\partial^2 f}{\partial x_n} (a)
			\end{pmatrix} 
		\end{equation}
  \end{proof}

	So, we can see that the \hyperref[def:hessian]{Hessian} matrix must be symmetric for a twice-differentiable function. 

  \begin{theorem}[Clairut's Theorem]
    \label{thm:clairuts-theorem}
		Given $f: E \subset \mathbb{R}^n \to \mathbb{R}$ is $C^2$ at point $a \in E$, its second iterated partials are equal. 
    \begin{equation}
			\frac{\partial^2 f}{\partial x_i \partial x_j} (a) = \frac{\partial^2 f}{\partial x_j \partial x_i} (a), \quad \forall 1 \leq i = j \leq n
    \end{equation}
  \end{theorem}
  \begin{proof}
    For clarity, denote $x_i, x_j$ as $x, y$ and ignore the rest of the variables. Then, the \hyperref[def:partial-derivative]{partial derivatives} $\partial_{x y} f$ and $\partial_{y x} f$ at a point $(x_0, y_0)$ can be expressed as double limits: 
    \begin{equation}
      \partial_{x y} f (x_0, y_0) = \lim_{y \rightarrow y_0} \frac{\partial_x f (x_0, y) - \partial_x f (x_0, y_0)}{y - y_0}
    \end{equation}
    where $\partial_x f: D \subset \mathbb{R}^n \to \mathbb{R}$. We can use the two limit definitions of partial derivatives
    \begin{equation}
      \partial_x f (x_0, y) = \lim_{x \rightarrow x_0} \frac{f(x, y) - f(x_0, y)}{x-x_0} \text{ and } \partial_x f (x_0, y_0) = \lim_{x \rightarrow x_0} \frac{f(x, y_0) - f(x_0, y_0)}{x-x_0}
    \end{equation}
    and substitute them to get the two partials
    \begin{align}
      \partial_{xy} f (x_0, y_0) & = \lim_{y \rightarrow y_0} \frac{ \lim_{x \rightarrow x_0} \frac{f(x, y) - f(x_0, y)}{x-x_0} - \lim_{y \rightarrow y_0} \frac{f(x, y_0) - f(x_0, y_0)}{x-x_0}}{y - y_0} \\
      & =  \lim_{y \rightarrow y_0} \lim_{x \rightarrow x_0} \bigg( \frac{f(x, y) - f(x_0, y) - f(x, y_0) + f(x_0, y_0)}{(x - x_0) (y - y_0)} \bigg) \\
      \partial_{yx} f (x_0, y_0) & = \lim_{x \rightarrow x_0} \frac{ \lim_{y \rightarrow y_0} \frac{f(x, y) - f(x, y_0)}{y-y_0} - \lim_{y \rightarrow y_0} \frac{f(x_0, y) - f(x_0, y_0)}{y-y_0}}{x - x_0} \\
      & = \lim_{x \rightarrow x_0} \lim_{y \rightarrow y_0} \bigg( \frac{f(x, y) - f(x, y_0) - f(x_0, y) + f(x_0, y_0)}{(y-y_0) (x-x_0)} \bigg)
    \end{align}
    Now invoking our assumption that $f$ is $C^2$, the two limits, which approach $(x_0, y_0)$ along different paths, both exist and are equal to 
    \begin{equation}
      \lim_{(x, y) \rightarrow (x_0, y_0)} \frac{f(x, y) - f(x_0, y) - f(x, y_0) + f(x_0, y_0)}{(x - x_0) (y - y_0)}
    \end{equation}
    and therefore $\partial_{x y} f = \partial_{y x} f$. 
  \end{proof}

	Once we have computed the \hyperref[def:hessian]{Hessian}, let's take the eigendecomposition of it. Since $H f(a)$ is a real symmetric matrix, by the spectral theorem, it will have $n$ real eigenvalues $\lambda_1, \ldots, \lambda_n$ (in descending values) and corresponding orthonormal eigenvectors $\mathbf{v}_1, \ldots, \mathbf{v}_n$. Now given the gradient $\nabla f(a)$ at $a$, we can approximate $\nabla f(a + h)$ at $a + h$ using its total derivative as 
	\begin{equation}
  	\nabla f(a + h) \approx \nabla f(a) + D \nabla f(a) h = \nabla f(a) + H f(a) h
	\end{equation}
  The eigenvalues of $H f(a)$ will tell us how "fast" the gradient changes at $a$. That is, given a small displacement vector $h$, we can take an orthonormal decomposition of it in the form 
	\begin{equation}
  	h = \sum_{i} h_i \mathbf{v}_i
	\end{equation}
  and now the approximate gradient can be written as 
	\begin{equation}
  	\nabla f (a + h) = \nabla f(a) + \sum_i h_i \lambda_i \mathbf{v}_i
	\end{equation}

  Therefore, bigger $\lambda_i$'s will contribute to a greater change in $f(a)$, and smaller ones will contribute less. We can use this information to speed up convergence by scaling along different axes of $h$ when sampling. 

	In order to determine whether a \hyperref[real-def:critical-point]{critical point} $a$ is a relative maximum, minimum, or neither, we use the \hyperref[real-thm:second-der-test]{second derivative test}. There is an exact analogue for the multivariate case. 

  \begin{theorem}[2nd Derivative Test]
    \label{thm:2nd-derivative-test}
    Let $a$ be a \hyperref[real-def:critical-point]{critical point} of $C^2$ function $f: E \subset \mathbb{R}^n \to \mathbb{R}$. That is, $D f(a) = \mathbf{0}$. Then, 
    \begin{enumerate}
      \item $a$ is a strict local minimum if $H f(a)$ is positive definite.
      \item $a$ is a strict local maximum if $H f(a)$ is negative definite. 
			\item $a$ is a \textit{saddle point} (neither a local minimum nor a local maximum) if $H f(a)$ is indefinite, i.e. it has both a positive and a negative eigenvalue.
    \end{enumerate}
    If $H f(a)$ is only semidefinite with an eigenvalue $0$, the test is inconclusive. 
  \end{theorem}
  \begin{proof}
    Since $f \in C^2$, the \hyperref[thm:2nd-order-taylor-polynomial]{second order Taylor polynomial} together with $Df(a) = \mathbf{0}$ gives 
    \begin{equation}
      f(a + h) - f(a) = \frac{1}{2} h^T H f(a) h + \varepsilon(h), \qquad \varepsilon(h) = o(\|h\|^2) \text{ as } h \to 0
    \end{equation}
    By \hyperref[thm:clairuts-theorem]{Clairaut's theorem}, $H f(a)$ is symmetric, so by the spectral theorem it has real eigenvalues $\lambda_1 \geq \ldots \geq \lambda_n$ with orthonormal eigenvectors $\mathbf{v}_1, \ldots, \mathbf{v}_n$. Expanding $h = \sum_i h_i \mathbf{v}_i$ gives $h^T H f(a) h = \sum_i \lambda_i h_i^2$, and so 
    \begin{equation}
      \lambda_n \|h\|^2 \leq h^T H f(a) h \leq \lambda_1 \|h\|^2
    \end{equation}
    \begin{enumerate}
      \item Suppose $H f(a)$ is positive definite, so $\lambda_n > 0$. Since $\varepsilon(h) = o(\|h\|^2)$, there is a $\delta > 0$ such that $|\varepsilon(h)| \leq \frac{\lambda_n}{4} \|h\|^2$ for $\|h\| < \delta$. Then for $0 < \|h\| < \delta$, 
      \begin{equation}
        f(a + h) - f(a) \geq \frac{\lambda_n}{2} \|h\|^2 - \frac{\lambda_n}{4} \|h\|^2 = \frac{\lambda_n}{4} \|h\|^2 > 0
      \end{equation}
      so $a$ is a strict local minimum. 

      \item Suppose $H f(a)$ is negative definite. Then $-f$ is $C^2$ with $D(-f)(a) = \mathbf{0}$ and $H(-f)(a) = -H f(a)$ positive definite, so $a$ is a strict local minimum of $-f$ by (1), i.e. a strict local maximum of $f$. 

      \item Suppose $\lambda_1 > 0 > \lambda_n$. Along the eigenvector $\mathbf{v}_1$, 
      \begin{equation}
        f(a + t \mathbf{v}_1) - f(a) = \frac{\lambda_1}{2} t^2 + \varepsilon(t \mathbf{v}_1) = t^2 \bigg( \frac{\lambda_1}{2} + \frac{\varepsilon(t \mathbf{v}_1)}{t^2} \bigg) > 0
      \end{equation}
      for all small $t \neq 0$, since the last term tends to $0$. Similarly $f(a + t \mathbf{v}_n) - f(a) < 0$ for all small $t \neq 0$. So every neighborhood of $a$ contains points where $f$ is larger and points where $f$ is smaller than $f(a)$, and $a$ is neither a local minimum nor a local maximum. 
    \end{enumerate}
    Finally, the semidefinite case is inconclusive since $f(x) = x^4$ and $f(x) = -x^4$ both have a critical point at $0$ with $f^{\prime\prime}(0) = 0$, but the first is a minimum and the second is a maximum. 
  \end{proof}
	\begin{proof}
	  
	\end{proof}

  Visually, this makes sense since given a \hyperref[real-def:critical-point]{critical point} $a$, the derivative matrix would be $0$, meaning that the 2nd degree Taylor expansion of $f$ near $a$ would be in form
	\begin{equation}
  	f(x) \approx f(a) + \frac{1}{2} (x - a)^T H f(a) (x - a)
	\end{equation}

  If $H f(a)$ is positive definite, then by definition $\frac{1}{2} (x - a)^T H f(a) (x - a) > 0$ for all $x$ near $a$, and so $f$ would increase in every direction within the neighborhood of $a$. The logic follows similarly for negative definite matrix $H f(a)$. If $H f(a)$ is not positive nor negative definite, then $\frac{1}{2} (x - a)^T H f(a) (x - a)$ could be positive or negative, depending on which direction vector $h = x - a$ we choose for computing the \hyperref[def:directional-derivative]{directional derivative}. Therefore, $f$ will increase for certain $h$ and decrease for other $h$. 

\subsection{Convexity}

	Recall what a \hyperref[real-def:convex_functions]{convex function} is in univariate analysis. We can extend this to the multivariate case. 

	\begin{definition}[Convex Set]
	  \label{def:convex-set}
	  A set $D \subset \mathbb{R}^n$ is \textbf{convex} if for all $x, y \in D$, 
		\begin{equation}
			tx + (1 - t) y \in D \quad \forall t \in [0, 1]
		\end{equation}
		That is, $D$ contains the straight line connecting $x$ and $y$. 
	\end{definition}

	\begin{definition}[Convex Function]
		\label{def:convex_function}
		Let $D \subset \mathbb{R}^n$ be convex.\footnote{Note that we require the domain to be convex.} $f: D \to \mathbb{R}$ is \textbf{convex} over $D$ if it satisfies any of the equivalent definitions. 
		\begin{enumerate}
		  \item \textit{General}. 
				\begin{equation}
					f\big( t x + (1 - t) y \big) \leq t f(x) + (1 - t) f(y) \quad \forall t \in [0, 1], \forall x, y \in D
				\end{equation}

			\item \textit{First Derivative}. If $f$ is differentiable, then\footnote{This theorem for $C^1$ functions is quite intuitive, since for a convex function, the tangent plane on its graph must never be "above" the graph. In other words, the first order approximation must be a global underestimate of $f$. }
				\begin{equation}
					f(x) + \langle \nabla f(x), y - x \rangle = f(x) + D f(x) (y - x) \leq f(y) \quad \forall x, y \in D
				\end{equation}

			\item \textit{Second Derivative}. If $f$ is twice differentiable, then 
				\begin{equation}
				  Hf(x) \succeq 0 \quad \forall x \in D^\circ
				\end{equation}
				where $D^\circ$ is the \hyperref[pst-def:interior]{interior} of $D$.\footnote{For \hyperref[lin-def:symmetric-matrices]{symmetric matrices} $A, B$, $A \succeq B$ if $A - B$ is positive semidefinite.}
		\end{enumerate}
		If any of the inequalities are strict, then $f$ is \textbf{strictly convex}, and $f$ is called \textbf{(strictly) concave} if $-f$ is (strictly) convex. 
	\end{definition}

	Clearly, strict convexity implies convexity. 

	\begin{definition}[Coercive Function]
		\label{def:coercive}
	  A function $f: \mathbb{R}^d \to \mathbb{R}$ is \textbf{coercive} if 
		\begin{equation}
			\lim_{\|x\| \to +\infty} f(x) = +\infty
		\end{equation}
	\end{definition}

	\begin{definition}[Co-Coercive Function]
		\label{def:cocoercive}
		A function $F: \mathbb{R}^d \to \mathbb{R}^d$ is \textbf{$\beta$-co-coercive} if for some $\beta > 0$, 
		\begin{equation}
			\langle F(x) - F(w), x - w \rangle \geq \beta \|F(x) - F(w)\|^2
		\end{equation}
		holds for all $x, w \in \mathbb{R}^d$. 
	\end{definition}

	\begin{definition}[Strongly Monotone Function]
		\label{def:strongly_monotone}
		A function $F: \mathbb{R}^d \to \mathbb{R}^d$ is \textbf{$c$-strongly monotone} if for some $c > 0$, 
		\begin{equation}
			\langle F(x) - F(y), x - y \rangle \geq c \|x - y\|^2
		\end{equation}
		holds for all $x, y \in \mathbb{R}^d$. 
	\end{definition}

	\begin{definition}[$\mu$-Strongly Convex]
	  \label{def:mu-strongly-convex}
		A differentiable function $f: \mathbb{R}^d \to \mathbb{R}$ is \textbf{$\mu$-strongly convex} if it satisfies the equivalent conditions below.\footnote{$f$ is just convex if $\mu = 0$.} 
		\begin{enumerate}
		  \item \textit{Strong Convexity}. For all $w, x \in \mathbb{R}^d$, 
				\begin{equation}
					f(x) \geq f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - w\|^2
				\end{equation}
				Note that $f$ is convex when $\mu = 0$. 

			\item \textit{Strongly Monotone Gradient}. The gradient $\nabla f$ is \hyperref[def:strongly_monotone]{strongly monotone}, i.e. for all $x, w \in \mathbb{R}^d$, 
				\begin{equation}
				  \langle \nabla f(x) - \nabla f(w), x - w \rangle \geq \mu \|x - w\|^2
				\end{equation}

			\item \textit{Second Derivative}. If $f$ is $C^2$, then 
				\begin{equation}
					 Hf(x) \succeq \mu I 
				\end{equation}
		\end{enumerate}
	\end{definition} 
	\begin{proof}
	  We prove directions. 
		\begin{enumerate}
		  \item $(1 \rightarrow 2)$. From strong convexity, we can see that 
				\begin{align}
					\langle \nabla f(w), x - w \rangle & \leq f(x) - f(w) - \frac{\mu}{2} \|x - w\|^2 \\ 
					\langle \nabla f(x), w - x \rangle & \leq f(w) - f(x) - \frac{\mu}{2} \|x - w\|^2
				\end{align}
				Therefore, 
				\begin{align}
					\langle \nabla f(x) - \nabla f(w), x - w \rangle 
						& = \langle \nabla f(x), x - w \rangle - \langle \nabla f(w), x - w \rangle \\ 
						& \geq \big\{ -f(w) + f(x) + \frac{\mu}{2} \|x - w\|^2 \big\} + \big\{ - f(x) + f(w) + \frac{\mu}{2} \| x - w\|^2 \big\} \\ 
						& = \mu \|x - w\|^2
				\end{align}

			\item $(2 \rightarrow 1)$. We define the straight line path 
				\begin{equation}
					\varphi: [0, 1] \to \mathbb{R}, \quad \varphi(t) = f\big( w + t(x - w) \big)
				\end{equation}
				Then, by FTC
				\begin{align}
					f(x) = \varphi(1) & = \varphi(0) + \int_0^1 \varphi^\prime (t) \,dt \\ 
														& = \varphi(0) + \int_0^1 \langle \nabla f(w + t(x - w)), x - w \rangle \,dt  \\ 
														& \geq  f(w) + \int_0^1 \mu t \|x - w\|^2 + \langle \nabla f(w), x - w \rangle \,dt \\
														& = f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - w\|^2
				\end{align}
				where for the inequality, we invoke strong monotonicity and have 
				\begin{equation}
				  \langle \nabla f(w + t(x - w)) - \nabla f(w), t (x - w) \rangle \geq \mu t^2 \|x - w\|^2
				\end{equation}
				After dividing by $t$ and rearranging, we have 
				\begin{equation}
				  \langle \nabla f(w + t(x - w)), x - w \rangle \geq \mu t \|x - w\|^2 + \langle \nabla f(w), x - w \rangle
				\end{equation}

		  \item $(1 \rightarrow 3)$. Given $x, w \in \mathbb{R}^d$, let $g(t) = f(w + t(x - w))$, which is $C^2$. Take the second order Taylor expansion and invoke strong convexity to get the inequality 
				\begin{align}
					g(t) & = f(w) + t \langle \nabla f(w), x - w \rangle + \frac{1}{2} t^2 \langle \nabla^2 f(w) (x - w), x - w \rangle + o(\|t(x - w)\|^2) \\ 
							 & \geq f(w) + \langle \nabla f(w), t (x - w) \rangle + \frac{\mu t^2}{2} \|x - w\|^2
				\end{align}
				Cleaning this up gives 
				\begin{equation}
					\langle \nabla^2 f(w) (x - w), x - w \rangle + \frac{o(t^2 \|x - w\|^2)}{t^2} \geq \mu \|x - w\|^2
				\end{equation}
				This holds for all $t \in [0, 1]$, so by letting $t \to 0$, we have 
				\begin{equation}
				  \langle \nabla^2 f(w) (x - w), x - w \rangle \geq \mu \|x - w\|^2 \implies Hf(w) \succeq \mu I
				\end{equation}

		  \item $(3 \rightarrow 1)$. We can see that 
				\begin{align}
					g^\prime (t) & = \langle \nabla f(w + t(x - w)), x - w \rangle \\ 
					g^{\prime\prime} (t) & = (x - w)^T H f(w + t(x - w)) (x - w) \\ 
															 & \geq \mu (x - w)^T (x - w) = \mu \|x - w\|^2
				\end{align}
				So, using \hyperref[real-thm:integral_form_remainder]{Taylor's expansion with the integral form of the remainder}, we have 
				\begin{align}
					f(x) = g(1) & = g(0) + g^\prime (0) (1 - 0) + \int_0^1 g^{\prime\prime} (t) (1 - t) \,dt \\ 
											& \geq g(0) + g^\prime (0) + \mu \|x - w\|^2 \int_0^1 (1 - t) \,dt \\ 
											& = f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - w\|^2
				\end{align}
		\end{enumerate}
	\end{proof}

	\begin{lemma}[Strong Convexity Implies Strict Convexity]
		If $f: D \subset \mathbb{R}^n \to \mathbb{R}$ is $\mu$-strongly convex for $\mu > 0$, then $f$ is strictly convex. 
	\end{lemma}
	\begin{proof}
	  We have 
		\begin{equation}
			f(y) \geq f(x) + \langle \nabla f(x), y - x \rangle + \frac{\mu}{2} \| y - x\|^2 > f(x) + \langle \nabla f(x), y - x \rangle
		\end{equation}
	\end{proof}

	This is a first order description of \hyperref[def:mu-strongly-convex]{$\mu$-strongly convex functions}. 

	\begin{example}[Strongly Convex and non-Strongly Convex Functions]
	  Here are some examples. For twice-differentiable $f$ on $\mathbb{R}^n$, $\mu$-strong convexity is equivalent to $\nabla^2 f(x) \succeq \mu I$ for all $x$, which makes most cases easy to check. 
		\begin{enumerate}
			\item \textit{Strongly convex}. $f(x) = \frac{1}{2} x^T A x - b^T x$ with $A \succ 0$ is $\lambda_{\min}$-strongly convex since it is $C^2$ and 
				\begin{equation}
					Hf(x) = A \succeq \lambda_{\min} I
				\end{equation}

		  \item \textit{Strongly convex}. $f(x) = \cosh(x) = \frac{e^x + e^{-x}}{2}$ is $1$-strongly convex since 
				\begin{equation}
				  f^{\prime\prime}(x) = \cosh(x) \geq 1
				\end{equation}

		  \item \textit{Convex but not strongly convex (flat directions)}. $f(w) = \frac{1}{2}\|Xw - y\|^2$ has Hessian $X^T X$, which is singular if $\operatorname{rank} X < d$ (e.g.\ $n < d$), so $\lambda_{\min}(X^T X) = 0$. Linear functions $f(x) = a^T x$ are the extreme case, with $\nabla^2 f = 0$. 

		  \item \textit{Convex but not strongly convex (vanishing curvature)}. $f(x) = x^4$ is strictly convex with a unique minimizer, but it is not strongly convex. Indeed, 
				\begin{equation}
					f^{\prime\prime}(x) = 12x^2 \to 0 \text{ as } x \to 0
				\end{equation}
				So strict convexity is strictly weaker than strong convexity. 


			\item \textit{Convex but not strongly convex (vanishing curvature)}. $f(x) = e^x$ is strictly convex but is not strongly convex. Indeed, 
				\begin{equation}
					f^{\prime\prime}(x) = e^x \to 0 \text{ as } x \to -\infty
				\end{equation}

			\item \textit{Convex but not strongly convex}. $f(x) = \log(1 + e^{-x})$ is strictly convex but not strongly convex. Indeed, 
				\begin{equation}
					f^{\prime\prime}(x) = \frac{e^{x}}{(1 + e^{x})^2} \to 0 \text{ as } |x| \to \infty
				\end{equation}

		  \item \textit{Not convex}. $f(x) = x^2 + 3\sin^2 x$ has $f^{\prime\prime}(x) = 2 + 6\cos 2x$, which is as low as $-4$, so it is not even convex. 
		\end{enumerate}
	\end{example}

\subsection{k-Times Continuously Differentiable Functions}

	Continuing our derivations for second order derivatives, we can recursively define the $k$th order derivative. Recall that a linear map $\mathbb{R}^n \to V$ is an element of $V \otimes (\mathbb{R}^n)^\dagger$, where $(\mathbb{R}^n)^\dagger$ is the \hyperref[lin-def:dual-space]{dual space} of $\mathbb{R}^n$. Set $D^0 f \coloneqq f$, which takes values in $\mathbb{R}^m$. Given that $f: E \subset \mathbb{R}^n \to \mathbb{R}^m$ is $(k-1)$-times differentiable on a neighborhood of $a \in E$, we have a function 
	\begin{equation}
		D^{k-1} f: x \mapsto D^{k-1} f(x) \in \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes (k-1)}
	\end{equation}
	defined near $a$. If $D^{k-1} f$ is differentiable at $a$, then its derivative is a linear map from $\mathbb{R}^n$ into $\mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes (k-1)}$, i.e. a tensor $D^k f(a) = D(D^{k-1}f)(a) \in \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes k}$.\footnote{Generally, multivariate calculus courses do not go into \hyperref[lin-def:tensor-algebra]{tensor algebra}, so these higher order derivatives are omitted; you only work with the \hyperref[def:hessian]{Hessian} for scalar-valued functions. But we won't be scared off and will present this in full generality, and this gives us two major benefits. First, we can compute higher order derivative as we have seen. Second, we can compute functions that also input or output tensors, such as matrix-valued functions. This knowledge becomes very useful for optimization, e.g. when you optimize a neural network with respect to matrices (in fully-connected layers) and 3-tensors (e.g. \hyperref[real-thm:convolution]{convolutions}) and is a must-know for building autograd libraries. } 

	\begin{definition}[$k$th Total/Frechet Derivative]
	  \label{def:kth-frechet-derivative}
		Let $f: E \subset \mathbb{R}^n \to \mathbb{R}^m$ with $E$ open, $a \in E$, and $k \geq 1$. Suppose $f$ is $(k-1)$-times differentiable on an open neighborhood of $a$, so that the map $D^{k-1} f: x \mapsto D^{k-1} f(x) \in \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes (k-1)}$ is defined near $a$. $f$ is \textbf{$k$-times differentiable} at $a$ if $D^{k-1} f$ is differentiable at $a$, i.e. it satisfies either of the equivalent conditions. 

		\begin{enumerate}
			\item \textit{Linear Approximation}. There exists a tensor $D^k f(a) \in \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes k}$\footnote{Here $D^k f(a)[h] \in \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes (k-1)}$ denotes the contraction of $h \in \mathbb{R}^n$ with the first $(\mathbb{R}^n)^\dagger$ factor, and $D^k f(a)[h_1, \ldots, h_k] \in \mathbb{R}^m$ denotes the contraction of $h_i$ with the $i$th $(\mathbb{R}^n)^\dagger$ factor.} such that 
				\begin{equation}
					\lim_{h \to 0} \frac{\| D^{k-1} f(a + h) - D^{k-1} f(a) - D^k f(a)[h] \|_{\mathrm{op}}}{\| h \|} = 0 
				\end{equation} 

			\item \textit{Fundamental Increment Lemma}. There exists a tensor $D^k f(a) \in \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes k}$ such that 
				\begin{equation}
					D^{k-1} f(a + h) = D^{k-1} f(a) + D^k f(a)[h] + \hyperref[real-def:little-o-notation]{o(\|h\|)}\footnote{That is, there exists a function $\Phi: \mathbb{R}^n \to \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes (k-1)}$ defined for sufficiently small nonzero $h \in \mathbb{R}^n$ such that $\lim_{h \to 0} \frac{\|\Phi(h)\|}{\|h\|} = 0$. }
				\end{equation}
		\end{enumerate}
		If such a tensor $D^k f(a)$ exists, then it is unique, and we call this the \textbf{$k$th total derivative}---or \textbf{$k$th Frechet derivative}---of $f$ at $a$. We say $f$ is $k$-times differentiable on $E$ if this holds at every $a \in E$, and $f$ is $C^k$ if moreover $D^k f: E \to \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes k}$ is continuous. 
	\end{definition}

  Unwinding the recursion, an element $T \in \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes k}$ is a $k$-linear map $(\mathbb{R}^n)^k \to \mathbb{R}^m$: viewing $T$ as a linear map $\mathbb{R}^n \to \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes (k-1)}$, it is identified with the $k$-linear map
  \begin{equation}
    T[h_1, \ldots, h_k] \coloneqq \big( \cdots \big( T[h_1] [h_2] \big) \cdots \big) [h_k]
  \end{equation}
  The norm on $\mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes k}$ is the operator norm $\|T\| = \sup_{\|h_i\| \leq 1} \|T[h_1, \ldots, h_k]\|$ (all norms are equivalent in finite dimensions anyway). Note that the points $x + h$ vary in the \textit{base point} of the map $D^{k-1} f$, while the vectors $h_1, \ldots, h_k$ are the \textit{inputs} of the multilinear form $D^k f(x)$; these two roles are not to be confused.

  The Fréchet derivatives above are clean, but to actually compute with them we need to express them in coordinates. Just like the first derivative $Df(a)$ has entries given by the \hyperref[def:partial-derivative]{partial derivatives} $\partial_{x_i} f(a)$, the entries of $D^k f(a)$ will be given by partial derivatives taken $k$ times in a row. \hyperref[def:directional-derivative]{Directional derivatives} lose meaning in these higher order regimes, and we just work with iterated partials. 

  \begin{definition}[$k$th Partial Derivative]
    \label{def:kth-partial-derivative}
    Let $E \subset \mathbb{R}^n$ be open and $f: E \to \mathbb{R}$. Given a sequence of indices $i_1, \ldots, i_k \in \{1, \ldots, n\}$, the \textbf{$k$th iterated partial derivative} of $f$ is defined recursively by
    \begin{equation}
      \partial_{x_{i_1} \ldots x_{i_k}} f \coloneqq \partial_{x_{i_k}} \big( \partial_{x_{i_1} \ldots x_{i_{k-1}}} f \big)
    \end{equation}
    at every point of $E$ where the right hand side exists, with the base case being the usual \hyperref[def:partial-derivative]{partial derivative} $\partial_{x_i} f$.
  \end{definition}

  Note the order convention: we read the subscripts from left to right, so $\partial_{x_{i_1} \ldots x_{i_k}} f$ means that we differentiate with respect to $x_{i_1}$ \textit{first} and $x_{i_k}$ \textit{last}. This is consistent with how we wrote $\partial_{xy} f$ in the proof of \hyperref[thm:clairuts-theorem]{Clairaut's theorem}, and with the entries of the \hyperref[def:hessian]{Hessian}. If the same variable is repeated we use exponents, e.g. $\partial_{x_1}^2 \partial_{x_2} f = \partial_{x_1 x_1 x_2} f$. Since there are $n^k$ sequences of indices, there are $n^k$ iterated partials of order $k$, which is exactly the number of entries in the tensor $D^k f(a) \in \mathbb{R} \otimes ((\mathbb{R}^n)^\dagger)^{\otimes k}$.

  \begin{definition}[$C^k$ Functions]
    \label{def:ck_functions}
    Let $E \subset \mathbb{R}^n$ be open. A function $f: E \to \mathbb{R}$ is a \textbf{$C^k$ function} if it satisfies either of the equivalent definitions: 
    \begin{enumerate}
      \item $f$ is \hyperref[def:kth-frechet-derivative]{$k$-times differentiable} on $E$ and the map $x \in E \mapsto D^k f(x) \in \mathbb{R}^m \otimes ((\mathbb{R}^n)^\dagger)^{\otimes k}$ is continuous. 
      \item All \hyperref[def:kth-partial-derivative]{iterated partial derivatives} $\partial_{x_{i_1} \ldots x_{i_m}} f$ of orders $m = 1, \ldots, k$ exist and are continuous on $E$. 
    \end{enumerate}
    The \hyperref[lin-def:vector-space]{vector space} of all $C^k$ functions is denoted $C^k (E; \mathbb{R})$, or $C^k(E)$. 
  \end{definition}
  \begin{proof}
    We prove by induction on $k$ that (1) and (2) are equivalent, and along the way we show that the entries of $D^k f(x)$ are exactly the iterated partials, namely 
    \begin{equation}
      D^k f(x) [e_{i_1}, \ldots, e_{i_k}] = \partial_{x_{i_k} \ldots x_{i_1}} f (x) 
    \end{equation}
    (the order of the indices is reversed because $D^k f(x) h_1$ is the derivative of $D^{k-1} f$ in the direction $h_1$, so the \textit{first} slot is the \textit{last} partial taken). 

    For $k = 1$, this is exactly the \hyperref[def:continuously-differentiable]{definition of continuously differentiable} together with \hyperref[thm:action-of-derivative-on-basis-vectors]{$Df(x) e_i = \partial_{x_i} f(x)$}, combined with \hyperref[thm:continuous-partials-implies-differentiability]{continuous partials implying differentiability}. Now let $k \geq 2$ and assume that the claim holds for $k - 1$. Write $T \coloneqq ((\mathbb{R}^n)^\dagger)^{\otimes (k-1)}$, so that $D^{k-1} f: E \to T$, and note that by equivalence of norms on the finite dimensional space $T$, a map into $T$ is continuous (resp. differentiable) if and only if all of its entries are. 
    \begin{enumerate}
      \item $(1 \to 2)$. If $f$ is $k$-times differentiable, then $D^{k-1} f$ is differentiable on $E$, hence continuous by \hyperref[thm:differentiability-implies-continuity]{differentiability implies continuity}. So $f$ satisfies (1) for $k-1$, and by the induction hypothesis all iterated partials of order $\leq k-1$ exist and are continuous, with the $(j_1, \ldots, j_{k-1})$th entry of $D^{k-1} f(y)$ equal to $g_J(y) \coloneqq \partial_{x_{j_{k-1}} \ldots x_{j_1}} f(y)$. Since $D^{k-1} f$ is differentiable at $x$, so is each entry $g_J$, and by $Dg(J)(x) e_i = \partial_{x_i} g_J(x)$ the partial $\partial_{x_i} g_J (x) = \partial_{x_{j_{k-1}} \ldots x_{j_1} x_i} f(x)$ exists and equals the $(i, j_1, \ldots, j_{k-1})$th entry of $D^k f(x)$. This entry is continuous in $x$ since $D^k f$ is. As $(i, j_1, \ldots, j_{k-1})$ ranges over all index sequences of length $k$, we get that all $k$th order iterated partials exist and are continuous. 

      \item $(2 \to 1)$. By the induction hypothesis, $f$ is $(k-1)$-times differentiable with $D^{k-1} f$ continuous, whose entries are the functions $g_J$ above. By assumption, each $g_J$ has all partials $\partial_{x_i} g_J = \partial_{x_{j_{k-1}} \ldots x_{j_1} x_i} f$ existing and continuous, so by \hyperref[thm:continuous-partials-implies-differentiability]{continuous partials implying differentiability}, each $g_J$ is differentiable on $E$ with $Dg(J) (x) h = \sum_i \partial_{x_i} g_J(x) h_i$. Therefore $D^{k-1} f$, whose entries are the $g_J$, is differentiable on $E$, i.e. $f$ is $k$-times differentiable, and the entries of $D^k f(x)$ are the continuous functions $\partial_{x_i} g_J(x)$. So $D^k f$ is continuous. 
    \end{enumerate}
  \end{proof}

  In particular, a $C^k$ function is $C^m$ for all $m \leq k$, since condition (2) for $k$ includes condition (2) for $m$. The proof also shows that the $n^k$ iterated partials of order $k$ are precisely the $n^k$ entries of the tensor $D^k f(x)$, so we are free to go back and forth between the tensor and coordinate pictures. 

  Whenever we want to get the $k$th iterated \hyperref[def:partial-derivative]{partial derivative} of $f$, we will assume that $f \in C^k$. Again, this is overkill, but it is conventional since we don't really work with the set of functions with existing partial derivatives. Note that mathematicians throw around the word "smooth" a lot. Usually, it means one of three things: it is of class $C^1$, $C^\infty$, or $C^k$ where $k$ is however high it needs to be to satisfy our assumptions. For example, if I say let us differentiate smooth $f$ two times, then I am assuming that $f \in C^2 (\mathbb{R}^n)$. 

  Visualizing $C^k$-functions is easy for low orders. A $C^0$ function produces a graph that isn't "ripped" or "punctured," since this is exactly what a discontinuity would look like. A $C^1$ function requires the surface to be smooth in such a way that there is a well defined affine tangent subspace at every point. This means that there cannot be any sharp "points" or "edges" on the graph since a tangent subspace cannot be well defined. 

  So there are $n^k$ iterated partials of order $k$. But \hyperref[thm:clairuts-theorem]{Clairaut's theorem} tells us that for $k = 2$ many of them coincide, so we expect that for $C^k$ functions, the order in which we differentiate doesn't matter at all.

  \begin{corollary}[Symmetry in $k$th Iterated Partials]
    \label{thm:symmetry-in-kth-iterated-partials}
    Given $f \in C^k(E)$, its $k$th iterated partials are invariant under permuting the order of differentiation. That is, for any indices $i_1, \ldots, i_k \in \{1, \ldots, n\}$ and any permutation $\sigma \in S_k$,
    \begin{equation}
      \partial_{x_{i_{\sigma(1)}} x_{i_{\sigma(2)}} \ldots x_{i_{\sigma(k)}}} f = \partial_{x_{i_1} x_{i_2} \ldots x_{i_k}} f \text{ on } E
    \end{equation}
    In other words, $D^k f(x)$ is a symmetric tensor, i.e. an element of the \hyperref[lin-def:symmetric-power]{$k$th symmetric power} $\mathrm{Sym}^k \big( (\mathbb{R}^n)^\dagger \big) \otimes \mathbb{R}^m \subset \big( (\mathbb{R}^n)^\dagger \big)^{\otimes k} \otimes \mathbb{R}^m$.
  \end{corollary}
  \begin{proof}
    The case $k = 1$ is trivial, and $k = 2$ is \hyperref[thm:clairuts-theorem]{Clairaut's theorem}. Now let $k \geq 3$. We first reduce to a special type of permutation. Every permutation in $S_k$ is a product of \textit{adjacent transpositions} $\tau_j = (j \;\, j+1)$ for $1 \leq j \leq k - 1$ (this is just the statement that you can sort any list by repeatedly swapping neighbors, e.g. bubble sort). Therefore, it suffices to show that swapping two adjacent indices does not change the partial. That is, for each $j = 1, \ldots, k-1$,
    \begin{equation}
      \partial_{x_{i_1} \ldots x_{i_{j}} x_{i_{j+1}} \ldots x_{i_k}} f = \partial_{x_{i_1} \ldots x_{i_{j+1}} x_{i_{j}} \ldots x_{i_k}} f
    \end{equation}
    since applying this repeatedly for each adjacent transposition in the product gives us the claim for general $\sigma$.

    To show this, let $g \coloneqq \partial_{x_{i_1} \ldots x_{i_{j-1}}} f$ (and $g \coloneqq f$ if $j = 1$). Since $f \in C^k$, the function $g$ has continuous partials up to order $k - (j-1) \geq 2$, since $j \leq k-1$. In particular, $g \in C^2(E)$. By \hyperref[thm:clairuts-theorem]{Clairaut's theorem} applied to $g$ at every point of $E$,
    \begin{equation}
      \partial_{x_{i_j} x_{i_{j+1}}} g = \partial_{x_{i_{j+1}} x_{i_j}} g \text{ on } E
    \end{equation}
    By definition, the left hand side is $\partial_{x_{i_1} \ldots x_{i_{j+1}}} f$ and the right hand side is $\partial_{x_{i_1} \ldots x_{i_{j-1}} x_{i_{j+1}} x_{i_j}} f$. These are the same function on $E$, so taking the remaining partials $\partial_{x_{i_{j+2}}}, \ldots, \partial_{x_{i_k}}$ (in this order) of both sides gives the same result, which is the desired equality.
  \end{proof}

  An immediate consequence is that an iterated partial of a $C^k$ function does not depend on the sequence $(i_1, \ldots, i_k)$ itself, but only on \textit{how many times} each variable appears in it. This lets us use much more compact notation.

  \begin{definition}[Multi-Index Notation]
    \label{def:multi-index}
    A \textbf{multi-index} is a tuple $\alpha = (\alpha_1, \ldots, \alpha_n) \in \mathbb{N}^n$ (where $\mathbb{N}$ includes $0$). We define its
    \begin{enumerate}
      \item \textit{order} $|\alpha| \coloneqq \alpha_1 + \ldots + \alpha_n$,
      \item \textit{factorial} $\alpha! \coloneqq \alpha_1! \, \alpha_2! \cdots \alpha_n!$,
      \item \textit{power} $h^\alpha \coloneqq h_1^{\alpha_1} h_2^{\alpha_2} \cdots h_n^{\alpha_n}$ for $h = (h_1, \ldots, h_n) \in \mathbb{R}^n$, and
      \item \textit{partial derivative} of a $C^{|\alpha|}$ function $f$ by
      \begin{equation}
        \partial^\alpha f \coloneqq \partial_{x_1}^{\alpha_1} \partial_{x_2}^{\alpha_2} \cdots \partial_{x_n}^{\alpha_n} f
      \end{equation}
      which means to differentiate $\alpha_1$ times in $x_1$, then $\alpha_2$ times in $x_2$, and so on.
    \end{enumerate}
  \end{definition}

  The point of the \hyperref[thm:symmetry-in-kth-iterated-partials]{symmetry corollary} is that this isn't an arbitrary choice of ordering. If $f \in C^k$ and $(i_1, \ldots, i_k)$ is \textit{any} sequence in which the index $m$ appears exactly $\alpha_m$ times (so $|\alpha| = k$), then we can permute this sequence into the sorted one $(1, \ldots, 1, 2, \ldots, 2, \ldots, n, \ldots, n)$, and therefore
  \begin{equation}
    \partial_{x_{i_1} \ldots x_{i_k}} f = \partial^\alpha f
  \end{equation}
  So the $n^k$ iterated partials of order $k$ collapse to $\binom{n + k - 1}{k}$ distinct ones, one for each multi-index with $|\alpha| = k$. Conversely, the number of sequences $(i_1, \ldots, i_k)$ that give the same $\alpha$ is the multinomial coefficient
  \begin{equation}
    \frac{k!}{\alpha!} = \frac{k!}{\alpha_1! \cdots \alpha_n!}
  \end{equation}
  since this counts the ways to arrange $\alpha_1$ copies of $1$, $\alpha_2$ copies of $2$, etc. into $k$ slots. This count is exactly the weight that will appear in front of $\partial^\alpha f$ when we expand the $k$th order Taylor term.

	\begin{example}[Derivatives of Polynomial]
    Let $f: \mathbb{R}^2 \to \mathbb{R}$ be the polynomial $f(x_1, x_2) = x_1^2 x_2 + x_2^3$. 
    \begin{enumerate}
      \item \textit{First derivative.} There are $2$ partials, which form the gradient 
      \begin{equation}
        Df(x) = \begin{pmatrix} \partial_{x_1} f & \partial_{x_2} f \end{pmatrix} = \begin{pmatrix} 2 x_1 x_2 & x_1^2 + 3 x_2^2 \end{pmatrix}
      \end{equation}

      \item \textit{Second derivative.} There are $2^2 = 4$ partials, forming the Hessian 
      \begin{equation}
        D^2 f(x) = \begin{pmatrix} \partial_{x_1 x_1} f & \partial_{x_1 x_2} f \\ \partial_{x_2 x_1} f & \partial_{x_2 x_2} f \end{pmatrix} = \begin{pmatrix} 2 x_2 & 2 x_1 \\ 2 x_1 & 6 x_2 \end{pmatrix}
      \end{equation}
      which is symmetric by \hyperref[thm:clairuts-theorem]{Clairaut's theorem}. 

      \item \textit{Third derivative.} There are $2^3 = 8$ partials, which form a $2 \times 2 \times 2$ tensor. Differentiating the Hessian entries once more, we get 
      \begin{align}
        \partial_{x_1 x_1 x_1} f & = 0 & \partial_{x_1 x_1 x_2} f & = 2 & \partial_{x_1 x_2 x_1} f & = 2 & \partial_{x_2 x_1 x_1} f & = 2 \\ 
        \partial_{x_1 x_2 x_2} f & = 0 & \partial_{x_2 x_1 x_2} f & = 0 & \partial_{x_2 x_2 x_1} f & = 0 & \partial_{x_2 x_2 x_2} f & = 6 
      \end{align}
      Since $f$ is a cubic, $D^3 f(x)$ is constant in $x$, and consequently $D^4 f = 0$. 
    \end{enumerate}
    By the \hyperref[thm:symmetry-in-kth-iterated-partials]{symmetry corollary}, only $\binom{4}{3} = 4$ of the $8$ third order partials are distinct, one for each $\alpha \in \{(3,0), (2,1), (1,2), (0,3)\}$: 
    \begin{equation}
      \partial^{(3,0)} f = 0, \quad \partial^{(2,1)} f = 2, \quad \partial^{(1,2)} f = 0, \quad \partial^{(0,3)} f = 6
    \end{equation}
    The multi-index $\alpha = (2, 1)$ has $\alpha! = 2$ and $k!/\alpha! = 3$, and indeed it appears exactly $3$ times in the table above, as $\partial_{x_1 x_1 x_2} f$, $\partial_{x_1 x_2 x_1} f$, and $\partial_{x_2 x_1 x_1} f$. Similarly $(0, 3)$ appears once, while $(3, 0)$ appears once and $(1, 2)$ appears $3$ times, both with value $0$. Summing each value with its multiplicity gives the cubic form 
    \begin{equation}
      D^3 f(x) [h, h, h] = \sum_{|\alpha| = 3} \frac{3!}{\alpha!} \partial^\alpha f \, h^\alpha = 3 \cdot 2 \, h_1^2 h_2 + 6 h_2^3 = 6 h_1^2 h_2 + 6 h_2^3
    \end{equation}
    which is exactly the weight that will appear in the third order Taylor term. 
  \end{example}

	Let's move onto a harder example. 

	\begin{example}[Derivative of Maximum Eigenvalue of Function Hessian]
		Let $f: \mathbb{R}^n \to \mathbb{R}$ be $C^3$. We define the \textit{sharpness} of $f$ at $a \in \mathbb{R}^n$ to be 
		\begin{equation}
			\lambda_{\max}: \mathbb{R}^n \to \mathbb{R}, \quad \lambda_{\max}(a) = \lambda_{\max}(D^2f(a))
		\end{equation}
		We wish to find the derivative of the sharpness. For simplicity of notation, let's denote $H(a) \coloneqq Hf(a)$, with $a^\prime = a + \delta$ for some small $\delta$. We wish to basically bound $\lambda_{\max} (a^\prime) - \lambda_{\max} (a)$. We know that 
		\begin{equation}
			\lambda_{\max} (a) = \max_{\|u\| = 1} u^T H(a) u 
		\end{equation}
		Let $v, v^\prime$ be the top unit eigenvectors of $H(a), H(a^\prime)$, respectively. 
		\begin{enumerate}
		  \item \textit{Lower Bound}. Then, 
				\begin{equation}
					\lambda_{\max} (a^\prime) \geq v^T H(a^\prime) v, \quad \lambda_{\max} (a) = v^T H(a) v
				\end{equation}
				Combining these two, we get 
				\begin{equation}
					\lambda_{\max}(a^\prime) - \lambda_{\max} (a) \geq v^T (H(a^\prime) - H(a)) v
				\end{equation}

			\item \textit{Upper Bound}. Note that  
				\begin{equation}
					\lambda_{\max} (a) \geq v^{\prime T} H(a) v^\prime, \quad \lambda_{\max} (a^\prime) = v^{\prime T} H(a^\prime) v^\prime 
				\end{equation}
				which implies that $\lambda_{\max}(a^\prime) - \lambda_{\max} (a) \leq v^{\prime T} (H(a^\prime) - H(a)) v^\prime$. 
		\end{enumerate}
		Combining these two, we get 
		\begin{equation}
			v^T (H(a^\prime) - H(a)) v \leq \lambda_{\max} (a^\prime) - \lambda_{\max} (a) \leq v^{\prime T} (H(a^\prime) - H(a)) v^\prime
		\end{equation}
		Now, since $f$ is $C^3$, we can take
		\begin{equation}
			H(a + \delta) - H(a) = D^3 f(a) [\delta] + o(\|\delta\|) 
		\end{equation}
		Substituting in gives us 
		\begin{equation}
			D^3 f(a)[\delta, v, v] + o(\|\delta\|) \leq \lambda_{\max} (a^\prime) - \lambda_{\max} (a) \leq D^3 f(a) [\delta, v^\prime, v^\prime] + o(\|\delta\|) 
		\end{equation}
		Subtracting $D^3 f(a)[\delta, v, v]$ from all three terms and dividing by $\|\delta\|$ gives 
		\begin{equation}
			\frac{o(\|\delta\|)}{\|\delta\|} \leq \frac{\lambda_{\max} (a^\prime) - \lambda_{\max} (a) - D^3 f(a)[\delta, v, v]}{\|\delta\|} \leq \frac{D^3 f(a)[\delta, v^\prime, v^\prime] - D^3 f(a)[\delta, v, v] + o(\|\delta\|)}{\|\delta\|}
		\end{equation}
		where the $o(\|\delta\|)$ terms are the (scalar) quantities $v^T \, o(\|\delta\|) \, v$ and $v^{\prime T} \, o(\|\delta\|) \, v^\prime$, which are bounded in absolute value by $\|o(\|\delta\|)\|$ since $\|v\| = \|v^\prime\| = 1$. The LHS clearly goes to $0$ as $\delta \to 0$. For the RHS, we split the difference by adding and subtracting $D^3 f(a)[\delta, v, v^\prime]$ and use the fact that $D^3 f(a)$ is \hyperref[lin-thm:operator_norm_bound]{trilinear and bounded by its operator norm}, 
		\begin{align}
			\big| D^3 f(a)[\delta, v^\prime, v^\prime] - D^3 f(a)[\delta, v, v] \big| & = \big| D^3 f(a)[\delta, v^\prime - v, v^\prime] + D^3 f(a)[\delta, v, v^\prime - v] \big| \\ 
			& \leq \|D^3 f(a)\| \|\delta\| \|v^\prime - v\| \|v^\prime\| + \|D^3 f(a)\| \|\delta\| \|v\| \|v^\prime - v\| \\ 
			& = 2 \|D^3 f(a)\| \|\delta\| \|v^\prime - v\|
		\end{align}
		Therefore, the RHS is bounded above by $2 \|D^3 f(a)\| \|v^\prime - v\| + \frac{o(\|\delta\|)}{\|\delta\|}$. We now need $v^\prime \to v$ as $\delta \to 0$, and this is where we assume that $\lambda_{\max}(a)$ is a \textit{simple} eigenvalue of $H(a)$, so that $v$ is unique up to sign. Since $H$ is continuous, $\lambda_{\max}$ is continuous, and so any limit point $w$ of $v^\prime$ as $\delta \to 0$ is a unit vector satisfying $H(a) w = \lambda_{\max}(a) w$, i.e. $w = \pm v$. Choosing the sign of $v^\prime$ so that $\langle v, v^\prime \rangle \geq 0$ then gives $v^\prime \to v$. Hence the RHS goes to $0$ as well, and by the squeeze theorem, 
		\begin{equation}
			\lim_{\delta \to 0} \frac{\lambda_{\max} (a + \delta) - \lambda_{\max} (a) - D^3 f(a)[\delta, v, v]}{\|\delta\|} = 0
		\end{equation}
		By the \hyperref[def:derivative]{definition of the derivative}, $\lambda_{\max}$ is differentiable at $a$ with 
		\begin{equation}
			D\lambda_{\max}(a) = D^3 f(a)[v, v] \in (\mathbb{R}^n)^\dagger
		\end{equation}
		In coordinates, $\partial_{x_i} \lambda_{\max}(a) = \sum_{j, k} v_j v_k \, \partial_{x_k x_j x_i} f(a)$, i.e. the derivative of the sharpness in direction $\delta$ is the third derivative contracted with $\delta$ and twice with the top eigenvector. 
	\end{example}

\subsection{Taylor's Formula}

  Just as in the \hyperref[real-thm:taylor-peano]{univariate case}, there are two separate theorems with two different purposes. The first, due to Peano, tells us \textit{how fast} the error of the polynomial approximation vanishes and works for vector-valued functions $f: E \to \mathbb{R}^m$. The second, due to Lagrange, gives an \textit{explicit formula} for the error and works for scalar-valued functions $f: E \to \mathbb{R}$. Their assumptions and proofs are again completely different. Throughout, $E \subset \mathbb{R}^n$ is open, and for $T \in ((\mathbb{R}^n)^\dagger)^{\otimes k} \otimes \mathbb{R}^m$ and $h \in \mathbb{R}^n$ we write
  \begin{equation}
    T [h^{\otimes k}] \coloneqq T[h, h, \ldots, h] \in \mathbb{R}^m
  \end{equation}
  for the $k$-linear map evaluated with $h$ in every slot. Note that this is a homogeneous polynomial of degree $k$ in $h$, since $T[(th)^{\otimes k}] = t^k T[h^{\otimes k}]$. We say that $g(h) = o(\|h\|^k)$ as $h \to 0$ if $\|g(h)\| / \|h\|^k \to 0$, in the sense of \hyperref[real-def:little-o-notation]{little-o notation}. 

  \begin{definition}[Taylor Polynomial]
    \label{def:taylor-polynomial}
    Let $f: E \to \mathbb{R}^m$ be $n$-times differentiable at $a \in E$. The \textbf{Taylor polynomial of order $n$} of $f$ at $a$ is the polynomial in $h \in \mathbb{R}^n$ given by
    \begin{equation}
      P_n f_a (h) \coloneqq \sum_{k=0}^n \frac{1}{k!} D^k f(a) [h^{\otimes k}] = f(a) + Df(a) h + \frac{1}{2} D^2 f(a) [h^{\otimes 2}] + \ldots + \frac{1}{n!} D^n f(a) [h^{\otimes n}]
    \end{equation}
    and we approximate $f(x)$ by $P_n f_a (x - a)$. 
  \end{definition}

  The tensor form is clean, but to compute with it we need the coordinate form. This is where all the work on iterated partials pays off. 

  \begin{lemma}[Coordinate Form of the $k$th Term]
    \label{thm:coordinate-form-of-kth-term}
    If $f: E \to \mathbb{R}$ is $C^k$, then for all $x \in E$ and $h \in \mathbb{R}^n$, 
    \begin{equation}
      D^k f(x) [h^{\otimes k}] = \sum_{|\alpha| = k} \frac{k!}{\alpha!} \partial^\alpha f(x) \, h^\alpha
    \end{equation}
    Therefore, $P_n f_a (h) = \sum_{|\alpha| \leq n} \frac{\partial^\alpha f(a)}{\alpha!} h^\alpha$ for $f \in C^n$. 
  \end{lemma}
  \begin{proof}
    Write $h = \sum_i h_i e_i$. Since $D^k f(x)$ is $k$-linear, we can expand it over the basis, and then use the \hyperref[def:ck_functions]{identity for its entries}, 
    \begin{equation}
      D^k f(x) [h^{\otimes k}] = \sum_{i_1, \ldots, i_k = 1}^n h_{i_1} \cdots h_{i_k} \, D^k f(x) [e_{i_1}, \ldots, e_{i_k}] = \sum_{i_1, \ldots, i_k = 1}^n h_{i_1} \cdots h_{i_k} \, \partial_{x_{i_k} \ldots x_{i_1}} f(x)
    \end{equation}
    Now group the sequences $(i_1, \ldots, i_k)$ by the multi-index $\alpha$ where $\alpha_j$ is the number of times $j$ appears. For each such sequence, $h_{i_1} \cdots h_{i_k} = h^\alpha$, and by the \hyperref[thm:symmetry-in-kth-iterated-partials]{symmetry corollary}, the partial is $\partial^\alpha f(x)$. There are exactly $k!/\alpha!$ sequences for each $\alpha$ with $|\alpha| = k$, which gives the result. 
  \end{proof}

  For the polynomial $f(x_1, x_2) = x_1^2 x_2 + x_2^3$ in our example above, this is exactly the computation we did for $D^3 f(x) [h^{\otimes 3}] = 6 h_1^2 h_2 + 6 h_2^3$. 


  Let's start with Peano's again, and ask what the successive approximations of $f: E \to \mathbb{R}^m$ near $a \in E$ should look like, i.e. we look for polynomials $P$ with
  \begin{equation}
    f(a + h) = P(h) + o(\|h\|^n) \text{ as } h \to 0
  \end{equation}
  \begin{enumerate}
    \item \textit{Constant Approximation}. For $n = 0$ the only possibility is $P = f(a)$, and this works if and only if $f$ is continuous at $a$. 

    \item \textit{Linear Approximation}. For $n = 1$, the polynomial $P(h) = f(a) + Df(a) h$ works by the \hyperref[def:derivative]{definition of the derivative}. 

    \item \textit{Quadratic Approximation}. For $n = 2$, we want a polynomial $P(h) = c_0 + c_1 h + \frac{1}{2} H[h^{\otimes 2}]$ with $c_1$ linear and $H$ bilinear. Which bilinear form $H$ should we choose? We show that it is forced to be $D^2 f(a)$. In the univariate case, we found the coefficients by dividing through and using L'Hopital's rule, which has no multivariate analogue. Instead, we reduce to the univariate case by restricting to lines. Suppose that $f \in C^2$ and 
    \begin{equation}
      f(a + h) = f(a) + Df(a) h + \frac{1}{2} H(h, h) + o(\|h\|^2)
    \end{equation}
    Since only the diagonal $H(h, h)$ enters, we may assume $H$ is symmetric. Fix a direction $v$ and let $g(t) \coloneqq f(a + tv)$. Setting $h = tv$ gives 
    \begin{equation}
      g(t) = f(a) + Df(a) v \, t + \frac{1}{2} H(v, v) \, t^2 + o(t^2)
    \end{equation}
    On the other hand, $g$ is a $C^2$ function of one variable, so the \hyperref[real-thm:taylor-peano]{univariate Peano theorem} gives $g(t) = g(0) + g^\prime(0) t + \frac{1}{2} g^{\prime\prime}(0) t^2 + o(t^2)$. By the \hyperref[thm:chain-rule]{chain rule} applied to $t \mapsto a + tv$, whose derivative is $v$, we have $g^\prime (t) = Df(a + tv) v$. Since evaluating at the fixed vector $v$ is linear, applying the chain rule again to $t \mapsto Df(a + tv)$ gives $g^{\prime\prime}(t) = D^2 f(a + tv) [v, v]$, so 
    \begin{equation}
      g^\prime (0) = Df(a) v, \qquad g^{\prime\prime}(0) = D^2 f(a) [v, v]
    \end{equation}
    Subtracting the two expansions of $g$, 
    \begin{equation}
      \frac{1}{2} \big( H(v, v) - D^2 f(a) [v, v] \big) t^2 = o(t^2) 
    \end{equation}
    and dividing by $t^2$ and letting $t \to 0$ gives $H(v, v) = D^2 f(a) [v, v]$ for every $v$. Finally, a symmetric bilinear form is determined by its diagonal via the polarization identity $H(u, v) = \frac{1}{4} \big( H(u + v, u + v) - H(u - v, u - v) \big)$, and $D^2 f(a)$ is symmetric by \hyperref[thm:clairuts-theorem]{Clairaut's theorem}, so $H = D^2 f(a)$. 
  \end{enumerate}

  Now we can state Peano's theorem. We state it for $C^n$ functions, which makes the derivatives $D^k f(a)$ symmetric.\footnote{The theorem is true if $f$ is only $n$-times differentiable \textit{at} $a$, matching the univariate statement. However, this needs the symmetry of $D^k f(a)$ at a single point, which is a refinement of Clairaut's theorem, so we skip it.} 

  \begin{theorem}[Taylor's Theorem with Peano's Form of Remainder]
    \label{thm:multivariate-taylor-peano}
    If $f: E \to \mathbb{R}^m$ is $C^n$ for $n \geq 1$, then for any $a \in E$, 
    \begin{equation}
      f(a + h) = \bigg( \sum_{k=0}^n \frac{1}{k!} D^k f(a) [h^{\otimes k}] \bigg) + o(\|h\|^n) \text{ as } h \to 0
    \end{equation}
  \end{theorem}
  \begin{proof}
    We prove by induction on $n$. For $n = 1$ this is the definition of the derivative. Now let $n \geq 2$ and assume it holds for $n - 1$ (for all $C^{n-1}$ maps into any $\mathbb{R}^{m'}$). Since $E$ is open, pick $r > 0$ with $B(a, r) \subset E$ and define the error 
    \begin{equation}
      R(h) \coloneqq f(a + h) - P_n f_a (h) \qquad \text{for } \|h\| < r
    \end{equation}
    Then $R(0) = 0$, and we'd like to show $R(h) = o(\|h\|^n)$. Since $f$ is differentiable and $P_n f_a$ is a polynomial, $R$ is differentiable on $B(0, r)$, and 
    \begin{equation}
      DR(h) = Df(a + h) - D(P_n f_a)(h)
    \end{equation}
    We compute the derivative of the polynomial. Since $D^k f(a)$ is symmetric (by the \hyperref[thm:symmetry-in-kth-iterated-partials]{symmetry corollary}, applied to each component of $f$), the derivative of $h \mapsto D^k f(a) [h^{\otimes k}]$ in direction $u$ is $k \, D^k f(a) [u, h, \ldots, h]$, so 
    \begin{equation}
      D(P_n f_a)(h) \, u = \sum_{k=1}^n \frac{1}{(k-1)!} D^k f(a) [u, h, \ldots, h]
    \end{equation}
    On the other hand, consider the map $Df: E \to (\mathbb{R}^n)^\dagger \otimes \mathbb{R}^m \simeq \mathbb{R}^{mn}$. Unwinding the recursive \hyperref[def:kth-frechet-derivative]{definition}, $D^j (Df)(a) [h_1, \ldots, h_j] \, u = D^{j+1} f(a) [h_1, \ldots, h_j, u]$, so $Df$ is $C^{n-1}$ and by symmetry, its Taylor polynomial of order $n-1$ is 
    \begin{equation}
      P_{n-1} (Df)_a (h) \, u = \sum_{j=0}^{n-1} \frac{1}{j!} D^{j+1} f(a) [h, \ldots, h, u] = \sum_{k=1}^{n} \frac{1}{(k-1)!} D^k f(a) [u, h, \ldots, h] 
    \end{equation}
    which is exactly $D(P_n f_a)(h) \, u$. So $D(P_n f_a)(h) = P_{n-1}(Df)_a(h)$, and by the induction hypothesis applied to $Df$, 
    \begin{equation}
      DR(h) = Df(a + h) - P_{n-1}(Df)_a (h) = o(\|h\|^{n-1})
    \end{equation}
    Now fix $\epsilon > 0$, and choose $\delta \in (0, r)$ such that $\|DR(y)\| \leq \epsilon \|y\|^{n-1}$ whenever $\|y\| < \delta$. For $\|h\| < \rho < \delta$, we have $\|DR(y)\| \leq \epsilon \rho^{n-1}$ on the convex open ball $B(0, \rho)$, so by the \hyperref[thm:mean-value-theorem]{mean value theorem} applied to $R$ between $0$ and $h$, 
    \begin{equation}
      \|R(h)\| = \|R(h) - R(0)\| \leq \epsilon \rho^{n-1} \|h\|
    \end{equation}
    Letting $\rho \downarrow \|h\|$ gives $\|R(h)\| \leq \epsilon \|h\|^n$ for all $\|h\| < \delta$, and so $R(h) = o(\|h\|^n)$. 
  \end{proof}

  Note that this is where the multivariate and univariate proofs differ. There, we used L'Hopital's rule $n-1$ times; here, we use induction on the derivative $Df$ itself, and the mean value theorem takes the place of L'Hopital. The reason we needed the general codomain $\mathbb{R}^{m'}$ in the induction is that $Df$ takes values in a matrix space, even if $f$ is scalar-valued. 

  \begin{corollary}[2nd Order Taylor Polynomial]
    \label{thm:2nd-order-taylor-polynomial}
    The second order approximation of a $C^2$ function $f: E \to \mathbb{R}$ about a point $a$ is 
    \begin{equation}
      f (a + h) = f(a) + D f(a) h + \frac{1}{2} h^T H f(a) h + o(\|h\|^2)
    \end{equation}
    where $D f(a)$ is the total derivative at $a$ and $H f(a)$ is the \hyperref[def:hessian]{Hessian} matrix at $a$. 
  \end{corollary}
  \begin{proof}
    This is Peano's theorem for $n = 2$, together with $D^2 f(a) [h^{\otimes 2}] = \sum_{i, j} h_i h_j \, \partial_{x_j x_i} f(a) = h^T H f(a) h$ by the \hyperref[def:ck_functions]{entries of $D^2 f(a)$} (and symmetry of the Hessian). 
  \end{proof}

  Now we look at Lagrange's, whose purpose is to get an explicit formula for the error, and like the univariate case, it is an extension of the mean value theorem. Given $f: E \to \mathbb{R}$ and $a, a + h \in E$, we want to compare $f(a + h)$ with $f(a)$. Unlike Peano's, we cannot do this for vector-valued functions since the \hyperref[real-thm:mvt]{mean value theorem} (with equality) fails for them. 

  The idea is to reduce to the univariate case by restricting $f$ to the line segment from $a$ to $a + h$. Define 
  \begin{equation}
    \varphi(t) \coloneqq f(a + th)
  \end{equation}
  Then $\varphi(0) = f(a)$ and $\varphi(1) = f(a + h)$, so what we want is a Taylor expansion of $\varphi$ at $0$ evaluated at $1$. All we need is to compute the derivatives of $\varphi$. 

  \begin{lemma}[Derivatives Along a Line]
    \label{thm:derivatives-along-a-line}
    If $f: E \to \mathbb{R}$ is $k$-times differentiable on $E$, then for $a, h$ with $a + th \in E$, 
    \begin{equation}
      \varphi^{(j)}(t) = D^j f(a + th) [h^{\otimes j}] \qquad \text{for } j = 0, 1, \ldots, k
    \end{equation}
  \end{lemma}
  \begin{proof}
    By induction on $j$. The case $j = 0$ is the definition. Assume it for $j < k$, and let $g_j (x) \coloneqq D^j f(x) [h^{\otimes j}]$, so that $\varphi^{(j)}(t) = g_j(a + th)$. Since $D^j f$ is differentiable and evaluating a $j$-linear form at the fixed vectors $(h, \ldots, h)$ is a linear map, $g_j$ is differentiable with $D(g_j)(x) u = D^{j+1} f(x) [u, h, \ldots, h]$ by the recursive \hyperref[def:kth-frechet-derivative]{definition}. Then by the \hyperref[thm:chain-rule]{chain rule} applied to $t \mapsto a + th$, whose derivative is $h$, 
    \begin{equation}
      \varphi^{(j+1)}(t) = D(g_j)(a + th) \, h = D^{j+1} f(a + th) [h^{\otimes (j+1)}]
    \end{equation}
  \end{proof}

  \begin{theorem}[Taylor's Theorem with Lagrange's Form of Remainder]
    \label{thm:multivariate-taylor-lagrange}
    Suppose $f: E \to \mathbb{R}$ is $C^{n+1}$, and the segment $\{a + th : t \in [0, 1]\}$ lies in $E$. Then, there exists a $\theta \in (0, 1)$ s.t. 
    \begin{equation}
      f(a + h) = \bigg( \sum_{k=0}^n \frac{1}{k!} D^k f(a) [h^{\otimes k}] \bigg) + \frac{1}{(n+1)!} D^{n+1} f(a + \theta h) [h^{\otimes (n+1)}]
    \end{equation}
    In coordinates, this reads 
    \begin{equation}
      f(a + h) = \bigg( \sum_{|\alpha| \leq n} \frac{\partial^\alpha f(a)}{\alpha!} h^\alpha \bigg) + \sum_{|\alpha| = n+1} \frac{\partial^\alpha f(a + \theta h)}{\alpha!} h^\alpha
    \end{equation}
  \end{theorem}
  \begin{proof}
    Since $E$ is open, $\varphi(t) = f(a + th)$ is defined on an open interval containing $[0, 1]$. By the \hyperref[thm:derivatives-along-a-line]{lemma}, $\varphi$ is $(n+1)$-times differentiable with $\varphi^{(n+1)}(t) = D^{n+1} f(a + th)[h^{\otimes (n+1)}]$, which is continuous in $t$ since $f \in C^{n+1}$. So $\varphi$ satisfies the hypotheses of the \hyperref[real-thm:taylor-lagrange]{univariate Lagrange form}, and there is a $\theta \in (0, 1)$ with 
    \begin{equation}
      \varphi(1) = \sum_{k=0}^n \frac{\varphi^{(k)}(0)}{k!} + \frac{\varphi^{(n+1)}(\theta)}{(n+1)!}
    \end{equation}
    Substituting $\varphi^{(k)}(0) = D^k f(a) [h^{\otimes k}]$ and $\varphi^{(n+1)}(\theta) = D^{n+1} f(a + \theta h)[h^{\otimes (n+1)}]$ gives the first formula. The coordinate form follows from the \hyperref[thm:coordinate-form-of-kth-term]{coordinate lemma}. 
  \end{proof}

  Just like in the univariate case, only the final term depends on the unknown point $\theta$, and a bound on the $(n+1)$th derivative controls the whole error. 

  \begin{corollary}[Uniform Bound on the Remainder]
    Under the assumptions above, if $\|D^{n+1} f(x)\| \leq M$ on the segment from $a$ to $a + h$, then 
    \begin{equation}
      \big| f(a + h) - P_n f_a (h) \big| \leq \frac{M}{(n+1)!} \|h\|^{n+1}
    \end{equation}
  \end{corollary}
  \begin{proof}
    By the \hyperref[lin-thm:operator_norm_bound]{operator norm bound}, $|D^{n+1} f(x) [h^{\otimes (n+1)}]| \leq \|D^{n+1} f(x)\| \|h\|^{n+1}$. Apply this to the remainder in Lagrange's form. 
  \end{proof}

  Note that this also shows that Lagrange's form implies Peano's form for scalar-valued $f \in C^{n+1}$ (take $M$ to be the bound on a small ball, which exists by continuity of $D^{n+1}f$), with the stronger conclusion that the remainder is $O(\|h\|^{n+1})$.

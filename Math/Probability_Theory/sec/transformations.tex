\section{Transformations of Random Variables} 

  In many applications, it happens that we are interested not in the value of the random variable $X$, but a function of it. That is, given a \hyperref[def:probability-space]{probability space} $(\Omega, \mathcal{F}, \mathbb{P})$, let us have a random variable $X: \Omega \rightarrow \mathbb{R}$. We can then define another function $f: \mathbb{R} \rightarrow \mathbb{R}$ and consider the potential random variable $f \circ X : \Omega \rightarrow \mathbb{R}$. We say potential because we don't know yet whether $f \circ X$ is measurable (i.e. the preimage of every Borel set in $\mathbb{R}$ is in $\mathcal{F}$). This condition suffices if $f$ itself is a \hyperref[mt-def:measurable-function]{measurable function}, i.e. for every Borel set $B \in \mathcal{R}$, its preimage $f^{-1} (B)$ is Borel in $\mathbb{R}$, and by measurablility of $X$, its preimage under $X$ is $\mathcal{F}$-measurable, making $f \circ X$ a viable random variable. With this new random variable $f \circ X$, we would now like to answer the question: What is the probability law $\mathbb{P}_{f \circ X}$ of $\mathbb{R}$? 

  Now we will look at more general transformations that are not just minimum, maximum, deterministic sums, or random sums. Let us have a probability space $(\Omega, \mathcal{F}, \mathbb{P})$, a random variable $X: \Omega \rightarrow \mathbb{R}$, and a measurable function $f: \mathbb{R} \rightarrow \mathbb{R}$. Now given that we know the CDF (and therefore distribution) of $X$, we want to find the CDF of random variable $Y = f(X) = f \circ X$ (which we have established as a random variable already due to measurability of $f$): $F_Y (y) = \mathbb{P}(Y \leq y)$, which is just $\mathbb{P}_Y ((-\infty, y])$ (where $\mathbb{P}_Y$ is the probability law on $Y$). But rather than trying to take the preimage of the entire composite random variable $Y$ and calculating $\mathbb{P}\big( Y^{-1}((-\infty, y]) \big)$ under the probability on $\mathcal{F}$, let's just take the preimage one step at a time. Note that $f^{-1} \big( (-\infty, y] \big) = \{x \in \mathbb{R} \mid f(x) \leq y\}$. We can then write the CDF of $Y$ in terms of the probability law of $X$: 
  \begin{align*}
    F_Y (y) & = \mathbb{P}_X \big( f^{-1} ((-\infty, y]) \big) \\
    & = \mathbb{P}_X \big( \{x \in \mathbb{R} \mid f(x) \leq y\} \big) \\
    & = \mathbb{P} \big( X^{-1} \circ f^{-1} ((-\infty, y]) \big) 
  \end{align*}
  Depending on how complicated $f$ is, this may be easy or not, but conceptually, this is no problem. But theoretically, this is as far as we can go in general. We now show formulas for the density of $Y = g(X)$ under more specific assumptions about the transformation. 

  \begin{theorem}[Density of a Monotone Transformation]
    \label{thm:density-of-a-monotone-transformation}
    Let $X$ be a \hyperref[def:continuous-random-variable]{continuous random variable} with (continuous) PDF $f_X$, and let $g: \mathbb{R} \to \mathbb{R}$ be continuously differentiable with $g^\prime (x) \neq 0$ for all $x$ (so $g$ is strictly monotone and has a differentiable inverse on its image $J = g(\mathbb{R}) = (a, b)$). Then, $Y = g(X)$ is a continuous random variable with PDF 
    \begin{equation}
      f_Y (y) = \frac{f_X \big( g^{-1} (y) \big)}{\big| g^\prime \big( g^{-1} (y) \big) \big|} = f_X \big( g^{-1} (y) \big) \, \big| (g^{-1})^\prime (y) \big| \text{ for } y \in J
    \end{equation}
    and $f_Y (y) = 0$ for $y \not\in J$. 
  \end{theorem}
  \begin{proof}
    Since $g$ is differentiable and strictly monotone, it is invertible on $J$, and $(g^{-1})^\prime (y) = 1/g^\prime (g^{-1}(y))$. Fix $y \in J$. 
    \begin{enumerate}
      \item If $g$ is increasing, then $\{g(X) \leq y\} = \{X \leq g^{-1} (y)\}$, so using the \hyperref[real-thm:change-of-variable]{change of variable} $x = g^{-1} (t)$ (so $dx = (g^{-1})^\prime (t) \,dt$), 
      \begin{equation}
        F_Y (y) = \int_{-\infty}^{g^{-1} (y)} f_X (x) \,dx = \int_{a}^{y} \frac{f_X \big( g^{-1} (t) \big)}{g^\prime \big( g^{-1} (t) \big)} \,dt
      \end{equation}
      \item If $g$ is decreasing, then $\{g(X) \leq y\} = \{X \geq g^{-1} (y)\}$, and the same substitution (which now reverses the limits, and $(g^{-1})^\prime < 0$) gives 
      \begin{equation}
        F_Y (y) = \int_{g^{-1} (y)}^{\infty} f_X (x) \,dx = \int_{a}^{y} \frac{f_X \big( g^{-1} (t) \big)}{- g^\prime \big( g^{-1} (t) \big)} \,dt
      \end{equation}
    \end{enumerate}
    In both cases the integrand is $f_X (g^{-1}(t)) / |g^\prime (g^{-1}(t))|$, and $F_Y (y)$ is the integral of it from $-\infty$ to $y$ (it is $0$ for $y \leq a$ and $1$ for $y \geq b$), so it is the PDF of $Y$. 
  \end{proof}

  The factor $\frac{1}{|g^\prime (g^{-1}(y))|} = |(g^{-1})^\prime (y)|$ is the one-dimensional \hyperref[mv-def:jacobian]{Jacobian}, the same one that we use when we perform a change of variables in integration. The multivariate version is the following. 

  \begin{theorem}[Density of a Multivariate Transformation]
    \label{thm:density-of-a-multivariate-transformation}
    Let $\mathbf{X}: \Omega \to \mathbb{R}^n$ be a random vector whose law is absolutely continuous w.r.t. the Lebesgue measure on $\mathbb{R}^n$, with joint PDF $f_{\mathbf{X}}$. Let $U \subset \mathbb{R}^n$ be open with $\mathbb{P}(\mathbf{X} \in U) = 1$, and let $g: U \to V$ be a diffeomorphism ($C^1$ bijection with $C^1$ inverse) onto an open set $V \subset \mathbb{R}^n$. Then, $\mathbf{Y} = g(\mathbf{X})$ has joint PDF 
    \begin{equation}
      f_{\mathbf{Y}} (y) = f_{\mathbf{X}} \big( g^{-1} (y) \big) \, \big| \det J_{g^{-1}} (y) \big| = \frac{f_{\mathbf{X}} \big( g^{-1} (y) \big)}{\big| \det J_g \big( g^{-1} (y) \big) \big|} \text{ for } y \in V
    \end{equation}
    and $f_{\mathbf{Y}} (y) = 0$ for $y \not\in V$, where $J$ denotes the \hyperref[mv-def:jacobian]{Jacobian} matrix. 
  \end{theorem}
  \begin{proof}
    Let $B \subset V$ be Borel. Since $g$ is a bijection, $\{\mathbf{Y} \in B\} = \{\mathbf{X} \in g^{-1}(B)\}$ (up to the probability zero event $\mathbf{X} \notin U$), so 
    \begin{equation}
      \mathbb{P}(\mathbf{Y} \in B) = \mathbb{P}(\mathbf{X} \in g^{-1}(B)) = \int_{g^{-1}(B)} f_{\mathbf{X}} (x) \,dx
    \end{equation}
    By the \hyperref[mv-thm:integration-over-change-of-bases-in-rn]{change of variables theorem in $\mathbb{R}^n$} applied to the substitution $x = g^{-1}(y)$, which maps $B$ onto $g^{-1}(B)$, 
    \begin{equation}
      \int_{g^{-1}(B)} f_{\mathbf{X}} (x) \,dx = \int_B f_{\mathbf{X}} \big( g^{-1} (y) \big) \, \big| \det J_{g^{-1}} (y) \big| \,dy
    \end{equation}
    Hence $\mathbb{P}(\mathbf{Y} \in B) = \int_B f_{\mathbf{Y}} (y) \,dy$ for every Borel $B \subset V$, and $\mathbb{P}(\mathbf{Y} \in V) = 1$, so $f_{\mathbf{Y}}$ is the PDF of $\mathbf{Y}$. Finally, by the chain rule applied to $g \circ g^{-1} = \mathrm{id}$, $J_{g^{-1}} (y) = \big( J_g (g^{-1}(y)) \big)^{-1}$, so $|\det J_{g^{-1}} (y)| = 1 / |\det J_g (g^{-1}(y))|$. 
  \end{proof}

  For $n = 1$ this is the previous theorem. Let's move onto some examples. We start with a practical way to generate a Gaussian distribution, which is how most modern software computes. 

  \begin{example}[Box-Muller Transform]
    \label{ex:box-muller-transform}
    Given that you have a uniform random number generator in $[0, 1]$, you can generate a normal $N(0, 1)$ by transforming it using the inverse CDF of the normal. This is usually computationally heavy since the inverse CDF of the Gaussian requires expensive operations. An easier way is to use the \textbf{Box-Muller transform}, where you take two uniforms $u_1, u_2$ and transform it as 
    \begin{align*}
        x_1 & = \sqrt{-2 \ln(u_1)} \cos(2 \pi u_2) \\
        x_2 & = \sqrt{-2 \ln(u_1)} \sin(2 \pi u_2)
    \end{align*}
    Once you have $x \sim N(0, 1)$, you can use $\mu + \sigma x \sim N(\mu, \sigma^2)$. You can extend this to a $n$-dimensional normal distribution $\mathbf{x} \sim N(\mathbf{0}, \mathbf{I})$ and transform it to get $\boldsymbol{\mu} + \boldsymbol{\Sigma}^{1/2} \mathbf{x} \sim N(\boldsymbol{\mu}, \boldsymbol{\Sigma})$. 
  \end{example}

  \begin{example}[Chi-Squared Distribution]
    \label{ex:chi-squared-distribution}
    Let $X \sim \mathcal{N}(0, 1)$ and $Y = f(X) = X^2$. Note that $X$ takes values in $(-\infty, +\infty)$ and $Y$ in $[0, +\infty)$. Then, we can write 
    \begin{align*}
      F_Y (y) & = \mathbb{P}(Y \leq y) \\ 
      & = \mathbb{P}_Y ( (-\infty, y]) \\
      & = \mathbb{P}_Y (  [0, y]) & (\text{range of } Y) \\
      & = \mathbb{P}_X ( f^{-1} ([0, y]) ) & (\text{work in prob. law of } X) \\
      & = \mathbb{P}_X ( [-\sqrt{y}, \sqrt{y}] ) \\
      & = \int_{-\sqrt{y}}^{\sqrt{y}} f_X (x) \,dx 
    \end{align*}
    Rewriting this using the notation $\mathbb{P}(\cdot)$, we have 
    \begin{align*}
      F_Y (y) & = \mathbb{P}(Y \leq y) \\
      & = \mathbb{P}(X^2 \leq y) \\
      & = \mathbb{P}( -\sqrt{y} \leq X \leq \sqrt{y}) \\
      & = 2 \mathbb{P}(0 \leq X \leq \sqrt{y}) & (\text{Symmetry of Gaussian})\\
      & = \frac{2}{\sqrt{2} \pi} \int_0^{\sqrt{y}} e^{-x^2 / 2} \,dx 
    \end{align*}
    and this is clearly differentiable, since it is written like an integral. Doing so gives the PDF
    \begin{equation}
      f_Y (y) = \frac{1}{\sqrt{2 \pi y}} e^{-y/2} \text{ for } y \geq 0
    \end{equation}
    This describes the PDF of a \textbf{Chi-Squared} distribution. 
  \end{example}

  \begin{example}[Log-Normal Distribution]
    \label{ex:log-normal-distribution}
    Let $X \sim \mathcal{N}(0, 1)$ and $Y = f(X) = e^X$. Note that the range of $f$ is $(0, +\infty)$. So, 
    \begin{align*}
      F_Y (y) & = \mathbb{P}(Y \leq y) \\
      & = \mathbb{P}_Y ((-\infty, y]) \\
      & = \mathbb{P}_Y ( (0, y]) \\ 
      & = \mathbb{P}_X ( f^{-1} ((0, y]) ) \\
      & = \mathbb{P}_X ( (-\infty, \ln{y}] ) \\
      & = \int_{-\infty}^{\ln{y}} f_X (x)\,dx 
    \end{align*}
    Rewriting this using the notation $\mathbb{P}(\cdot)$, we have 
    \begin{align*}
      F_Y (y) & = \mathbb{P}(e^X \leq y) \\
      & = \mathbb{P}(X \leq \ln(y)) \\ 
      & = \int_{-\infty}^{\ln(y)} \frac{1}{\sqrt{2} \pi} e^{-x^2/ 2} \,dx 
    \end{align*}
    We can differentiate this to get 
    \begin{equation}
      f_Y (y) = \frac{1}{y \sqrt{2 \pi}} e^{-\frac{(ln{y})^2}{2}} \text{ for } y \geq 0
    \end{equation}
    This describes the PDF of a \textbf{log-normal} distribution. 
  \end{example}

  \begin{example}[Log-Normal Revisited]
    \label{ex:log-normal-revisited}
    Given $X \sim \mathcal{N}(0, 1)$ and $Y = e^X$ (which is monotonically increasing), we can simply plug in the formula to get the PDF: 
    \begin{equation}
      f_Y (y) = \frac{f_X (g^{-1} (y) )}{|g^\prime (g^{-1} (y))|} = \frac{f_X (\ln{y}) }{ | e^{\ln{y}} |} = \frac{1}{\sqrt{2 \pi} y} e^{-(\ln{y})^2 / 2}
    \end{equation}
    for $y > 0$. This domain is important since $\ln{y}$ is only defined for $y > 0$. 
  \end{example}

  \begin{example}[Non-Monotone Transformation of a Gaussian]
    \label{ex:non-monotone-transformation-of-a-gaussian}
    Given $X \sim \mathcal{N}(0, 1)$ and $Y = f(X) = X^2$, we cannot use the formula since $f$ is not monotonic on the range of $X$, which is $(-\infty, +\infty)$.  
  \end{example}

  \begin{example}[Squaring an Exponential Random Variable]
    \label{ex:squaring-an-exponential-random-variable}
    Given $X \sim \Exponential(\lambda)$ and $Y = f(X) = X^2$, it may seem like the formula is not applicable here, but $f$ \textit{is} monotonic on the range of $X$, which is $[0, + \infty)$. 
  \end{example}

  However, there is much less chance of error by deriving using first principles, so I would recommend using it always rather than these formulas. 

  Let's do the $n$-dimensional version of this. Given \hyperref[def:random-variable]{random variables} $X_1, X_2, \ldots, X_n$ iid random variables with joint density $f_{X_1 \ldots X_n} (x_1, \ldots, x_n)$, we define the transformation $g: \mathbb{R}^n \rightarrow \mathbb{R}^n$ as 
  \begin{equation}
    \begin{bmatrix} Y_1 \\ \vdots \\ Y_N \end{bmatrix} = \begin{bmatrix} g_1 (X_1) \\ \vdots \\ g_n (X_N) \end{bmatrix}
  \end{equation}
  Then, the PDF of $Y$ will be 
  \begin{align*}
    f_{Y_1 \ldots Y_n} (y_1, \ldots, y_n) & = f_{X_1 \ldots X_n} \big( \mathbf{g}^{-1} (\mathbf{y}) \big) \cdot | \mathbf{J}(\mathbf{y})| \\
    & = f_{X_1 \ldots X_n} \big( g_1^{-1}(y_1), \ldots, g_n^{-1} (y_n) \big) \cdot | \mathbf{J}(\mathbf{y})| \\
  \end{align*}
  where 
  \begin{equation}
    \mathbf{J}(y) = \mathrm{det}\begin{pmatrix} 
    \frac{\partial x_1}{\partial y_1} & \ldots & \frac{\partial x_n}{\partial y_1} \\
    \vdots & \ddots & \vdots \\ 
    \frac{\partial x_1}{\partial y_n} & \ldots & \frac{\partial x_n}{\partial y_n} \end{pmatrix}
  \end{equation}


  This also works for joint random variables, which we will learn later. Given a joint random variable $(X_1, X_2, \ldots X_n): \Omega \rightarrow \mathbb{R}^n$, we can define a \hyperref[mt-def:measurable-function]{measurable function} $f: \mathbb{R}^n \to \mathbb{R}$ and define the scalar random variable $f \circ (X_1, \ldots X_n)$ on $\Omega$. But again, we want to find what the CDF of this composition. 

  Now we've made a remark before that given a multivariate distribution $\mathbf{X} = (X_1, \ldots, X_n)$, all of its marginal distributions being Gaussian does not mean that $\mathbf{X}$ is a \hyperref[def:multivariate-gaussian]{multivariate Gaussian}. We give a counterexample. 

  \begin{example}[Distribution of a Transformed IID Pair]
    \label{ex:distribution-of-a-transformed-iid-pair}
  Let $Y_1, Y_2$ be iid \hyperref[def:random-variable]{random variables} distributed according to the PDF 
  \begin{equation}
    f_Y (y) = \sqrt{\frac{2}{\pi}} e^{-y^2 / 2} \text{ for } y > 0
  \end{equation}
  which we can interpret as a one-sided Gaussian. Let $W \sim \Bernoulli(\frac{1}{2})$ be independent of $Y_1$ and $Y_2$. Now, define the random variables 
  \begin{equation}
    X_1 = W \, Y_1 \text{ and } X_2 = W \, Y_2
  \end{equation}
  This is clearly not a \hyperref[def:multivariate-gaussian]{multivariate Gaussian}, even though the marginals are $X_1, X_2 \sim \mathcal{N}(0, 1)$. We could make the degenerate case that $X_1 = X_2$, which would make the image of $(X_1, X_2)$ just the line at $x_1 = x_2$, but we can think of this as a degenerate Gaussian with a singular $\boldsymbol{\Sigma}$.

\subsection{Sums of Random Variables}

  Now given two random variables $X, Y: \Omega \rightarrow \mathbb{R}$ that each push their own probability laws $\mathbb{P}_X, \mathbb{P}_Y$ onto $\mathbb{R}$, their sum $Z = X + Y$ is also a random variable that pushes its own probability law $\mathbb{P}_Z$. We must actually prove that $Z$ is a random variable, which we can do by proving that the preimage of every $(-\infty, x]$ is $\mathcal{F}$-measurable. Equivalently (by complementation), we must prove that the preimage of every $(x, +\infty)$ (that is, all sets of form $\{ \omega \mid Z(\omega) > z\}$) is $\mathcal{F}$-measurable. Now we can write $z$ as the sum of two numbers $z = q + (z - q)$, where $q \in \mathbb{R}$, and say that 
  \begin{equation}
    \{ \omega \mid Z(\omega) > z\} = \bigcup_{q \in \mathbb{R}} \{ \omega \mid X (\omega) > q , \; Y(\omega) > z - q\}
  \end{equation}
  But using the fact that $\mathbb{Q}$ is dense in $\mathbb{R}$, we can turn this from an uncountable union to a countable union and say 
  \begin{align}
    \{ \omega \mid Z(\omega) > z\} & = \bigcup_{q \in \mathbb{Q}} \{ \omega \mid X (\omega) > q , \; Y(\omega) > z - q\} \\
    & = \bigcup_{q \in \mathbb{Q}} \big( \{\omega \mid X(\omega) > q\} \cap \{ \omega \mid Y(\omega) > z - q\} \big) 
  \end{align}
  and since I have a countable union of (an intersection of) these $\mathcal{F}$-measurable sets, $\{ \omega \mid Z(\omega) > z\}$ is $\mathcal{F}$-measurable, and we are done. This equation above even gives us a hint of how to compute the CDF of $Z$. 

  \begin{theorem}[Sums and Products of Random Variables]
    \label{thm:sums-and-products-of-random-variables}
    Given \hyperref[def:random-variable]{random variables} $X_1, X_2, \ldots, X_n$ of \hyperref[def:probability-space]{probability space} $(\Omega, \mathcal{F}, \mathbb{P})$, 
    \begin{enumerate}
      \item $X_1 + \ldots + X_n$ is a random variable.
      \item $X_1 \cdot \ldots \cdot X_n$ is a random variable. 
    \end{enumerate}
  \end{theorem}

  For simplicity, we will only consider jointly discrete or jointly \hyperref[def:continuous-random-variable]{continuous random variables}. The probability law $\mathbb{P}_Z$ can be confusing to define, since given some Borel set $B \in \mathcal{R}$, we must now look at the preimage under the \textit{sum} $X + Y$. A simpler way to approach this is to consider the joint distribution $X, Y$ and look at its distribution, which we call the \textbf{\hyperref[real-thm:convolution]{convolution}} of $X$ and $Y$. This is especially simple to consider for \hyperref[def:discrete-random-variable]{discrete random variables}. 

  \begin{definition}[Sums of Discrete Random Variables]
    \label{def:sums-of-discrete-random-variables}
    Take two \hyperref[def:discrete-random-variable]{discrete random variables} $X, Y$ with their \hyperref[def:joint-pmf]{joint PMF} and their sum $Z = X + Y$. We can see that the PMF of $Z$ is 
    \begin{equation}
      \mathbb{P}(Z = z) = \sum_{(x, y) \,:\, x + y = z} \mathbb{P}(X = x, Y = y) = \sum_{x \in \mathcal{X}} \mathbb{P}(X = x, Y = z - x)
    \end{equation}
    The CDF is very simple, since we just have to sum over all $(x, y)$ such that their sum is less than $z$: 
    \begin{equation}
      \mathbb{P}(Z \leq z) = \sum_{(x, y) \,:\, x + y \leq z} \mathbb{P}(X = x, Y = y)
    \end{equation}
    If $X$ and $Y$ are independent, then their joint distribution is the product of their singular distributions, and so we have 
    \begin{equation}
      \mathbb{P}(Z = z) = \sum_{x \in \mathcal{X}} \mathbb{P}(X = x) \, \mathbb{P}(Y = z - x)
    \end{equation}
    which is called the \hyperref[real-thm:convolution]{convolution} $p_X \ast p_Y$ of the PMFs $p_X (x) = \mathbb{P}(X = x)$ and $p_Y (y) = \mathbb{P}(Y = y)$. 
  \end{definition}

  \begin{example}[Sums of Poisson RVs]
    \label{ex:sums-of-poisson-rvs}
    Let $X_1$ and $X_2$ be independent \hyperref[def:poisson-random-variable]{Poisson random variables} with parameters $\lambda_1, \lambda_2 > 0$, and let $Z = X_1 + X_2$. The PMF of each $X_i$ is 
    \begin{equation}
      p_{X_i} (k) = \frac{e^{-\lambda_i} \lambda_i^k}{k!} \text{ for } k \in \mathbb{Z}
    \end{equation}
    and taking the \hyperref[real-thm:convolution]{convolution} gives the PMF of $Z$: 
    \begin{align*}
      p_Z (z) & = (p_{X_1} \ast p_{X_2}) (z) \\
      & = \sum_{k=-\infty}^{+\infty} \frac{e^{-\lambda_1} \lambda_1^k}{k!} \cdot \frac{e^{-\lambda_2} \lambda_2^{z - k}}{(z - k)!} \\
      & = \sum_{k=0}^{z} \frac{e^{-\lambda_1} \lambda_1^k}{k!} \cdot \frac{e^{-\lambda_2} \lambda_2^{z - k}}{(z - k)!} \\ 
      & = \frac{e^{-(\lambda_1 + \lambda_2)}}{z!} \sum_{k=0}^z \binom{z}{k} \lambda_1^k \lambda_2^{z - k} \\
      & = \frac{e^{-(\lambda_1 + \lambda_2)} (\lambda_1 + \lambda_2)^z}{z!} 
    \end{align*}
    for $z \in \mathbb{N}_0$, which is the PMF of another Poisson. So, $Z \sim \Poisson(\lambda_1 + \lambda_2)$. 
  \end{example}

  This has a nice visualization, since the joint distribution of $X$ and $Y$ over $\mathbb{R}^2$ is being "summed up/integrated" over the diagonals of $\mathbb{R}^2$, i.e. the lines where $x + y = z$ for some $z$, sort of like marginalizing over these diagonals. This creates a new "diagonally marginal distribution" $Z$. 

  \begin{definition}[Sums of Continuous Random Variables]
    \label{def:sums-of-continuous-random-variables}
    Take two \hyperref[def:continuous-random-variable]{continuous random variables} $X, Y$ with their joint PDF $f_{X, Y} (x, y)$ and their sum $Z = X + Y$. To calculate the CDF, we must basically integrate the joint PDF over the borel set $\{(x, y) \in \mathbb{R}^2 \mid x + y \leq z\}$. 
    \begin{align*}
      \mathbb{P}(Z \leq z) = F_Z (z) & = \int_{(x, y) \,:\, x + y \leq z} f_{X, Y} (x, y) \,dy\,dx \\
      & = \int_{-\infty}^{+\infty} \int_{-\infty}^{z - x} f(x, y) \,dy \,dx
    \end{align*}
    We can see that the PDF of $Z$ is 
    \begin{equation}
      f_{Z} (z) = \int_{\mathbb{R}} f_{X, Y} (x, z - x) \, dx
    \end{equation}
    If $X$ and $Y$ are independent, then 
    \begin{equation}
      f_{Z} (z) = \int_{\mathbb{R}} f_{X} (x) \, f_Y (z - x) \,dx \coloneqq f_X \ast f_Y
    \end{equation}
    where $f_Z = f_X \ast f_Y$ is the \hyperref[real-thm:convolution]{convolution} of $f_X$ and $f_Y$. 
  \end{definition}

  \begin{definition}[Convolution]
    \label{def:convolution}
    Given two functions $f, g: \mathbb{R} \to \mathbb{R}$, the \textbf{convolution} of $f$ and $g$ is a new function $f \ast g$ defined  
    \begin{equation}
      (f \ast g) (t) \coloneqq \int_\mathbb{R} f(t)\, g(t - \tau) \, d \tau
    \end{equation}
  \end{definition}

  Usually, when we take \hyperref[def:convolution]{convolutions}, it is not pretty and even for nice distributions like two Gaussians, convolving them is quite complicated. What we can do is transform them (using Laplace, Fourier, etc.) to make calculations easier and more elegant, which we will talk about later.  

  \begin{example}[Sum of Two Independent Exponentials]
    \label{ex:sum-of-two-independent-exponentials}
    Let $X_1$ and $X_2$ be independent exponential with parameters $\lambda_1, \lambda_2$, with individual PDFs $f_{X_i} (x) = \lambda_i e^{-\lambda_i x}$ for $x \geq 0$. Let $Z = X_1 + X_2$. Then, 
    \begin{align*}
      f_Z (z) = (f_{X_1} \ast f_{X_2})(z) & = \int_{-\infty}^\infty \lambda_1 e^{-\lambda_1 x} \, \lambda_2 e^{-\lambda_2 (z -x)} \, dx \\
      & = \int_{0}^z \lambda_1 e^{-\lambda_1 x} \, \lambda_2 e^{-\lambda_2 (z -x)} \, dx \\ 
      & = \lambda_1 \lambda_2 e^{-\lambda_2 z} \int_0^z e^{(\lambda_2 - \lambda_1) x}\,dx \\
      & = \begin{cases} \frac{\lambda_1 \lambda_2}{\lambda_2 - \lambda_1} \big( e^{-\lambda_1 z} - e^{-\lambda_2 z} \big) & \text{ if } \lambda_1 \neq \lambda_2 \\
      \lambda^2 z e^{-\lambda z} & \text{ if } \lambda_1 = \lambda_2 = \lambda \end{cases} 
    \end{align*}
    When $\lambda_1 = \lambda_2$, this is the Erlang distribution (i.e. $\GammaDist(2, \lambda)$, see \hyperref[thm:sum-of-independent-exponentials-is-gamma]{the theorem below}), which has many applications. The case $\lambda_1 \neq \lambda_2$ is called the hypoexponential distribution. 
  \end{example}

  The following theorems are all proved by taking the \hyperref[def:convolution]{convolution} of the densities (or PMFs) of independent random variables, as in \hyperref[ex:sums-of-poisson-rvs]{the Poisson example above}. 

  \begin{theorem}[Sum of Independent Bernoullis is Binomial]
    \label{thm:sum-of-independent-bernoullis-is-binomial}
    If $X_1, \ldots, X_n$ are independent $\Bernoulli(p)$ \hyperref[def:indicator-bernoulli-random-variable]{random variables}, then $S_n \coloneqq X_1 + \cdots + X_n \sim \Binomial(n, p)$. 
  \end{theorem}
  \begin{proof}
    We induct on $n$. For $n = 1$ this is trivial. Suppose $S_n \sim \Binomial(n, p)$. Since $S_{n+1} = S_n + X_{n+1}$ with $S_n$ and $X_{n+1}$ independent, convolving with the PMF of $X_{n+1}$ (which is supported on $\{0, 1\}$) gives, for $0 \leq k \leq n+1$, 
    \begin{align*}
      \mathbb{P}(S_{n+1} = k) & = \mathbb{P}(S_n = k) \, \mathbb{P}(X_{n+1} = 0) + \mathbb{P}(S_n = k - 1) \, \mathbb{P}(X_{n+1} = 1) \\
      & = \binom{n}{k} p^k (1 - p)^{n - k + 1} + \binom{n}{k-1} p^{k} (1 - p)^{n - k + 1} \\
      & = \bigg[ \binom{n}{k} + \binom{n}{k - 1} \bigg] p^k (1 - p)^{n + 1 - k} = \binom{n+1}{k} p^k (1 - p)^{n+1-k}
    \end{align*}
    by Pascal's rule (with the convention $\binom{n}{j} = 0$ for $j < 0$ or $j > n$). This is the PMF of a $\Binomial(n+1, p)$. 
  \end{proof}

  \begin{theorem}[Sum of Independent Geometrics is Negative Binomial]
    \label{thm:sum-of-independent-geometrics-is-negative-binomial}
    If $X_1, \ldots, X_n$ are independent $\Geometric(p)$ \hyperref[def:geometric-random-variable]{random variables}, then $S_n \coloneqq X_1 + \cdots + X_n$ has PMF 
    \begin{equation}
      \mathbb{P}(S_n = m) = \binom{m - 1}{n - 1} p^n (1 - p)^{m - n} \text{ for } m \geq n
    \end{equation}
    That is, $S_n - n \sim \NegBinomial(n, 1 - p)$, the \hyperref[def:negative-binomial-distribution]{negative binomial} counting the number of failures before the $n$th success. 
  \end{theorem}
  \begin{proof}
    We induct on $n$. For $n = 1$ the formula is the geometric PMF. Suppose it holds for $n$. Convolving $S_n$ with $X_{n+1}$, for $m \geq n + 1$, 
    \begin{align*}
      \mathbb{P}(S_{n+1} = m) & = \sum_{j=n}^{m-1} \mathbb{P}(S_n = j) \, \mathbb{P}(X_{n+1} = m - j) \\
      & = \sum_{j=n}^{m-1} \binom{j-1}{n-1} p^n (1 - p)^{j - n} \cdot (1 - p)^{m - j - 1} p \\
      & = p^{n+1} (1 - p)^{m - n - 1} \sum_{j=n}^{m-1} \binom{j-1}{n-1} = \binom{m-1}{n} p^{n+1} (1 - p)^{m - (n+1)}
    \end{align*}
    where the last step is the hockey-stick identity $\sum_{j=n}^{m-1} \binom{j-1}{n-1} = \binom{m-1}{n}$. This is the claimed PMF for $n + 1$. Finally, setting $k = m - n$ gives $\mathbb{P}(S_n - n = k) = \binom{k + n - 1}{k} p^n (1-p)^k$, which is the PMF of $\NegBinomial(n, 1-p)$. 
  \end{proof}

  \begin{theorem}[Sum of Independent Binomials is Binomial]
    \label{thm:sum-of-independent-binomials-is-binomial}
    If $X \sim \Binomial(n, p)$ and $Y \sim \Binomial(m, p)$ are independent, then $X + Y \sim \Binomial(n + m, p)$. 
  \end{theorem}
  \begin{proof}
    Convolving the PMFs, for $0 \leq k \leq n + m$, 
    \begin{align*}
      \mathbb{P}(X + Y = k) & = \sum_{j=0}^{k} \binom{n}{j} p^j (1 - p)^{n - j} \binom{m}{k - j} p^{k - j} (1 - p)^{m - k + j} \\
      & = p^k (1 - p)^{n + m - k} \sum_{j=0}^k \binom{n}{j} \binom{m}{k - j} = \binom{n + m}{k} p^k (1 - p)^{n + m - k}
    \end{align*}
    by Vandermonde's identity. This is the PMF of a $\Binomial(n + m, p)$. 
  \end{proof}

  \begin{theorem}[Sum of Independent Poissons is Poisson]
    \label{thm:sum-of-independent-poissons-is-poisson}
    If $X \sim \Poisson(\lambda)$ and $Y \sim \Poisson(\gamma)$ are independent, then $X + Y \sim \Poisson(\lambda + \gamma)$. 
  \end{theorem}
  \begin{proof}
    This is exactly the convolution computed in \hyperref[ex:sums-of-poisson-rvs]{the example above} with $\lambda_1 = \lambda$ and $\lambda_2 = \gamma$, which gives the PMF $\frac{e^{-(\lambda + \gamma)} (\lambda + \gamma)^z}{z!}$ of a $\Poisson(\lambda + \gamma)$. 
  \end{proof}

  \begin{theorem}[Sum of Independent Gaussians is Gaussian]
    \label{thm:sum-of-independent-gaussians-is-gaussian}
    If $X \sim \Normal(\mu_1, \sigma_1^2)$ and $Y \sim \Normal(\mu_2, \sigma_2^2)$ are independent, then $X + Y \sim \Normal(\mu_1 + \mu_2, \sigma_1^2 + \sigma_2^2)$. 
  \end{theorem}
  \begin{proof}
    Shifting $X$ and $Y$ by their means only shifts the sum by $\mu_1 + \mu_2$, so we may assume $\mu_1 = \mu_2 = 0$. Let $\sigma^2 = \sigma_1^2 + \sigma_2^2$. The convolution of the \hyperref[def:gaussian-random-variable]{Gaussian} densities is 
    \begin{equation}
      f_{X+Y} (z) = \frac{1}{2 \pi \sigma_1 \sigma_2} \int_{-\infty}^\infty \exp \bigg( -\frac{x^2}{2 \sigma_1^2} - \frac{(z - x)^2}{2 \sigma_2^2} \bigg) \,dx
    \end{equation}
    Completing the square in $x$, with $a = \frac{1}{\sigma_1^2} + \frac{1}{\sigma_2^2} = \frac{\sigma^2}{\sigma_1^2 \sigma_2^2}$, 
    \begin{equation}
      \frac{x^2}{\sigma_1^2} + \frac{(z - x)^2}{\sigma_2^2} = a \bigg( x - \frac{\sigma_1^2 z}{\sigma^2} \bigg)^2 + \frac{z^2}{\sigma^2}
    \end{equation}
    Since $\int_{-\infty}^\infty e^{-a (x - c)^2 / 2} \,dx = \sqrt{2 \pi / a} = \frac{\sqrt{2 \pi} \, \sigma_1 \sigma_2}{\sigma}$, we get 
    \begin{equation}
      f_{X+Y} (z) = \frac{1}{2 \pi \sigma_1 \sigma_2} \cdot \frac{\sqrt{2 \pi} \, \sigma_1 \sigma_2}{\sigma} \, e^{-z^2 / 2 \sigma^2} = \frac{1}{\sigma \sqrt{2 \pi}} e^{-z^2 / 2 \sigma^2}
    \end{equation}
    which is the density of a $\Normal(0, \sigma_1^2 + \sigma_2^2)$. 
  \end{proof}

  \begin{theorem}[Sum of Independent Gamma and Exponential is Gamma]
    \label{thm:sum-of-independent-gamma-and-exponential-is-gamma}
    If $X \sim \GammaDist(n, \lambda)$ and $Y \sim \Exponential(\lambda)$ are independent, then $X + Y \sim \GammaDist(n + 1, \lambda)$. 
  \end{theorem}
  \begin{proof}
    Both densities are supported on $[0, \infty)$, so for $z \geq 0$ the \hyperref[def:convolution]{convolution} is 
    \begin{align*}
      f_{X+Y} (z) & = \int_0^z \frac{\lambda^n x^{n-1}}{\Gamma(n)} e^{-\lambda x} \cdot \lambda e^{-\lambda (z - x)} \,dx = \frac{\lambda^{n+1} e^{-\lambda z}}{\Gamma(n)} \int_0^z x^{n-1} \,dx \\
      & = \frac{\lambda^{n+1} e^{-\lambda z}}{\Gamma(n)} \cdot \frac{z^n}{n} = \frac{\lambda^{n+1} z^n}{\Gamma(n+1)} e^{-\lambda z}
    \end{align*}
    using $\Gamma(n+1) = n \Gamma(n)$. This is the density of a $\GammaDist(n+1, \lambda)$. 
  \end{proof}

  \begin{theorem}[Sum of Independent Exponentials is Gamma]
    \label{thm:sum-of-independent-exponentials-is-gamma}
    If $X_1, \ldots, X_n$ are independent $\Exponential(\lambda)$ \hyperref[def:exponential-random-variable]{random variables}, then $X_1 + \cdots + X_n \sim \GammaDist(n, \lambda)$. 
  \end{theorem}
  \begin{proof}
    We induct on $n$. For $n = 1$, $\Exponential(\lambda) = \GammaDist(1, \lambda)$ since both have density $\lambda e^{-\lambda x}$. If $X_1 + \cdots + X_n \sim \GammaDist(n, \lambda)$, then adding the independent $X_{n+1} \sim \Exponential(\lambda)$ gives $\GammaDist(n+1, \lambda)$ by the \hyperref[thm:sum-of-independent-gamma-and-exponential-is-gamma]{previous theorem}. For $n = 2$ and $\lambda_1 = \lambda_2$ this recovers the Erlang density computed in \hyperref[ex:sum-of-two-independent-exponentials]{the example above}. 
  \end{proof}

  \begin{theorem}[Sum of Independent Gammas is Gamma]
    \label{thm:sum-of-independent-gammas-is-gamma}
    If $X \sim \GammaDist(n, \lambda)$ and $Y \sim \GammaDist(m, \lambda)$ are independent, then $X + Y \sim \GammaDist(n + m, \lambda)$. 
  \end{theorem}
  \begin{proof}
    For $z \geq 0$, the \hyperref[def:convolution]{convolution} of the \hyperref[def:gamma-random-variable]{gamma} densities is 
    \begin{equation}
      f_{X+Y} (z) = \frac{\lambda^{n+m} e^{-\lambda z}}{\Gamma(n) \Gamma(m)} \int_0^z x^{n-1} (z - x)^{m-1} \,dx
    \end{equation}
    Substituting $x = z t$, the integral becomes $z^{n+m-1} \int_0^1 t^{n-1} (1 - t)^{m-1} \,dt = z^{n+m-1} B(n, m) = z^{n+m-1} \frac{\Gamma(n) \Gamma(m)}{\Gamma(n + m)}$, where $B$ is the normalizing constant of the \hyperref[def:beta-random-variable]{Beta distribution}. Therefore, 
    \begin{equation}
      f_{X+Y} (z) = \frac{\lambda^{n+m} z^{n+m-1}}{\Gamma(n + m)} e^{-\lambda z}
    \end{equation}
    which is the density of a $\GammaDist(n + m, \lambda)$. 
  \end{proof}

  Now we consider a \hyperref[def:random-variable]{random variable} where the number of terms we are summing is a random variable. Let $\{X_i\}_i$ be a countable sequence of \hyperref[def:independent-random-variables]{independent random variables} with CDF $F_{X_i}$. Let $N$ be a positive integer-valued random variable with PMF $p_N(n) = \mathbb{P}(N = n)$. Assume that $N$ is independent of $X_i$'s. Now, consider the function 
  \begin{equation}
    S_N \coloneqq \sum_{i=1}^N X_i
  \end{equation}
  To interpret this, consider the sample space $\Omega$. We have all $X_i$'s and $N$ defined on the same $\Omega$. Once $\omega \in \Omega$ realizes, the $\{X_i\}$'s will realize as a sequence of numbers, and $N$ will realize as a positive integer. We simply sum them up according to the rule $S_N$, and by this definition, $S_N$ is a real-valued function on $\Omega$. We first have to prove that $S_N$ is a random variable (since we only know that a \textit{fixed} sum of random variables is a random variable), and then we must find the CDF of $S_N$ $\mathbb{P}(S_N \leq x)$. 

  First, note that the realization of $N$ partitions the sample space as 
  \begin{equation}
    \Omega = \bigsqcup_{n = 1}^\infty \{\omega \mid N(\omega) = n\}
  \end{equation}
  Once I have this partition, I can invoke the partition rule and write 
  \begin{align*}
    \mathbb{P}(S_N \leq x) & = \sum_{k=1}^\infty \mathbb{P}(S_N \leq x \mid N = k) \, \mathbb{P}(N = k) \\
    & = \sum_{k=1}^\infty \mathbb{P}(S_k \leq x \mid N = k) \, \mathbb{P}(N = k) & (\text{conditioned on } N = k) \\
    & = \sum_{k=1}^\infty \mathbb{P}(S_k \leq x) \, \mathbb{P}(N = k) & (N \text{ is indep. of } X_i \text{s})
  \end{align*}
  where $\mathbb{P}(N = k)$ is known since we know the PMF of $N$, and the CDFs $\mathbb{P}(S_k \leq x)$ can be computed by computing the deterministic sums and computing their CDF. 

  \begin{example}[Geometric Sum of Exponentials]
    \label{ex:geometric-sum-of-exponentials}
    Let $X_i$'s be iid $\Exponential(\lambda)$, and $N \sim \Geometric(p)$. We know that the deterministic sum of iid exponentials gives an Erlang. So, $S_N = \sum_{i=1}^N X_i$, and its CDF is 
    \begin{equation}
      \mathbb{P}(S_N \leq x) = \sum_{k=1}^\infty \mathbb{P}(S_k \leq x) \, \mathbb{P}(N = k)
    \end{equation}
    where $\mathbb{P}(N = k) = (1 - p)^{k - 1} p$. The PDF of the Erlang is 
    \begin{equation}
      p_{S_k} (x) = \frac{\lambda^n x^{n-1}}{(n - 1)!} e^{-\lambda x}
    \end{equation}
    and doing the brute force calculations gives a clean $S_N \sim \Exponential(\lambda p)$. 
  \end{example}

\subsection{Order Statistics}

	We would like to generalize the minimum and maximum functions of \hyperref[def:random-variable]{random variables}. 

	\begin{definition}[Minimum/Maximum of Random Variables]
	  \label{def:minimum-maximum-of-random-variables}
		Let $X_1, X_2, \ldots, X_n$ be \hyperref[def:random-variable]{random variables} of $(\Omega, \mathcal{F}, \mathbb{P})$. Then, $Y_n = \min\{X_1, \ldots, X_n\}$ and $Z_n = \max\{X_1, \ldots, X_n\}$\footnote{Note that $Y_n$ and $Z_n$ are also functions of $\Omega$ to $\mathbb{R}$.} are random variables. 
	\end{definition}
	\begin{proof}
		The random vector has joint CDF $F_{X_1 \ldots X_n} (x_1, \ldots, x_n)$. To prove that they are \hyperref[def:random-variable]{random variables}, we just have to prove that $\min$ and $\max$ are \hyperref[mt-def:measurable-function]{measurable functions} from $\mathbb{R}^n$ to $\mathbb{R}$, which we can do by proving that the preimage of all semi-infinite interval $(-\infty, x]$ are Borel in $\mathbb{R}^n$. 
		\begin{enumerate}
			\item The preimage of $(-\infty, x]$ under $\max$ is just the set of all $n$-vectors whose max is less than $x$, which is just the semi-infinite cuboid $(-\infty, x]^n \subset \mathbb{R}^n$, which is Borel in $\mathbb{R}^n$. 
			\item The preimage of $(-\infty, x]$ under $\min$ is the set of all $n$-vectors whose min is less than $x$, i.e. at least one element must be less than $x$. But this is just the complement of all vectors that have elements all greater than $x$, which is $\mathbb{R}^n \setminus (x, +\infty)^n \subset \mathbb{R}^n$, which is Borel in $\mathbb{R}^n$. 
		\end{enumerate}
	\end{proof}


  Now we must determine the CDF of $Y_n$ and $Z_n$. 
  \begin{enumerate}
    \item We have 
    \begin{align*}
      F_{Z_n} (z) & = \mathbb{P}(\{ \omega \mid Z_n (\omega) \leq z \}) \\
      & = \mathbb{P}(\{ \omega \mid X_1 (\omega) \leq z, \ldots, X_n (\omega) \leq z\}) \\
      & = F_{X_1 \ldots X_n} (z, \ldots, z)
    \end{align*}
    where the last equality is describes simply the joint CDF of the joint distribution $(X_1, \ldots, X_n)$. If we assume independence of $X_i$'s, it simplifies out to 
    \begin{equation}
      \prod_{i} F_{X_i} (z)
    \end{equation}
    and if iid, then we have $[F_{X} (z) ]^n$, where $X$ is the common distribution. 
    \item For $Y_n$, we work with complements again and have 
    \begin{align*}
      F_{Y_n} (y) & = \mathbb{P}(\{ \omega \mid Y_n (\omega) \leq y \}) \\ 
      & = 1 - \mathbb{P}(\{ \omega \mid Y_n (\omega) > y \}) \\
      & = 1 - \mathbb{P}(\{ \omega \mid X_1 (\omega) > y, \ldots X_n (\omega) > y \}) \\
    \end{align*}
    where $\mathbb{P}(\{ \omega \mid X_1 (\omega) > y, \ldots X_n > y \})$ can be calculated from the joint distribution. If we assume independence of $X_i$, it simplifies out to 
    \begin{equation}
      1 - \prod_{i} \mathbb{P}(\{\omega \mid X_i(\omega) > y \}) = 1 - \prod_{i} \big( 1 - F_{X_i} (y) \big)
    \end{equation}
    and if iid, then we have $1 - [1 - F_{X} (y)]^n$. 
  \end{enumerate}

  \begin{example}[Uniforms]
    \label{ex:uniforms}
    Let $X_1, X_2$ be iid distributed as $\Uniform[0, 1]$, and let $Z = \max(X_1, X_2)$ with $Y = \min(X_1, X_2)$, i.e. $Z$ is the greater of the two and $Y$ is the lesser. We would expect the PDF of $Z$ to have more mass towards $1$ and the PDF of $Y$ to have more mass towards $0$. Our common CDF is 
    \begin{equation}
      F_{X} (x) = \begin{cases} 0 & \text{ if } x < 0 \\
      x & \text{ if } 0 \leq x \leq 1 \\
      1 & \text{ if } 1 < x \end{cases}
    \end{equation}
    Let's calculate the CDF of $Z$. 
    \begin{align*}
      F_{Z} (z) & = \mathbb{P}(\{\omega \mid Z(\omega) \leq z\}) \\
      & = \mathbb{P}(\{ \omega \mid X_1 (\omega) \leq z, X_2 (\omega) \leq z\}) \\
      & = F_{X_1, X_2} (z, z) \\
      & = [F_{X} (z)]^2 = \begin{cases} 0 & \text{ if } x < 0 \\
      x^2 & \text{ if } x \in [0, 1] \\
      1 & \text{ if } 1 < x \end{cases}
    \end{align*}
    This CDF is differentiable everywhere except the two points $0$ and $1$, so we can get the PDF to be $f_Z (z) = 2 z$ for $z \in (0, 1)$ and $0$ otherwise. For the values of $f_Z$ at $0$ and $1$, we can fill it in with anything we want (since the measure of these sets are $0$), so we will just defined $f_Z (0) = 0$ and $f_Z(1) = 2$, getting 
    \begin{equation}
      f_Z (z) = \begin{cases} 2 z & \text{ if } z \in [0, 1] \\
      0 & \text{ if else} \end{cases}
    \end{equation}
    Let's calculate the CDF of $Y$. 
    \begin{align*}
      F_{Y} (y) & = \mathbb{P}(\{ \omega \mid Y(\omega) \leq y\}) \\
      & = 1 - \mathbb{P}(\{ \omega \mid Y(\omega) > y\}) \\
      & = 1 - \mathbb{P}(\{\omega \mid X_1 (\omega) > y, X_2 (\omega) > y \}) \\
      & = 1 - \mathbb{P}(\{\omega \mid X_1 (\omega) > y\}) \, \mathbb{P}(\{ X_2 (\omega) > y \}) \\ 
      & = 1 - [1 - F_X (y)]^2 = \begin{cases} 0 & \text{ if } y < 0 \\
      1 - (1 - y)^2 & \text{ if } y \in [0, 1] \\
      1 & \text{ if } y > 1 \end{cases} 
    \end{align*}
    and differentiating it (with setting any values of the PDF at the nondifferentiable points $0$ and $1$) gives 
    \begin{equation}
      f_Y (y) = \begin{cases} 2 - 2y & \text{ if } y \in [0, 1] \\
      0 & \text{ if else} \end{cases}
    \end{equation}
  \end{example}

  \begin{example}[Exponentials]
    \label{ex:exponentials}
    Let $X_1, X_2, \ldots, X_n$ be independent \hyperref[def:exponential-random-variable]{exponential random variables} with parameters $\lambda_1, \ldots, \lambda_n$, respectively (not identical!). Then, for each $X_i$, its CDF is 
    \begin{equation}
      F_{X_i} (x) = 1 - e^{-\lambda_i x} \text{ for } x \geq 0
    \end{equation}
    and let $Y = \min(X_1, \ldots, X_n)$. Then, we have 
    \begin{align*}
      F_Y (y) & = 1 - \prod_{i=1}^n [ 1 - F_{X_i} (y)] \\
      & = 1 - \prod_{i=1}^n e^{-\lambda_i x} \\
      & = 1 - e^{- ( \sum_{i=1}^n \lambda_i ) x}
    \end{align*}
    which is the CDF of an exponential distribution. So, 
    \begin{equation}
      Y \sim \Exponential(\lambda_1 + \ldots + \lambda_n)
    \end{equation}
    This is nice, since the minimum of a bunch of exponentials is an exponential. However, this is not the case for the maximum. 
  \end{example}

  This has nice practical applications. For example, recall the memoryless property of the exponential, which nicely models radioactive decay. If we have $n$ elements each decaying at some $\Exponential(\lambda_i)$ rate, then we can model the time at which the first alpha particle will emit amongst all $n$ elements will also be an exponential. These processes where the inter-emission times are exponentials are called Poisson process, which we will discuss later. 

  \begin{definition}[Order Statistic]
    \label{def:order-statistic}
    Let $X_1, X_2, ..., X_n$ be a finite collection of independent, identically distributed \hyperref[def:random-variable]{random variables}. Suppose that they are continuously distributed with density $f$ and CDF $F$. Define the random variable $X_{(k)}$ to be the $k$th ranked value, called the \textbf{$k$th order statistic}. This means that 
    \begin{equation}
      X_{(1)} = \min\{X_1, X_2, ..., X_n\}, \;\; X_{(n)} = \max\{X_1, X_2, ..., X_n\}
    \end{equation}
    and in general, for any $k \in \{1, 2, ..., n\}$, 
    \begin{equation}
      X_{(k)} = X_j \text{ if } \sum_{l=1}^n \mathbb{I}_{X_l < X_j} = k - 1
    \end{equation}
    which means that exactly $k-1$ of the values of $X_l$ are less than $X_j$. Since $F$ is continuous, 
    \begin{equation}
      X_{(1)} < X_{(2)} < ... < X_{(n)}
    \end{equation}
    holds with probability $1$. This leads us to define the random variable $X_{(k)}$ representing the $k$th order statistic.
    \begin{equation}
      f_{(k)} (y) = \begin{cases} 
      n \, \binom{n-1}{k-1} y^{k-1} (1-y)^{n-k} & y \in (0, 1) \\
      0 & y \not\in (0,1)
      \end{cases}
    \end{equation}
    That is, $X_{(k)}$ has the Beta$(k, n-k_1)$ distribution. 
  \end{definition}

\subsection{Probability Generating Function}

	The PGF is only defined for \hyperref[def:discrete-random-variable]{discrete random variable}, and is analogous to the Z-transform in singal processing. 

	\begin{definition}[Probability Generating Function]
	  \label{def:probability-generating-function}
		Let $X$ be a \hyperref[def:discrete-random-variable]{discrete random variable} taking values in $\mathbb{N}_0$. Then, the \textbf{probability generating function} of $X$ is defined 
		\begin{equation}
			G_X (z) \coloneqq \mathbb{E}[z^X] = \sum_{i=0}^\infty z^i \, \mathbb{P}(X = i)
		\end{equation}
		Now there is the problem of convergence, but we will not pay attention to this technicality for now and just consider the PGF as a tool. 
	\end{definition}

	\begin{theorem}[PGF of Bernoulli Distribution]
	  \label{thm:pgf-of-bernoulli-distribution}
		The PGF of \hyperref[def:indicator-bernoulli-random-variable]{$X \sim \Bernoulli(p)$} is 
		\begin{equation}
			G_X (z) = 1 - p + p z \text{ for all } z \in \mathbb{R}
		\end{equation}
	\end{theorem}
	\begin{proof}
		Since $X$ only takes values in $\{0, 1\}$ with $\mathbb{P}(X = 0) = 1 - p$ and $\mathbb{P}(X = 1) = p$, 
		\begin{equation}
			G_X (z) = \mathbb{E}[z^X] = z^0 (1 - p) + z^1 p = 1 - p + p z
		\end{equation}
	\end{proof}

	\begin{theorem}[PGF of Binomial Distribution]
	  \label{thm:pgf-of-binomial-distribution}
		The PGF of \hyperref[def:binomial-random-variable]{$X \sim \Binomial(n, p)$} is 
		\begin{equation}
			G_X (z) = (1 - p + p z)^n \text{ for all } z \in \mathbb{R}
		\end{equation}
	\end{theorem}
	\begin{proof}
		By the binomial theorem, 
		\begin{equation}
			G_X (z) = \sum_{k=0}^n z^k \binom{n}{k} p^k (1 - p)^{n-k} = \sum_{k=0}^n \binom{n}{k} (p z)^k (1 - p)^{n-k} = (p z + 1 - p)^n
		\end{equation}
	\end{proof}

	\begin{theorem}[PGF of Geometric Distribution]
	  \label{thm:pgf-of-geometric-distribution}
		The PGF of \hyperref[def:geometric-random-variable]{$X \sim \Geometric(p)$} is 
		\begin{equation}
			G_X (z) = \frac{p z}{1 - z(1 - p)} \text{ for } |z| < \frac{1}{1 - p}
		\end{equation}
	\end{theorem}
	\begin{proof}
		Since $X$ takes values in $\mathbb{N}$ with $\mathbb{P}(X = i) = (1-p)^{i-1} p$, we have a geometric series with ratio $z(1-p)$, which converges when $|z(1-p)| < 1$: 
		\begin{equation}
			G_X (z) = \sum_{i=1}^\infty z^i \, (1 - p)^{i-1} p = p z \sum_{i=0}^\infty \big( z (1 - p) \big)^i = \frac{p z}{1 - z(1 - p)}
		\end{equation}
	\end{proof}

	\begin{theorem}[PGF of Negative Binomial Distribution]
	  \label{thm:pgf-of-negative-binomial-distribution}
		The PGF of \hyperref[def:negative-binomial-distribution]{$X \sim \NegBinomial(r, p)$} is 
		\begin{equation}
			G_X (z) = \bigg( \frac{1 - p}{1 - p z} \bigg)^r \text{ for } |z| < \frac{1}{p}
		\end{equation}
	\end{theorem}
	\begin{proof}
		Recall the negative binomial series $\sum_{k=0}^\infty \binom{k + r - 1}{k} x^k = (1 - x)^{-r}$ for $|x| < 1$. Then, 
		\begin{equation}
			G_X (z) = \sum_{k=0}^\infty z^k \binom{k + r - 1}{k} (1 - p)^r p^k = (1 - p)^r \sum_{k=0}^\infty \binom{k + r - 1}{k} (p z)^k = \frac{(1 - p)^r}{(1 - p z)^r}
		\end{equation}
		which converges when $|p z| < 1$. 
	\end{proof}

	\begin{theorem}[PGF of Poisson Distribution]
	  \label{thm:pgf-of-poisson-distribution}
		The PGF of \hyperref[def:poisson-random-variable]{$X \sim \Poisson(\lambda)$} is 
		\begin{equation}
			G_X (z) = e^{\lambda(z - 1)} \text{ for all } z \in \mathbb{R}
		\end{equation}
	\end{theorem}
	\begin{proof}
		By definition and the Taylor series of the exponential, 
		\begin{equation}
			G_X (z) = \mathbb{E}[z^X] = \sum_{i=0}^\infty z^i \, \frac{e^{-\lambda} \lambda^i}{i!} = e^{-\lambda} \sum_{i=0}^\infty \frac{(\lambda z)^i}{i!} = e^{-\lambda} e^{\lambda z} = e^{\lambda(z - 1)}
		\end{equation}
	\end{proof}

	\begin{lemma}[Properties of PGF]
	  \label{thm:properties-of-pgf}
		Given \hyperref[def:random-variable]{random variable} $X$ and its PGF $G_X$, we have the following: 
		\begin{enumerate}
			\item Evaluate at $z = 1$: 
			\begin{equation}
				G_X (1) = \mathbb{E}[1^X] = \mathbb{E}[1] = 1
			\end{equation}
			\item Derivative at $z = 1$: 
			\begin{equation}
				\frac{d G_X (z)}{d z} \bigg|_{z = 1} = \mathbb{E}[X]
			\end{equation}
			\item $k$th derivative at $z = 1$: 
			\begin{equation}
				\frac{d^k G_X (z)}{d z^k} \bigg|_{z = 1} = \mathbb{E}[X (X-1) (X-2) \ldots (X-k +1)]
			\end{equation}
			\item Transformation: Given the sum $Z = X + Y$ (where $X, Y$ are independent), rather than computing its \hyperref[def:convolution]{convolution}, the PGF of $Z$ is simply the product of the PGFs of $X$ and $Y$: 
			\begin{equation}
				G_Z (z) = G_X (z) \, G_Y (z)
			\end{equation}
			For example, since a $\Poisson(\lambda)$ random variable has PGF of form $e^{\lambda (z - 1)}$, if we have two Poissons $X$ and $Y$ with parameters $\lambda, \mu$, then we can easily multiply their PGFs to get the PGF of $Z = X + Y$, which is $e^{(\lambda + \mu)(z - 1)}$, which is the PGF of a $\Poisson(\lambda + \mu)$ random variable. 
		\end{enumerate}
	\end{lemma}

\subsection{Moment Generating Function}

	\begin{definition}[Moment Generating Function (MGF)]
	  \label{def:moment-generating-function-mgf}
		The \textbf{moment generating function} associated with a \hyperref[def:random-variable]{random variable} $X$ is a function $M_X: \mathbb{R} \to [0, \infty]$ defined 
		\begin{equation}
			M_X (s) \coloneqq \mathbb{E}[e^{s X}]
		\end{equation}
		It is like an exponential moment. The region of convergence of $M_X$ is the set $D_X = \{s \mid M_X (s) < \infty\}$. 
		and we always have $M_X (0) = 1$, so $0 \in D_X$ always. 
	\end{definition}

	\begin{lemma}[Properties of MGF]
	  \label{thm:properties-of-mgf}
		Let $X$ be a \hyperref[def:random-variable]{random variable} with MGF $M_X (s)$. 
		\begin{enumerate}
			\item \textit{Converges at $0$}. $M_X (0) = 1$, so $0$ is always in the region of convergence. 

			\item \textit{Affine Precomposition}. If $Y = a X + b$, then 
				\begin{equation}
					M_Y (s) = e^{b s} M_X (a s)
				\end{equation}

			\item \textit{Addition of Independent RVs}. If $X$ and $Y$ are independent and $Z = X + Y$, then 
				\begin{equation}
					M_Z (s) = M_X (s) \, M_Y (s)
				\end{equation}
		\end{enumerate}
	\end{lemma}
	\begin{proof}
		Listed. 
		\begin{enumerate}
			\item $M_X (0) = \mathbb{E}[e^{0 X}] = \mathbb{E}[1] = 1$. 
			\item We have 
			\begin{align*}
				M_Y (s) & = \mathbb{E} [e^{s(a X + b)}] \\
				& = \mathbb{E}[ e^{a s X } e^{b s}] \\
				& = \mathbb{E}[e^{(as) X}] \, \mathbb{E}[e^{b s}] \\
				& = e^{b s} M_X (a s)
			\end{align*}
			where the penultimate step was due to independence of constant RV with any other RVs. 
			\item We can see 
			\begin{equation}
				M_Z (s) = \mathbb{E}[ e^{s (X + Y)}] = \mathbb{E}[e^{s X} \, e^{s Y}] = \mathbb{E}[e^{s X}] \, \mathbb{E}[e^{s Y}] = M_X (s) \, M_Y (s)
			\end{equation}
			since $X, Y$ independent means that any function of $X$ and $Y$ are independent. 
		\end{enumerate}
	\end{proof}

	\begin{theorem}[Inversion Theorem]
	  \label{thm:inversion-theorem}
		Suppose $M_X (s)$ is finite for all $s \in [-\epsilon, \epsilon]$ for some $\epsilon > 0$. Then, $M_X$ uniquely determines the CDF of $X$. This implies that if $X$ and $Y$ are \hyperref[def:random-variable]{random variables} such that $M_X (s) = M_Y (s)$ for all $s \in [-\epsilon, \epsilon]$ for some $\epsilon > 0$, then $X$ and $Y$ have the same CDF. 
	\end{theorem}
	\begin{proof}
	  
	\end{proof}

	This theorem is useful for comparing \hyperref[def:random-variable]{random variables} with the MGFs, but a limitation is that it is not always clear that the MGF is defined beyond $0$. 

	\begin{theorem}[Moment Generating Property]
	  \label{thm:moment-generating-property}
		Suppose $M_X (s) < \infty$ for $s \in [-\epsilon, \epsilon]$ with $\epsilon > 0$. Then, the derivatives at $s = 0$ generate the moments of $X$: 
		\begin{equation}
			\frac{d^m M_X (s)}{d s^m} \bigg|_{s = 0} = \mathbb{E}[X^m]
		\end{equation}
	\end{theorem}
	\begin{proof}
		A hand-wavy proof is that we can take the derivative and put it "in" the expectation. 
		\begin{equation}
			\frac{d}{ds} \mathbb{E}[e^{s X}] = \mathbb{E} \big[ \frac{d}{ds} e^{s X} \big] = \mathbb{E}[X e^{s X}]
		\end{equation}
		which evaluates to $\mathbb{E}[X]$ when $s = 0$. Differentiating $m$ times just gets $\mathbb{E}[X^m e^{s X}]$. However, this should be questioned, since the expectation is an integral and we are putting the derivative inside the integral. 
	\end{proof}

	\begin{theorem}[MGF of Bernoulli Distribution]
	  \label{thm:mgf-of-bernoulli-distribution}
		The MGF of \hyperref[def:indicator-bernoulli-random-variable]{$X \sim \Bernoulli(p)$} is 
		\begin{equation}
			M_X (s) = 1 - p + p e^s \text{ for all } s \in \mathbb{R}
		\end{equation}
	\end{theorem}
	\begin{proof}
		Since $\mathbb{P}(X = 0) = 1 - p$ and $\mathbb{P}(X = 1) = p$, 
		\begin{equation}
			M_X (s) = \mathbb{E}[e^{s X}] = e^{0} (1 - p) + e^{s} p = 1 - p + p e^s
		\end{equation}
	\end{proof}

	\begin{theorem}[MGF of Binomial Distribution]
	  \label{thm:mgf-of-binomial-distribution}
		The MGF of \hyperref[def:binomial-random-variable]{$X \sim \Binomial(n, p)$} is 
		\begin{equation}
			M_X (s) = (1 - p + p e^s)^n \text{ for all } s \in \mathbb{R}
		\end{equation}
	\end{theorem}
	\begin{proof}
		By the binomial theorem, 
		\begin{equation}
			M_X (s) = \sum_{k=0}^n e^{s k} \binom{n}{k} p^k (1 - p)^{n-k} = \sum_{k=0}^n \binom{n}{k} (p e^s)^k (1 - p)^{n-k} = (p e^s + 1 - p)^n
		\end{equation}
	\end{proof}

	\begin{theorem}[MGF of Geometric Distribution]
	  \label{thm:mgf-of-geometric-distribution}
		The MGF of \hyperref[def:geometric-random-variable]{$X \sim \Geometric(p)$} is 
		\begin{equation}
			M_X (s) = \frac{p e^s}{1 - (1 - p) e^s} \text{ for } s < -\ln(1 - p)
		\end{equation}
		and $M_X (s) = \infty$ for $s \geq -\ln(1-p)$. 
	\end{theorem}
	\begin{proof}
		We have a geometric series with ratio $(1 - p) e^s$, 
		\begin{equation}
			M_X (s) = \sum_{k=1}^\infty e^{s k} (1 - p)^{k-1} p = p e^s \sum_{k=0}^\infty \big( (1 - p) e^s \big)^k
		\end{equation}
		which converges to $\frac{p e^s}{1 - (1-p) e^s}$ exactly when $(1-p) e^s < 1$, i.e. $s < -\ln(1-p)$, and diverges otherwise. 
	\end{proof}

	\begin{theorem}[MGF of Negative Binomial Distribution]
	  \label{thm:mgf-of-negative-binomial-distribution}
		The MGF of \hyperref[def:negative-binomial-distribution]{$X \sim \NegBinomial(r, p)$} is 
		\begin{equation}
			M_X (s) = \bigg( \frac{1 - p}{1 - p e^s} \bigg)^r \text{ for } s < -\ln p
		\end{equation}
		and $M_X (s) = \infty$ for $s \geq -\ln p$. 
	\end{theorem}
	\begin{proof}
		Using the negative binomial series $\sum_{k=0}^\infty \binom{k + r - 1}{k} x^k = (1 - x)^{-r}$ for $|x| < 1$, 
		\begin{equation}
			M_X (s) = \sum_{k=0}^\infty e^{s k} \binom{k + r - 1}{k} (1 - p)^r p^k = (1 - p)^r \sum_{k=0}^\infty \binom{k + r - 1}{k} (p e^s)^k = \frac{(1 - p)^r}{(1 - p e^s)^r}
		\end{equation}
		which converges exactly when $p e^s < 1$, i.e. $s < -\ln p$. For $s \geq -\ln p$ the terms do not decay to $0$, so the series diverges. 
	\end{proof}

	\begin{theorem}[MGF of Poisson Distribution]
	  \label{thm:mgf-of-poisson-distribution}
		The MGF of \hyperref[def:poisson-random-variable]{$X \sim \Poisson(\lambda)$} is 
		\begin{equation}
			M_X (s) = e^{\lambda (e^s - 1)} \text{ for all } s \in \mathbb{R}
		\end{equation}
	\end{theorem}
	\begin{proof}
		By the Taylor series of the exponential, 
		\begin{equation}
			M_X (s) = \sum_{k=0}^\infty e^{s k} \frac{e^{-\lambda} \lambda^k}{k!} = e^{-\lambda} \sum_{k=0}^\infty \frac{(\lambda e^s)^k}{k!} = e^{-\lambda} e^{\lambda e^s} = e^{\lambda (e^s - 1)}
		\end{equation}
	\end{proof}

	\begin{theorem}[MGF of Exponential Distribution]
	  \label{thm:mgf-of-exponential-distribution}
		The MGF of \hyperref[def:exponential-random-variable]{$X \sim \Exponential(\lambda)$} is 
		\begin{equation}
			M_X (s) = \begin{cases} 
			\frac{\lambda}{\lambda - s} & \text{ for } s < \lambda \\
			\infty & \text{ for } s \geq \lambda 
			\end{cases}
		\end{equation}
	\end{theorem}
	\begin{proof}
		By definition, 
		\begin{equation}
			M_X (s) = \int_0^\infty \lambda e^{-\lambda x} e^{s x} \, dx = \lambda \int_0^\infty e^{-(\lambda - s) x} \, dx
		\end{equation}
		If $s < \lambda$, the exponent is negative and the integral evaluates to $\frac{1}{\lambda - s}$, giving $\frac{\lambda}{\lambda - s}$. If $s \geq \lambda$, the integrand is at least $1$ on $[0, \infty)$, so the integral diverges. 
	\end{proof}

	\begin{theorem}[MGF of Gaussian Distribution]
	  \label{thm:mgf-of-gaussian-distribution}
		The MGF of \hyperref[def:gaussian-random-variable]{$X \sim \Normal(\mu, \sigma^2)$} is 
		\begin{equation}
			M_X (s) = \exp \bigg( \mu s + \frac{\sigma^2 s^2}{2} \bigg) \text{ for all } s \in \mathbb{R}
		\end{equation}
		In particular, the standard Gaussian has $M_X (s) = e^{s^2 / 2}$. 
	\end{theorem}
	\begin{proof}
		Completing the square in the exponent, 
		\begin{equation}
			s x - \frac{(x - \mu)^2}{2 \sigma^2} = - \frac{(x - \mu - \sigma^2 s)^2}{2 \sigma^2} + \mu s + \frac{\sigma^2 s^2}{2}
		\end{equation}
		so 
		\begin{equation}
			M_X (s) = \int_{-\infty}^\infty \frac{1}{\sigma \sqrt{2 \pi}} e^{s x} e^{-\frac{(x - \mu)^2}{2 \sigma^2}} \,dx = e^{\mu s + \sigma^2 s^2 / 2} \int_{-\infty}^\infty \frac{1}{\sigma \sqrt{2\pi}} e^{-\frac{(x - \mu - \sigma^2 s)^2}{2 \sigma^2}}\,dx = e^{\mu s + \sigma^2 s^2 / 2}
		\end{equation}
		since the remaining integrand is the PDF of a $\Normal(\mu + \sigma^2 s, \sigma^2)$ random variable, which integrates to $1$. 
	\end{proof}

	\begin{theorem}[MGF of Cauchy Distribution]
	  \label{thm:mgf-of-cauchy-distribution}
		The MGF of a (standard) \hyperref[def:cauchy-random-variable-standardized]{Cauchy random variable} $X$ is 
		\begin{equation}
			M_X (s) = \begin{cases} 1 & \text{ if } s = 0 \\
			\infty & \text{ if } s \neq 0 \end{cases}
		\end{equation}
		So the region of convergence is just $\{0\}$. 
	\end{theorem}
	\begin{proof}
		At $s = 0$, $M_X (0) = 1$ always. For $s \neq 0$, 
		\begin{equation}
			M_X (s) = \int_{-\infty}^\infty \frac{e^{s x}}{\pi (1 + x^2)} \,dx
		\end{equation}
		and since the \hyperref[real-def:exponential-function]{exponential function} grows faster than any polynomial, $\frac{e^{s x}}{1 + x^2} \to \infty$ as $x \to \infty$ (if $s > 0$) or as $x \to -\infty$ (if $s < 0$). The integrand is then unbounded on a half-line, so the integral diverges. 
	\end{proof}

	\begin{theorem}[MGF of Gamma Distribution]
	  \label{thm:mgf-of-gamma-distribution}
		The MGF of \hyperref[def:gamma-random-variable]{$X \sim \GammaDist(\alpha, \beta)$} is 
		\begin{equation}
			M_X (s) = \bigg( \frac{\beta}{\beta - s} \bigg)^\alpha \text{ for } s < \beta
		\end{equation}
		and $M_X (s) = \infty$ for $s \geq \beta$. 
	\end{theorem}
	\begin{proof}
		For $s < \beta$, we have 
		\begin{equation}
			M_X (s) = \int_0^\infty e^{s x} \frac{\beta^\alpha x^{\alpha-1}}{\Gamma(\alpha)} e^{-\beta x} \,dx = \frac{\beta^\alpha}{\Gamma(\alpha)} \int_0^\infty x^{\alpha - 1} e^{-(\beta - s) x} \,dx
		\end{equation}
		Substituting $u = (\beta - s) x$, so $dx = du / (\beta - s)$, 
		\begin{equation}
			\int_0^\infty x^{\alpha - 1} e^{-(\beta - s) x} \,dx = \frac{1}{(\beta - s)^\alpha} \int_0^\infty u^{\alpha - 1} e^{-u} \,du = \frac{\Gamma(\alpha)}{(\beta - s)^\alpha}
		\end{equation}
		so $M_X (s) = \frac{\beta^\alpha}{(\beta - s)^\alpha}$. If $s \geq \beta$, the integrand is at least $x^{\alpha - 1}$ on $[0, \infty)$ and the integral diverges. 
	\end{proof}

	\begin{example}[Inverting a Moment Generating Function]
	  \label{ex:inverting-a-moment-generating-function}
		Given $X_1 \sim \Exponential(\lambda_1)$ and $X_2 \sim \Exponential(\lambda_2)$ are independent, the MGF of $Z = X_1 + X_2$ is 
		\begin{equation}
			M_Z (s) = M_X (s) \, M_Y (s) = \frac{\lambda_1 \lambda_2}{(\lambda_1 - s)(\lambda_2 - s)} \text{ for } s < \min\{\lambda_1, \lambda_2\}
		\end{equation}
		and we can perform our inverse transform on it. 
	\end{example}

\subsection{Characteristic Function}

	We can see that the MGF has its limitations: for some \hyperref[def:random-variable]{random variables} (like the Cauchy), its MGF was not defined at all beyond $\{0\}$. On the contrary, the characteristic function is always defined everywhere and is finite everywhere (in fact, is bounded by $1$, shown below). Also, it is a bit easier to invert (similar to how the Fourier transform is a bit easier to invert than the Laplace). 

	\begin{definition}[Characteristic Function]
	  \label{def:characteristic-function}
		Given a \hyperref[def:random-variable]{random variable} $X: \Omega \to \mathbb{R}$, the \textbf{characteristic function} is defined to be 
		\begin{align*}
			\varphi_X (t) & = \mathbb{E}[ e^{i t X} ] \\
			& = \mathbb{E}[\cos{(t X)}] + i \mathbb{E}[ \sin{(t X)}]
		\end{align*}
	\end{definition}

	If $X$ admits a PDF, then the \hyperref[def:characteristic-function]{characteristic function} is its Fourier transform with a small sign reversal. 
	\begin{equation}
		\varphi_X (t) = \int_\mathbb{R} e^{i t x} f_X (x)\,dx
	\end{equation}

	\begin{theorem}[Properties of CF]
	  \label{thm:properties-of-cf}
		Let $X$ be a \hyperref[def:random-variable]{random variable} with CF $\varphi_X (t)$. 
		\begin{enumerate}
			\item $\varphi_X (0) = 1$  and $|\varphi_X (t)| \leq 1$ for all $t \in \mathbb{R}$. 
			\item If $Y = a X + b$, then 
			\begin{equation}
				\varphi_Y (t) = e^{i b t} \varphi_X (a t)
			\end{equation}
			\item If $X$ and $Y$ are \hyperref[def:independent-random-variables]{independent random variables} and $Z = X + Y$, then 
			\begin{equation}
				\varphi_Z (t) = \varphi_X (t) \, \varphi_Y (t)
			\end{equation}
			\item $\varphi_X (t)$ is uniformly continuous on $\mathbb{R}$, i.e. for all $t \in \mathbb{R}$, there exists a $\phi(h) \downarrow 0$ as $h \downarrow 0$ such that 
			\begin{equation}
				|\varphi_X (t + h) - \varphi_X (t)| \leq \phi(h)
			\end{equation}
			\item $\varphi_X$ is a nonnegative-definite kernel, i.e. for any $n$ reals $t_1, \ldots, t_n$ and $n$ complex numbers $z_1, \ldots z_n$, we have 
			\begin{equation}
				\sum_{i, j} z_i \varphi_X (t_i - t_j) \, \bar{z}_j \geq 0
			\end{equation}
		\end{enumerate}
	\end{theorem}
	\begin{proof}
		Listed. 
		\begin{enumerate}
			\item We just set $\varphi_X (0) = \mathbb{E}[e^{i 0 X}] = \mathbb{E}[1] = 1$, and for \hyperref[def:continuous-random-variable]{continuous random variables}, we can bound 
			\begin{align*}
				\big| \varphi_X (t) \big| & = \bigg| \int_{-\infty}^\infty e^{i t x} f_X (x) \,dx  \bigg| \\
				& \leq \int_{-\infty}^\infty \big| e^{i t x} f_X (x) \big| \,dx \\
				& \leq \int_{-\infty}^\infty \big| e^{i t x} \big| \cdot \big|f_X (x) \big| \,dx \\
				& = \int_{-\infty}^\infty f_X (x) \,dx = 1 
			\end{align*}
			
			\item We have 
			\begin{equation}
				\varphi_Y (t) = \mathbb{E}[ e^{i t (a X + b)}] = \mathbb{E}[ e^{i a t X} \, e^{i b t}] = \mathbb{E}[ e^{i (at) X}] \, \mathbb{E}[e^{i b t}] = e^{i b t} \varphi_X (a t)
			\end{equation}
			
			\item We have 
			\begin{equation}
				\varphi_Z (t) = \mathbb{E}[ e^{i t (X + Y)}] = \mathbb{E}[ e^{i t X} \, e^{i t Y}] = \mathbb{E}[e^{i t X}] \, \mathbb{E}[e^{i t Y}] = \varphi_X (t) \, \varphi_Y (t)
			\end{equation}
			
			\item We have 
			\begin{align*}
				| \varphi_X (t + h) - \varphi_X (t)| & = | \mathbb{E} [ e^{i tX} (e^{i h X} - 1)] | \\ 
				& \leq \mathbb{E}[ | e^{i tX} (e^{i h X} - 1)|] \\
				& \leq \mathbb{E}[ |e^{i h X} - 1|] \ldots
			\end{align*}
		\end{enumerate}
	\end{proof}

	Now this next theorem states the uniqueness of each \hyperref[def:characteristic-function]{characteristic function}. It is a highly nontrivial result. 

	\begin{theorem}[Inversion Theorem]
	  \label{thm:inversion-theorem-2}
		If two \hyperref[def:random-variable]{random variables} have the same \hyperref[def:characteristic-function]{characteristic function}, then their CDFs are the same. Further, if $X$ is a \hyperref[def:continuous-random-variable]{continuous random variable}, then the PDF can be recovered from the characteristic function as follows: 
		\begin{equation}
			f_X (x) = \lim_{T \rightarrow \infty} \frac{1}{2 \pi} \int_{-T}^T e^{- i t x} \varphi_X (t) \,dt
		\end{equation}
		for every $x$ where $f_X (x)$ is continuous. 
	\end{theorem}

	Just like how we can recover moments from the MGF, we can always recover the moments from the \hyperref[def:characteristic-function]{characteristic function}, with the added advantage that the CF will always exist. 

	\begin{theorem}[Moment Generating Property]
	  \label{thm:moment-generating-property-2}
		Let $X$ be a \hyperref[def:random-variable]{random variable} and $\varphi_X (t)$ its CF. 
		\begin{enumerate}
			\item If $\varphi_X^{(k)} (t)$ ( the $k$th derivative) exists at $t = 0$, then 
			\begin{align*}
				\mathbb{E}[|X^k|] < \infty & \text{ for } k \text{ even} \\
				\mathbb{E}[|X^{k-1}|] < \infty & \text{ for } k \text{ odd}
			\end{align*}
			
			\item If $\mathbb{E}[|X^k|] < \infty$, then 
			\begin{equation}
				\varphi_X^{(k)} (0) = i^k \mathbb{E}[X^k]
			\end{equation}
			
			\item Further, given that the moments are finite, we can expand the CF by moments of $X$ as 
			\begin{equation}
				\varphi_X (t) = \sum_{j=0}^k \frac{\mathbb{E}[X^j]}{j!} (i t)^j + o(t^k)
			\end{equation}
		\end{enumerate}
	\end{theorem}

	\begin{example}[Bernoulli]
	  \label{ex:bernoulli-2}
		Given $X \sim \Bernoulli(p)$, we have 
		\begin{align*}
			\mathbb{E}[ e^{i t X}] & = \sum_{x \in \{0, 1\}} e^{i t x} \cdot \mathbb{P}(X = x) \\
			& = e^{i t 0} (1 - p) + e^{i t} p \\
			& = 1 - p + p e^{i t}
		\end{align*}
	\end{example}

	\begin{example}[Exponential]
	  \label{ex:exponential-2}
		Given $X \in \Exponential(\lambda)$, we have 
		\begin{align*}
			\mathbb{E}[e^{i t X}] & = \int_0^\infty e^{i t x} \lambda e^{-\lambda x} \,dx \\
			& = \int_0^\infty \lambda \, e^{-(\lambda - i t) x} \,dx \\
			& = \frac{\lambda}{\lambda - it} \text{ for all } t \in \mathbb{R}
		\end{align*}
		where the complex integral requires some complex analysis. 
	\end{example}

\subsection{Exercises} 

	\begin{exercise}[5 Intersecting Intervals]
		Five random intervals are generated within $[0,1]$. Each interval is formed by independently selecting two points uniformly at random from $[0,1]$ and taking the smaller as the left endpoint and the larger as the right endpoint. What is the probability that all five intervals share a common nonempty intersection?
	\end{exercise}
	\begin{solution}
		We let $X_i, Y_i \sim \Uniform(0, 1)$ for $i = 1, \ldots, 5$ denote the points sampled for each interval, and let $A_i = \min\{X_i, Y_i\}, B_i = \max\{X_i, Y_i\}$ denote the start and end points. Clearly, $A_i$ and $B_i$ are dependent, so we should calculate the joint pdf of it. We start with the joint CDF. 
		\begin{equation}
			F(a, b) = \mathbb{P}(A_i \leq a \wedge B_i \leq b) = \mathbb{P}(B_i \leq b \mid A_i \leq a) \mathbb{P}(A_i \leq a)
		\end{equation} 
		We know that $\mathbb{P}(A_i \leq a) = 2a - a^2$ when $0 \leq a \leq 1$. For $\mathbb{P}(B_i \leq b \mid A_i \leq a)$ we can solve it using different ways: 
		\begin{enumerate}
			\item Visualize the joint on a square. The red region represents when $A_i \leq a$, and the blue region reprsents the $\mathbb{P}(B_i \leq b \mid A_i \leq a$. 
				\begin{center}
					\begin{tikzpicture}[x=4.5cm, y=4.5cm, >=Stealth, scale=0.7]
						\definecolor{regionblue}{RGB}{30,120,255}
						\definecolor{regionred}{RGB}{220,50,50}
						\def\a{0.32}
						\def\b{0.72}

						% ---------- Shaded region {A_i <= a} ----------
						\fill[regionred, fill opacity=0.35] (0,0) -- (1,0) -- (1,\a) -- (\a,\a) -- (\a,1) -- (0,1) -- cycle;
						\draw[regionred, very thick] (0,0) -- (1,0) -- (1,\a) -- (\a,\a) -- (\a,1) -- (0,1) -- cycle;

						% ---------- Shaded region {A_i <= a, B_i <= b} ----------
						\fill[regionblue, fill opacity=0.35] (0,0) -- (\b,0) -- (\b,\a) -- (\a,\a) -- (\a,\b) -- (0,\b) -- cycle;
						\draw[regionblue, very thick] (0,0) -- (\b,0) -- (\b,\a) -- (\a,\a) -- (\a,\b) -- (0,\b) -- cycle;

						% ---------- Excised square (a,b]^2 ----------
						\draw[gray, dashed] (\a,\a) rectangle (\b,\b);

						% ---------- Unit square and axes ----------
						\draw[thick] (0,0) rectangle (1,1);
						\draw[->, thick] (0,0) -- (1.15,0) node[right] {$x$};
						\draw[->, thick] (0,0) -- (0,1.15) node[above] {$y$};

						% ---------- Ticks ----------
						\foreach \v/\lab in {\a/a, \b/b, 1/1} {
							\draw (\v,0) -- (\v,-0.02) node[below, font=\small] {$\lab$};
							\draw (0,\v) -- (-0.02,\v) node[left, font=\small] {$\lab$};
						}
					\end{tikzpicture}
				\end{center}
				For $0 \leq a \leq b \leq 1$ its area is $b^2 - (b - a)^2 = 2ab - a^2$, so that $F(a,b) = 2ab - a^2$ and
				\begin{equation}
					\mathbb{P}(B_i \leq b \mid A_i \leq a) = \frac{2ab - a^2}{2a - a^2} = \frac{2b - a}{2 - a}
				\end{equation}

			\item Complementary. We can see that $\mathbb{P}(B_i \leq b \mid A_i \leq a) = 1 - \mathbb{P}(B_i \geq b \mid A_i \leq a)$. This gives us the two events where one point must be in $[0, a]$ and the other point must be in $[b, 1]$. These points are mutually exclusive, so
				\begin{align}
					\mathbb{P}(B_i \geq b \mid A_i \leq a) & = \frac{\mathbb{P}(B_i \geq b,  A_i \leq a)}{\mathbb{P}(A_i \leq a)} \\ 
																								 & = \frac{\mathbb{P}(X_i \geq b, Y_i \leq a) + \mathbb{P}(Y_i \geq b, X_i \leq a)}{\mathbb{P}(A_i \leq a)} \\ 
																								 & = \frac{2a(1 - b)}{2a - a^2} = \frac{2 - 2b}{2 - a}
				\end{align}
				Therefore, $\mathbb{P}(B_i \leq b \mid A_i \leq a) = 1 - \frac{2 - 2b}{2 - a} = \frac{2b - a}{2 - a}$
		\end{enumerate}
		
		Therefore, $\mathbb{P}(A_i \leq a, B_i \leq b) = \frac{2b - a}{2 - a} (2a - a^2) = 2ab - a^2$, implying 
		\begin{equation}
			f(a, b) = 2 \text{ when } 0 \leq a \leq b \leq 1
		\end{equation}

		Now, let $\alpha = \max\{A_i\}, \beta = \min\{B_i\}$. The probability that the intersections of all 5 intervals being nontrivial is precisely $\mathbb{P}(\alpha \leq \beta)$ since this is the probability that the highest left endpoint is less than the smallest right endpoint. But note that $(A_i, B_i)$ are independent of $(A_j, B_j)$ for $i \neq j$. So, we can write 
		\begin{equation}
			\mathbb{P}(\alpha \leq a, \beta > b) = \prod_{i=1}^5 \mathbb{P}(A_i \leq a, B_i \geq b) = \prod_{i=1}^5 2 a (1 - b) =  32 a^5 (1 - b)^5 
		\end{equation}
		which implies that $F_{\alpha, \beta}(a, b) = \mathbb{P}(\alpha \leq a, \beta \leq b) = 1 - 32a^5 (1 - b)^5$. Now, we find the pdf
		\begin{equation}
			f_{\alpha, \beta} (a, b) = \frac{\partial^2}{\partial a \partial b} 1 - 32a^5 (1 - b)^5 = 800 a^4 (1 - b)^4
		\end{equation}
		which implies that 
		\begin{equation}
			\mathbb{P}(\alpha \leq \beta) = \int_0^1 \int_0^b f_{\alpha, \beta}(a, b) \,da \,db  = \frac{8}{63}
		\end{equation}
	\end{solution}
	\begin{solution}
		A more elegant solution is to think of all 10 points $X_1, \ldots, X_{10} \sim \Uniform(0, 1)$ already realized on the line. Then, for each point, we assign it to be a left endpoint or a right endpoint such that there are a total of 5 left and 5 right endpoints. We can do this iteratively. 
		\begin{enumerate}
		  \item You start be selecting the leftmost $X_i$, assign it to be a left endpoint, and you have $9$ choices for the right endpoint. 
			\item Select the leftmost $X_i$ out of the remaining $8$, and you have $7$ choices for the right endpoints. 
			\item Continue. 
		\end{enumerate}
		There are $9 \cdot 7 \cdot 5 \cdot 3 \cdot 1$ ways to do this. The pairings that we are interested in is where we have $5$ left endpoints followed by $5$ right endpoints, going from left to right. The number of ways to choose this can be done as follows. 
		\begin{enumerate}
		  \item You start be selecting the leftmost $X_i$, assign it to be a left endpoint, and you have $5$ choices for the right endpoint. 
			\item You select the leftmost $X_i$ out of the remaining $8$, assign it to be a left endpoint, and you have $4$ choices for the right endpoint. 
			\item Continue. 
		\end{enumerate}
		You keep doing this until you are done. Therefore, the probability is 
		\begin{equation}
			\frac{5 \cdot 4 \cdot 3 \cdot 2 \cdot 1}{9 \cdot 7 \cdot 5 \cdot 3 \cdot 1} = \frac{8}{63} 
		\end{equation}
	\end{solution}

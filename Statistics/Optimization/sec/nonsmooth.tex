\section{Nonsmooth Optimization}

\subsection{Convex Conjugacy}

	\begin{definition}[Proper Functions]
	  \label{def:proper-functions}
		Given a function $f: \mathbb{R}^d \to (-\infty, +\infty]$, it is called \textbf{proper} if it is finite somewhere. The \textbf{effective domain} is 
		\begin{equation}
			\dom(f) \coloneqq \{ x \in \mathbb{R}^d \colon f(x) < +\infty \}
		\end{equation}
	\end{definition}

	\begin{definition}[Convex, Fenchel Conjugate]
	  \label{def:convex-fenchel-conjugate}
		Given a proper function $f: \mathbb{R}^d \to (-\infty, +\infty]$, its \textbf{convex conjugate}, or \textbf{Fenchel conjugate}, is defined to be 
		\begin{equation}
			f^\ast: \mathbb{R}^d \to (-\infty, +\infty], \quad f^\ast (y) = \sup_{x \in \mathbb{R}^d} \{ \langle y, x \rangle - f(x) \}
		\end{equation}
		satisfying 
		\begin{enumerate}
			\item \textit{Convexity}. $f^\ast$ is convex: for all $y_1, y_2 \in \mathbb{R}^d$ and $t \in [0, 1]$, $f^\ast(t y_1 + (1 - t) y_2) \leq t f^\ast(y_1) + (1 - t) f^\ast(y_2)$. 
			\item \textit{Lower Semicontinuous}. $f^\ast$ is lower semicontinuous. 
		\end{enumerate}
	\end{definition}
	\begin{proof}
	  It is a pointwise supremum of continuous affine functions of $y$. 
	\end{proof}

	This generalizes the classical Legendre transform.

	  \label{thm:fenchel-young-inequality}

		\begin{equation}
		  f(x) + f^\ast (y) \geq \langle y, x \rangle
		\end{equation}
	\end{theorem}

	\begin{definition}[Subgradients, Subdifferentials]
	  \label{def:subgradients-subdifferentials}
		Given proper function $f$, a vector $y \in \mathbb{R}^d$ is a \textbf{subgradient} of $f$ at $x \in \dom(f)$ if 
		\begin{equation}
			f(z) \geq f(x) + \langle y, z - x \rangle \quad \forall z \in \mathbb{R}^d
		\end{equation}
		The set of all subgradients at $x$ is the \textbf{subdifferential} 
		\begin{equation}
			\partial f(x) \coloneqq \{ y \in \mathbb{R}^d \colon f(z) \geq f(x) + \langle y, z - x \rangle \; \forall z \}
		\end{equation}
	\end{definition}

  It is well known that the mean square error of a \hyperref[lin-def:linear_map]{linear map} is convex. However, when we impose---say the L1 penalty---the loss function is now not differentiable at $0$. The subgradient also acts as a linear approximation of $f$, but now at nondifferentiable points of \hyperref[mt-def:convex-function]{convex functions}, we have a set of linear approximations. It is clear that the subgradient of a convex function at a point where it is differentiable is uniquely the gradient ($\partial f(\mathbf{x}) = \{ \nabla f(\mathbf{x}) \}$), but for places like the absolute value, we can have infinite linear approximations.

  \end{equation}

	\begin{definition}[Biconjugate]
	  \label{def:biconjugate}
		Given proper $f$ and its convex conjugate $f^\ast$, the \textbf{biconjugate} of $f$ is the conjugate of $f^\ast$. 
		\begin{equation}
			f^{\ast\ast} (x) = \sup_{y \in \mathbb{R}^d} \{ \langle y, x \rangle - f^\ast (y) \}
		\end{equation}
	\end{definition}

	\begin{corollary}[Affine Lower Bound Reconstruction]
	  \label{thm:affine-lower-bound-reconstruction}
		Given proper $f$, we have 
		\begin{equation}
			f^{\ast\ast} \leq f
		\end{equation}
		That is, $f^{\ast\ast}$ reconstructs $f$ from its affine lower bounds. 
	\end{corollary}

	\begin{theorem}[Fenchel-Moreau]
	  \label{thm:fenchel-moreau}
	  If $f$ is proper, then the following are equivalent. 
		\begin{enumerate}
			\item $f = f^{\ast\ast}$. 
			\item $f$ is convex and lower-semicontinuous. 
		\end{enumerate}
	\end{theorem}
	\begin{proof}
	  
	\end{proof}

	Therefore, a proper lower semicontinuous \hyperref[mt-def:convex-function]{convex function} can be recovered entirely from its conjugate.

			\item $x \in \partial f^\ast (y)$. 
			\item $f(x) + f^\ast (y) = \langle y, x \rangle$.

\subsection{Subgradient Descent}

	Since $\partial f(x)$ is a \textit{set}, the first-order oracle of the smooth case no longer makes sense: there is nothing for $\mathcal{O}_f(x)$ to return at a point where $f$ is kinked. We must fix a choice.

	\begin{definition}[Weak Subgradient Oracle]
		\label{def:weak_subgradient_oracle}
		Let $\mathcal{F}$ be the class of proper \hyperref[mt-def:convex-function]{convex functions} $f : S \subset \mathbb{R}^d \to \mathbb{R}$. A \textbf{subgradient selection} for $f$ is any map
		\begin{equation}
			g_f : S \to \mathbb{R}^d, \quad g_f(x) \in \partial f(x) \quad \forall x \in S
		\end{equation}
		and the \textbf{weak subgradient oracle} associated with it is
		\begin{equation}
			\mathcal{O}_f : S \to \mathbb{R} \times \mathbb{R}^d, \quad \mathcal{O}_f (x) = \big( f(x), g_f(x) \big)
		\end{equation}
	\end{definition}

	Unlike the first-order oracle, $\mathcal{O}_f$ is \textit{not} determined by $f$ alone, since a proper convex $f$ admits many selections---in Figure \ref{fig:subgradient_abs}, any of the $g_i$ is a legal answer at $x = 0$. The convention is that an algorithm must work for \textit{every} admissible selection, i.e. the oracle is free to return the worst element of $\partial f(w_k)$ at each query. This is what makes the model a fair one for proving lower bounds. At any $x$ where $f$ is differentiable, $\partial f(x) = \{\nabla f(x)\}$, the selection is forced, and we recover the first-order oracle exactly.

	\begin{definition}[Subgradient Descent]
		\label{def:subgradient_descent}
		Let $\mathcal{F}$ be the class of proper \hyperref[mt-def:convex-function]{convex functions} $f: S \subset \mathbb{R}^d \to \mathbb{R}$ that are $G$-Lipschitz, i.e. $\|g\| \leq G$ for every $x \in S$ and every $g \in \partial f(x)$. Fix an initial point $w_0 \in S$, a sequence of positive step sizes $(\eta_k)$, and a \hyperref[def:weak_subgradient_oracle]{weak subgradient oracle}
		\begin{equation}
			\mathcal{O}_f : S \to \mathbb{R} \times \mathbb{R}^d, \quad \mathcal{O}_f (x) = \big( f(x), g_f(x) \big), \quad g_f(x) \in \partial f(x)
		\end{equation}
		Then \textbf{subgradient descent} is a deterministic algorithm defined by $\mathcal{O}_f$ and update maps $\Phi_k : (S \times (\mathbb{R} \times \mathbb{R}^d))^{k+1} \to S$
		\begin{align}
			w_{k+1} = \Phi_k \big( w_0, (f(w_0), g_0), \ldots, w_k, (f(w_k), g_k) \big)
				& \coloneqq w_k - \eta_k g_k \\
				& = \argmin_w \bigg\{ \underbrace{\langle g_k, w - w_k \rangle}_{\text{affine minorant}} + \underbrace{\frac{1}{2\eta_k} \|w - w_k\|^2}_{\text{anchor}} \bigg\}
		\end{align}
		where $g_k \coloneqq g_f(w_k) \in \partial f(w_k)$. After $K$ iterations, the output rule $\Psi_K$ returns either the \textbf{best iterate}
		\begin{equation}
			w_{\mathcal{A}_K} (f) = w_{k^\ast}, \quad k^\ast = \argmin_{0 \leq k \leq K} f(w_k)
		\end{equation}
		or the step-size weighted \textbf{ergodic average}
		\begin{equation}
			w_{\mathcal{A}_K} (f) = \bar{w}_K \coloneqq \frac{\sum_{k=0}^K \eta_k w_k}{\sum_{k=0}^K \eta_k}
		\end{equation}
	\end{definition}

	The update is the verbatim analogue of \hyperref[def:gradient_descent]{gradient descent}: minimize an affine model of $f$ plus a quadratic anchor. The only change is that the model is now a genuine \textit{global lower bound} on $f$, by the very definition of the subgradient, rather than a local Taylor approximation. Two other parts of the definition, however, really do differ from the smooth case, and both are forced on us rather than chosen.

	\textit{First, the output rule cannot be $w_{\mathcal{A}_K}(f) = w_K$.} Despite its name, \hyperref[def:subgradient_descent]{subgradient descent} is \textbf{not a descent method}. Where $f$ is nondifferentiable, $-g_k$ need not be a descent direction at all, so $f(w_{k+1}) > f(w_k)$ can happen for \textit{every} positive step size.

	\begin{example}[Subgradients of $|w_1| + 2|w_2|$]
	  \label{ex:subgradients-of-x12x2}
		Let $f(w_1, w_2) = |w_1| + 2|w_2|$ and take $w_k = (1, 0)$, so that $f(w_k) = 1$. Here $\partial f(w_k) = \{1\} \times [-2, 2]$, so $g_k = (1, 2)$ is a legal oracle response. Then $w_{k+1} = (1 - \eta, -2\eta)$ and
		\begin{equation}
			f(w_{k+1}) = |1 - \eta| + 4\eta = 1 + 3\eta > 1 = f(w_k)
		\end{equation}
		for every $0 < \eta \leq 1$. The objective increases no matter how small the step.
	\end{example}

	So $(f(w_k))_k$ is not monotone, and the final iterate carries no guarantee on its own. Taking the running minimum or averaging is what restores one, and both achieve the rate below.\footnote{The last iterate \textit{does} converge under carefully chosen schedules, but this is a considerably harder result than the averaged case (Shamir and Zhang, 2013).}

	\textit{Second, the step sizes cannot be a fixed constant chosen independently of the horizon.} With $\eta_k = \eta$ fixed, the iterates only reach a neighborhood of the optimum: the error bound floors out at $\eta G^2 / 2$, and no amount of further iteration removes it. The standard requirement is that $(\eta_k)$ be diminishing but nonsummable,
	\begin{equation}
		\eta_k \to 0, \quad \sum_{k=0}^\infty \eta_k = \infty
	\end{equation}
	which drives the floor to $0$ while still letting the iterates travel an unbounded distance from $w_0$.

	\begin{theorem}[Convergence of Subgradient Descent]
	  \label{thm:convergence-of-subgradient-descent}
		Let $f$ be convex and $G$-Lipschitz with minimizer $w^\ast$, and set $R = \|w_0 - w^\ast\|$. Running \hyperref[def:subgradient_descent]{subgradient descent} for a fixed horizon $K$ with the step size $\eta = \frac{R}{G \sqrt{K}}$ gives
		\begin{equation}
			\min_{0 \leq k \leq K} f(w_k) - f(w^\ast) \leq f(\bar{w}_K) - f(w^\ast) \leq \frac{GR}{\sqrt{K}}
		\end{equation}
	\end{theorem}
	\begin{proof}

	\end{proof}

	This $O(1/\sqrt{K})$ rate is far worse than the $O(1/K)$ that \hyperref[def:gradient_descent]{gradient descent} achieves on $L$-smooth \hyperref[mt-def:convex-function]{convex functions}, and it cannot be improved: it is the minimax rate over the class of $G$-Lipschitz convex functions in the black-box first-order model, so no cleverer choice of $\Phi_k$ and $\Psi_K$ does better. Note also that \hyperref[def:gradient_descent_momentum]{momentum} buys us nothing here. The gap is the price of nonsmoothness alone, which motivates the proximal methods of the next section: rather than probing the nonsmooth part of the objective with a subgradient, we handle it exactly.

\subsection{Proximal Methods}

  \begin{definition}[Proximal Operator]
    \label{def:proximal-operator}
    Given a lower semicontinuous \hyperref[mt-def:convex-function]{convex function} $f$ mapping from \hyperref[fa-def:hilbert-space]{Hilbert space} $X$ to $(-\infty, +\infty]$ (and proper), its \textbf{proximal operator} associated with a point $u$ is defined 
    \begin{equation}
      \prox_{f, \tau} (u) = \argmin_{x} \bigg( f(x) + \frac{1}{2\tau} \|x - u\|^2 \bigg)
    \end{equation}
    where $\tau > 0$ is a parameter that scales the quadratic term. This is basically the point that minimizes the sum of $f(x)$ and the square of the Euclidean distance between $x$ and $u$, scaled by $1/2\tau$. 
  \end{definition}

  Now given the loss function $L (\boldsymbol{\theta}) = L_{\mathrm{obj}} (\boldsymbol{\theta}) + L_{\mathrm{reg}} (\boldsymbol{\theta})$, we want to compute the \hyperref[def:proximal-operator]{proximal operator} on the regularization loss and update that with the gradient of the smooth objective loss. 
  \begin{equation}
    \boldsymbol{\theta}^{(k+1)} = \prox_{L_{\mathrm{reg}}, \tau} \big[ \boldsymbol{\theta}^{(k)} - \tau \nabla L_{\mathrm{obj}} (\boldsymbol{\theta}^{(k)}) \big]
  \end{equation}
  Let's compute the proximal operator of the L1 loss $h(\boldsymbol{\theta}) = \lambda \|\boldsymbol{\theta}\|_1$. We can parameterize this loss by the $\lambda$, so we will use the notation $\prox_{\lambda, \tau}$ rather than $\prox_{h, \tau}$. 
  \begin{align*}
    \prox_{\lambda, \tau} (\mathbf{u}) & = \argmin_{\boldsymbol{\theta}} \bigg( \lambda \|\boldsymbol{\theta}\|_1 + \frac{1}{2\tau} \|\boldsymbol{\theta} - \mathbf{u}\|_2^2 \bigg) \\
    & = \argmin_{\boldsymbol{\theta}} \bigg( \sum_{i=1}^n \lambda |\theta_i| + \frac{1}{2\tau} (\theta_i - u_i)^2 \bigg) 
  \end{align*}
  These are separable functions that can be decoupled and optimized component-wise. So, we really just want to find 
  \begin{equation}
    \theta_i^\ast = \argmin_{\theta_i} \bigg( \lambda |\theta_i| + \frac{1}{2\tau} (\theta_i - u_i)^2 \bigg)
  \end{equation}
  The sum of \hyperref[mt-def:convex-function]{convex functions} is convex, and so we should differentiate it and find where the gradient is $0$ to optimize it. 
  \begin{enumerate}
    \item When $\theta_i > 0$, then we minimize $\lambda \theta_i + \frac{1}{2\tau} (\theta_i - u_i)^2$, so taking the gradient and setting to $0$ gives 
    \begin{equation}
      \theta_i = u_i - \lambda \tau
    \end{equation}
    subject to the constraint that $\theta_i > 0$, or equivalently, that $u_i > \lambda \tau$. 

    \item When $\theta_i < 0$, then we minimize $-\lambda \theta_i + \frac{1}{2\tau} ( \theta_i - u_i)^2$, so taking the gradient and setting to $0$ gives 
    \begin{equation}
      \theta_i = u_i + \lambda \tau
    \end{equation}
    subject to the constraint that $\theta_i < 0$, or equivalently, that $u_i < -\lambda \tau$. 

    \item When $\theta_i = 0$, then we minimize $\lambda |\theta_i| + \frac{1}{2\tau} (\theta_i - u_i)^2$, which doesn't have derivative at $\theta_i = 0$. So, we can compute the subdifferential of it to get 
    \begin{equation}
      0 \in \partial \bigg( \lambda |\theta_i| + \frac{1}{2\tau} (\theta_i - u_i)^2 \bigg) = \lambda \partial (|\theta_i|) + \frac{1}{\tau} (\theta_i - u_i)
    \end{equation}
    Now at $\theta_i = 0$, the subdifferential can be any value in $[-1, 1]$, and the above reduces to 
    \begin{equation}
      0 \in \lambda [-1, 1] - \frac{1}{\tau} u_i
    \end{equation}
    this is equivalent to saying that $u_i/\tau$ is contained in the interval $[-\lambda, \lambda]$, meaning that $u_i \in [-\lambda \tau, \lambda \tau]$. 
  \end{enumerate}

  Ultimately we get that 
  \begin{equation}
    \prox_{\lambda, \tau} (u) = \begin{cases} u - \lambda \tau & \text{ if } u > \lambda \tau \\ 0 & \text{ if } |u| \leq \lambda \tau \\ u + \lambda \tau & \text{ if } u < - \lambda \tau \end{cases}
  \end{equation}
  which can be simplified to 
  \begin{equation}
    \prox_{\lambda, \tau} (u) = \mathrm{sign}(u) \max\{ |u| - \lambda \tau, 0\}
  \end{equation}

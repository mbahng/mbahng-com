\section{Smooth Optimization}

	Unfortunately, $f$ being merely differentiable does not give too many regularity conditions. A moderately stronger regularity condition is that the gradient itself is Lipschitz continuous. 

	\begin{definition}[$L$-Smooth]
	  \label{def:l-smooth}
	  A function $f: \mathbb{R}^d \to \mathbb{R}$ is \textbf{$L$-smooth} if the gradient is $L$-Lipschtiz (and hence $C^1$). 
		\begin{equation}
		  \| \nabla f(x) - \nabla f(y) \| \leq L \|x - y\|
		\end{equation}
	\end{definition}

	\begin{lemma}[Properties of $L$-Smoothness]
	  \label{thm:properties-of-l-smoothness}
		Let $f, g: \mathbb{R}^d \to \mathbb{R}$ with $f$ $L$-smooth and $g$ $M$-smooth. 
		\begin{enumerate}
		  \item \textit{Monotonicity}. $f$ is $L'$-smooth for every $L' \geq L$. 
			\item \textit{Scaling}. For $a \in \mathbb{R}$, $af$ is $|a|L$-smooth. 
			\item \textit{Addition}. $f + g$ is $(L + M)$-smooth.
			\item \textit{Affine Precomposition}. For $A \in \mathbb{R}^{m \times d}$ and $b \in \mathbb{R}^m$, if $f: \mathbb{R}^m \to \mathbb{R}$ is $L$-smooth, then $h(x) = f(Ax + b)$ is $L\|A\|_{op}^2$-smooth. 
			\item \textit{\hyperref[mv-def:hessian]{Hessian} Characterization}. If $f$ is $C^2$, then $f$ is $L$-smooth if and only if $\|\nabla^2 f(x)\|_{op} \leq L$ for all $x$. 
			\item \textit{Product}. If $f, g$ are bounded and have bounded gradients, say $|f| \leq B_f$, $|g| \leq B_g$, $\|\nabla f\| \leq G_f$, $\|\nabla g\| \leq G_g$, then $fg$ is $(B_f M + B_g L + 2 G_f G_g)$-smooth.\footnote{The boundedness assumptions in the \hyperref[mv-thm:product-rule-2]{product rule} cannot be dropped. For $f = g = \frac{1}{2}x^2$ on $\mathbb{R}$, both are $1$-smooth, but $fg = \frac{1}{4}x^4$ has $(fg)'' = 3x^2$, which is unbounded, so it is not $L$-smooth for any $L$ by (5).}
		\end{enumerate}
	\end{lemma}
	\begin{proof}
		Listed. 
		\begin{enumerate}
		  \item \textit{Monotonicity}. Immediate from definition. 
			\item \textit{Scaling}. Immediate from definition. 
			\item \textit{Addition}. by the triangle inequality, 
				\begin{equation}
					\|\nabla (f + g)(x) - \nabla (f+g)(y)\| \leq \|\nabla f(x) - \nabla f(y)\| + \|\nabla g(x) - \nabla g(y)\| \leq (L + M)\|x - y\|
				\end{equation}

			\item \textit{Affine Precomposition}. By the \hyperref[mv-thm:chain-rule]{chain rule}, $\nabla h(x) = A^T \nabla f(Ax + b)$, so 
				\begin{equation}
					\|\nabla h(x) - \nabla h(y)\| \leq \|A^T\|_{op} \|\nabla f(Ax + b) - \nabla f(Ay + b)\| \leq \|A\|_{op} L \|A(x - y)\| \leq L\|A\|_{op}^2 \|x - y\|
				\end{equation}

			\item \textit{\hyperref[mv-def:hessian]{Hessian}}

		\end{enumerate}


		(5) ($\Leftarrow$) By the fundamental theorem of calculus, $\nabla f(x) - \nabla f(y) = \int_0^1 \nabla^2 f(y + t(x - y)) (x - y) \, dt$, so $\|\nabla f(x) - \nabla f(y)\| \leq \sup_z \|\nabla^2 f(z)\|_{op} \|x - y\|$. ($\Rightarrow$) For any unit vector $u$ and $t > 0$, $\|\nabla^2 f(x) u\| = \lim_{t \to 0} \frac{1}{t}\|\nabla f(x + tu) - \nabla f(x)\| \leq L$. 

		(6) Since $\nabla (fg) = f \nabla g + g \nabla f$, we can write 
		\begin{align}
		  \nabla(fg)(x) - \nabla(fg)(y) & = f(x)\big(\nabla g(x) - \nabla g(y)\big) + \big(f(x) - f(y)\big) \nabla g(y) \\ 
		  & \quad + g(x)\big(\nabla f(x) - \nabla f(y)\big) + \big(g(x) - g(y)\big) \nabla f(y)
		\end{align}
		Since $\|\nabla f\| \leq G_f$ and $\|\nabla g\| \leq G_g$, $f$ and $g$ are $G_f$- and $G_g$-Lipschitz respectively (\hyperref[mv-thm:mean-value-theorem]{mean value theorem}), so taking norms gives 
		\begin{equation}
		  \|\nabla(fg)(x) - \nabla(fg)(y)\| \leq \big(B_f M + G_f G_g + B_g L + G_g G_f\big) \|x - y\|
		\end{equation}
	\end{proof}

	When we have $L$-smoothness, we can get some nice bounds. 

	\begin{theorem}[Taylor's 1st Order Bound]
	  \label{thm:taylors-1st-order-bound}
		If $f: \mathbb{R}^d \to \mathbb{R}$ is $L$-smooth, then 
		\begin{equation}
			|f(x) - f(w) - \langle \nabla f(w), x - w \rangle | \leq \frac{L}{2} \|x - w\|^2
		\end{equation}
		and therefore 
		\begin{equation}
			f(x) \leq f(w) + \langle \nabla f(w), x - w \rangle + \frac{L}{2} \|x - w\|^2
		\end{equation}
	\end{theorem}
	\begin{proof}
		Let $x = w + s$ and define $\theta: [0, 1] \to \mathbb{R}, \theta(t) = f(w + ts)$. Since $f \in C^1$, $f^\prime$ is continuous and hence Riemann integrable. By the \hyperref[real-thm:ftc2]{2nd fundamental theorem of calculus}, we can write 
		\begin{align}
			|f(x + s) - f(x) - \nabla f(x)^T s| & = \bigg| \theta(0) + \int_0^1 \theta^\prime (t) \,dt - \theta(0) - \nabla f(x)^T s \bigg|  \\ 
																					& = \bigg| \int_0^1 \langle \nabla f(x + ts) - \nabla f(x), s \rangle \,dt \bigg| \\ 
																					& \leq \int_0^1 | \langle \nabla f(x + ts) - \nabla f(x), s \rangle | \,dt \\ 
																					& \leq \int_0^1 \| \nabla f(x + ts) - \nabla f(x) \| \|s\| \,dt \\ 
																					& \leq \int_0^1 L \|ts\| \|s\|\,dt \\ 
																					& = \frac{1}{2} L \|s\|^2
		\end{align}
	\end{proof}

	This is useful since it upper-bounds a potentially complicated nonlinear function by a simple quadratic function. An algorithmic design principle that we will follow is that instead of minimizing $f$ directly, we iteratively minimize an approximation of $f$ centered at the previous iterate. 

	\begin{lemma}[Gradient Descent on $L$-Smooth Functions]
	  \label{thm:gradient-descent-on-l-smooth-functions}
		\hyperref[def:gradient_descent]{Gradient descent} on the \hyperref[def:function-class]{function class} $\mathcal{F}$ of $L$-smooth functions is equivalent to the following algorithm consisting of the oracle 
		\begin{equation}
			\mathcal{O}_f : S \to \mathbb{R} \times \mathbb{R}^d, \quad \mathcal{O}_f (x) = \big( f(x), \nabla f(x) \big)
		\end{equation}
		and update maps $\Phi_t : (S \times (\mathbb{R} \times \mathbb{R}^d))^{t+1} \to S$
		\begin{align}
			x_{t+1} & = \Phi_t \big(x_0, (f(x_0), \nabla f(x_0)), \ldots, x_t, (f(x_t), \nabla f(x_t)) \big) \\
							& \coloneqq x_t - \eta_t \nabla f(x_t) \\
							& = \argmin_w \bigg\{ \underbrace{\langle \nabla f(w_t), w - w_t \rangle}_{\text{linear approximation}} + \underbrace{\frac{1}{\eta_t} \|w - w_t\|^2}_{\text{anchor}} \bigg\} 
		\end{align}
		Note that the quadratic term is what makes this function convex, and therefore has a well-defined minimum. It is also a regularizer that prevents us from moving too far from the current point $w_k$. After $k$ iterations, the algorithm outputs $w_{\mathcal{A}_k} (f) = x_k$.

	\end{lemma}
	\begin{proof}
		Recall that at a local minimum of a differentiable function, the gradient must vanish. So, 
		\begin{equation}
			0 = \nabla \Phi_k (w_{k+1}) = \nabla f(w_k) + \frac{1}{\eta_k}(w_{k+1} - w_k) \implies w_{k+1} = w_k - \eta_k \nabla f(w_k)
		\end{equation}
		So $\eta$ is sort of like the step size. 
	\end{proof}

	Okay, so we have an equivalent formulation of GD. This formulation will allow us to establish our desired bounds. 

	\begin{lemma}[Descent Lemma]
	  \label{thm:descent-lemma}
		Let $f: \mathbb{R}^d \to \mathbb{R}$ be $L$-smooth. Then, 
		\begin{equation}
			f(w - \eta \nabla f(w)) \leq f(w) - \eta \big(1 - \frac{L \eta}{2} \big) \|\nabla f(w)\|^2
		\end{equation}
		Furthermore, if $\eta \in (0, \frac{2}{L})$, then the iterates are guaranteed to decrease, i.e. 
		\begin{equation}
			f(w - \eta \nabla f(w)) \leq f(w) 
		\end{equation}
	\end{lemma}
	\begin{proof}
	  This follows from Taylor's with $x = w - \eta \nabla f(w)$, since 
		\begin{align}
			f(w - \eta \nabla f(w)) & \leq f(w) + \langle \nabla f(w), - \eta \nabla f(w) \rangle + \frac{L}{2} \| \eta \nabla f(w)\|^2 \\ 
															& = f(w) - \eta \|\nabla f(w)\|^2 + \frac{L \eta^2}{2} \| \nabla f(w)\|^2 \\ 
															& = f(w) - \eta \big( 1 - \frac{L\eta}{2} \big) \|\nabla f(w)\|^2
		\end{align}
		Now if $0 < \eta < 2/L$, then $\eta < \frac{2}{L} \implies 0 < \frac{L\eta}{2}$, and with the fact that $0 < \eta$, 
		\begin{equation}
			\eta \big( 1 - \frac{L\eta}{2} \big) > 0 \implies f(w) - \eta \big( 1 - \frac{L\eta}{2} \big) \|\nabla f(w)\|^2 < f(w)
		\end{equation} 
	\end{proof}
	
	Note that when $\eta = \frac{1}{L}$, then $\eta \big( 1 - \frac{L \eta}{2} \big) = \frac{1}{2L}$. For this reason, we will often set the stepsize to be $\eta = \frac{1}{L}$. (CHECK)

	\begin{theorem}[Convergence to Criticality of GD]
		\label{thm:convergence_to_criticality_gd}
		Suppose that $f$ is $L$-smooth and $\inf{f} > -\infty$. Consider the iterates $w_{k+1} \gets w_k - \frac{1}{L} \nabla f(w_k)$. Then, 
		\begin{equation}
			\min_{k \in [T - 1]} \|\nabla f(w_k) \| \leq \sqrt{ \frac{2L \big( f(w_0) - \inf{f} \big)}{T}}
		\end{equation}
		Therefore, for any $\epsilon > 0$, we have
		\begin{equation}
			T \geq \frac{2L}{\epsilon^2} \big( f(w_0) - f(w^\ast) \big) \implies \min_{t \in [T-1]} \|\nabla f (w_t) \| \leq \epsilon
		\end{equation}
	\end{theorem}
	\begin{proof}
		We apply the \hyperref[thm:descent-lemma]{descent lemma} to each iterate and get 
		\begin{equation}
			f(w_{k+1}) \leq f(w_k) - \frac{1}{2L} \|\nabla f(w)\|^2 \implies \|\nabla f(w_k)\|^2 \leq 2L \big( f(w_k) - f(w_{k+1}) \big)
		\end{equation}
		Now the minimum has to be less than the average, so 
		\begin{align}
			\min_{k \in [T - 1]} \|\nabla f(w_k)\|^2 & \leq \frac{1}{T} \sum_{k=0}^{T-1} \|\nabla f(w_k) \|^2 \\ 
																							 & \leq \frac{2L}{T} \sum_{k=0}^{T-1} \big( f(w_k) - f(w_{k+1}) \big) \\ 
																							 & = \frac{2L}{T} \big( f(w_0) - f(w_T) \big) \\ 
																							 & \leq \frac{2L}{T} \big( f(w_0) - \inf(f) \big)  
		\end{align}
		Therefore, 
		\begin{equation}
			\min_{k \in [T-1]} \|\nabla f(w_k) \| = \sqrt{\min_{k \in [T-1]} \|\nabla f(w_k) \|^2} \leq \sqrt{\frac{2L \big( f(w_0) - \inf(f) \big)}{T}}
		\end{equation}
	\end{proof}

	This means that up to constants, $T \sim \epsilon^{-2}$ for a gradient bound. So if $\epsilon = 10^{-6}$, then we would need $\epsilon^{-2} = 10^{12}$ iterations! This is not good enough, so we would like some extra regularity assumptions, such as convexity.\footnote{In fact, most of these regularity/geometric assumptions that people used to prove guarantees amounted to strong convexity.}

\subsection{Convexity} 

	Recall what a \hyperref[real-def:convex_functions]{convex function} is. We can immediately use this to bound the suboptimality with the distance to the minimizer. 

	\begin{theorem}[Convergence of Gradient Descent for Smooth Convex Functions]
	  \label{thm:convergence-of-gradient-descent-for-smooth}
		Suppose $f: \mathbb{R}^d \to \mathbb{R}$ is convex and $L$-smooth, and admits a minimizer $w^\ast$. Then, \hyperref[def:gradient_descent]{gradient descent} with step size $\eta = \frac{1}{L}$ satisfies, for every $k \geq 1$,
		\begin{equation}
			f(w_k) - f^\ast \leq \frac{L \|w_0 - w^\ast\|^2}{2k}.
		\end{equation}
		Consequently, for every $\epsilon > 0$, 
		\begin{equation}
			k \geq \frac{L \|w_0 - w^\ast\|^2}{2\epsilon} \implies f(w_k) - f^\ast \leq \epsilon
		\end{equation}
		giving an iteration complexity of $O(\epsilon^{-1})$.
	\end{theorem}
	\begin{proof}
		By convexity,
		\begin{equation}
			f(w_k) - f^\ast \leq \langle \nabla f(w_k), w_k - w^\ast \rangle.
		\end{equation}
		On the other hand, by the \hyperref[thm:descent-lemma]{descent lemma} and the update rule,
		\begin{align}
			f(w_{k+1}) 
				&\leq f(w_k) + \langle \nabla f(w_k), w_{k+1} - w_k \rangle + \frac{L}{2} \|w_{k+1} - w_k\|^2 \\
				&= f(w_k) - \frac{1}{2L} \|\nabla f(w_k)\|^2.
		\end{align}
		Combining these inequalities gives
		\begin{equation}
			f(w_{k+1}) - f^\ast \leq \langle \nabla f(w_k), w_k - w^\ast \rangle - \frac{1}{2L} \|\nabla f(w_k)\|^2.
		\end{equation}
		Now, expanding the squared distance to the minimizer,
		\begin{align}
			\|w_{k+1} - w^\ast\|^2
			&= \left\|w_k - w^\ast - \frac{1}{L}\nabla f(w_k)\right\|^2 \\
			&= \|w_k - w^\ast\|^2 - \frac{2}{L} \langle \nabla f(w_k), w_k - w^\ast \rangle + \frac{1}{L^2}\|\nabla f(w_k)\|^2.
		\end{align}
		Rearranging yields the key inequality
		\begin{equation}
			f(w_{k+1}) - f^\ast \leq \frac{L}{2} \left( \|w_k - w^\ast\|^2 - \|w_{k+1} - w^\ast\|^2 \right).
		\end{equation}
		Summing over $t = 0, \ldots, k-1$, the right-hand side telescopes:
		\begin{align}
			\sum_{t=0}^{k-1} \big(f(w_{t+1}) - f^\ast\big)
				&\leq \frac{L}{2} \sum_{t=0}^{k-1} \left( \|w_t - w^\ast\|^2 - \|w_{t+1} - w^\ast\|^2 \right) \\
				&= \frac{L}{2} \left( \|w_0 - w^\ast\|^2 - \|w_k - w^\ast\|^2 \right) \\
				&\leq \frac{L}{2}\|w_0 - w^\ast\|^2.
		\end{align}
		Since the descent lemma ensures that
		$f(w_{t+1}) \leq f(w_t)$, we have
		\begin{equation}
			k\big(f(w_k) - f^\ast\big) \leq \sum_{t=0}^{k-1}\big(f(w_{t+1}) - f^\ast\big).
		\end{equation}
		Therefore,
		\begin{equation}
			f(w_k) - f^\ast
			\leq \frac{L\|w_0 - w^\ast\|^2}{2k}.
		\end{equation}
	\end{proof}

	\begin{lemma}[Co-Coercivity of Gradients]
		\label{thm:cocoercive_gradient}
		If $f: \mathbb{R}^d \to \mathbb{R}$ is convex and $L$-smooth, then $\nabla f$ is $\frac{1}{L}$-co-coercive, i.e. for all $x, w \in \mathbb{R}^d$, 
		\begin{equation}
			\langle \nabla f(x) - \nabla f(w), x - w \rangle \geq \frac{1}{L} \|\nabla f(x) - \nabla f(w)\|^2
		\end{equation}
	\end{lemma}
	\begin{proof}
		Fix $w$ and define $\varphi(z) = f(z) - \langle \nabla f(w), z \rangle$. Then $\varphi$ is convex and $L$-smooth, since $\nabla \varphi(z) = \nabla f(z) - \nabla f(w)$. Since $\nabla \varphi(w) = 0$ and $\varphi$ is convex, $w$ is a global minimizer of $\varphi$. Applying the \hyperref[thm:descent-lemma]{descent lemma} with step size $\eta = 1/L$, 
		\begin{equation}
			\varphi(w) \leq \varphi\big( x - \tfrac{1}{L} \nabla \varphi(x) \big) \leq \varphi(x) - \frac{1}{2L} \|\nabla \varphi(x)\|^2
		\end{equation}
		Substituting back the definition of $\varphi$ and rearranging, 
		\begin{equation}
			f(x) \geq f(w) + \langle \nabla f(w), x - w \rangle + \frac{1}{2L} \|\nabla f(x) - \nabla f(w)\|^2
		\end{equation}
		Swapping the roles of $x$ and $w$ and adding the two inequalities gives the claim. 
	\end{proof}

	\begin{definition}[$\mu$-Strongly Convex]
	  \label{def:mu-strongly-convex}
		A function $f: \mathbb{R}^d \to \mathbb{R}$ is \textbf{$\mu$-strongly convex} if it satisfies the equivalent conditions below.\footnote{$f$ is just convex if $\mu = 0$.} 
		\begin{enumerate}
		  \item \textit{Strong Convexity}. For all $w, x \in \mathbb{R}^d$, 
				\begin{equation}
					f(x) \geq f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - w\|^2
				\end{equation}
				Note that $f$ is convex when $\mu = 0$. 

			\item \textit{Strongly Monotone Gradient}. The gradient $\nabla f$ is \hyperref[mv-def:strongly_monotone]{strongly monotone}, i.e. for all $x, w \in \mathbb{R}^d$, 
				\begin{equation}
				  \langle \nabla f(x) - \nabla f(w), x - w \rangle \geq \mu \|x - w\|^2
				\end{equation}
		\end{enumerate}
	\end{definition} 
	\begin{proof}
	  We prove bidirectionally. 
		\begin{enumerate}
		  \item $(\rightarrow)$. From strong convexity, we can see that 
				\begin{align}
					\langle \nabla f(w), x - w \rangle & \leq f(x) - f(w) - \frac{\mu}{2} \|x - w\|^2 \\ 
					\langle \nabla f(x), w - x \rangle & \leq f(w) - f(x) - \frac{\mu}{2} \|x - w\|^2
				\end{align}
				Therefore, 
				\begin{align}
					\langle \nabla f(x) - \nabla f(w), x - w \rangle 
						& = \langle \nabla f(x), x - w \rangle - \langle \nabla f(w), x - w \rangle \\ 
						& \geq \big\{ -f(w) + f(x) + \frac{\mu}{2} \|x - w\|^2 \big\} + \big\{ - f(x) + f(w) + \frac{\mu}{2} \| x - w\|^2 \big\} \\ 
						& = \mu \|x - w\|^2
				\end{align}

			\item $(\leftarrow)$. We define the straight line path 
				\begin{equation}
					\varphi: [0, 1] \to \mathbb{R}, \quad \varphi(t) = f\big( w + t(x - w) \big)
				\end{equation}
				Then, by FTC
				\begin{align}
					f(x) = \varphi(1) & = \varphi(0) + \int_0^1 \varphi^\prime (t) \,dt \\ 
														& = \varphi(0) + \int_0^1 \langle \nabla f(w + t(x - w)), x - w \rangle \,dt  \\ 
														& \geq  f(w) + \int_0^1 \mu t \|x - w\|^2 + \langle \nabla f(w), x - w \rangle \,dt \\
														& = f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - w\|^2
				\end{align}
				where for the inequality, we invoke strong monotonicity and have 
				\begin{equation}
				  \langle \nabla f(w + t(x - w)) - \nabla f(w), t (x - w) \rangle \geq \mu t^2 \|x - w\|^2
				\end{equation}
				After dividing by $t$ and rearranging, we have 
				\begin{equation}
				  \langle \nabla f(w + t(x - w)), x - w \rangle \geq \mu t \|x - w\|^2 + \langle \nabla f(w), x - w \rangle
				\end{equation}
		\end{enumerate}
	\end{proof}

	This is a first order description of $\mu$-strongly \hyperref[mt-def:convex-function]{convex functions}. Here is a second order one. 

	\begin{lemma}[2nd Order Definition of Strongly Convex Functions]
	  \label{thm:2nd-order-definition-of-strongly-convex}
	  A $C^2$ function $f: \mathbb{R}^d \to \mathbb{R}$ is $\mu$-strongly convex iff $Hf(x) \succeq \mu I$.\footnote{For \hyperref[lin-def:symmetric-matrices]{symmetric matrices} $A, B$, $A \succeq$ if $A - B$ is positive semidefinite.}
	\end{lemma}
	\begin{proof}
		We prove bidirectionally. Given $x, w \in \mathbb{R}^d$, let $g(t) = f(w + t(x - w))$, which is $C^2$. 
		\begin{enumerate}
		  \item $(\rightarrow)$. Take the second order Taylor expansion and invoke strong convexity to get the inequality 
				\begin{align}
					g(t) & = f(w) + t \langle \nabla f(w), x - w \rangle + \frac{1}{2} t^2 \langle \nabla^2 f(w) (x - w), x - w \rangle + o(\|t(x - w)\|^2) \\ 
							 & \geq f(w) + \langle \nabla f(w), t (x - w) \rangle + \frac{\mu t^2}{2} \|x - w\|^2
				\end{align}
				Cleaning this up gives 
				\begin{equation}
					\langle \nabla^2 f(w) (x - w), x - w \rangle + \frac{o(t^2 \|x - w\|^2)}{t^2} \geq \mu \|x - w\|^2
				\end{equation}
				This holds for all $t \in [0, 1]$, so by letting $t \to 0$, we have 
				\begin{equation}
				  \langle \nabla^2 f(w) (x - w), x - w \rangle \geq \mu \|x - w\|^2 \implies Hf(w) \succeq \mu I
				\end{equation}

		  \item $(\leftarrow)$. We can see that 
				\begin{align}
					g^\prime (t) & = \langle f(w + t(x - w)), x - w \rangle \\ 
					g^{\prime\prime} (t) & = (x - w)^T H f(w + t(x - w)) (x - w) \\ 
															 & \geq \mu (x - w)^T (x - w) = \mu \|x - w\|^2
				\end{align}
				So, using \hyperref[real-thm:integral_form_remainder]{Taylor's expansion with the integral form of the remainder}, we have 
				\begin{align}
					f(x) = g(1) & = g(0) + g^\prime (0) (1 - 0) + \frac{1}{2} \int_0^1 g^{\prime\prime} (t) (1 - t) \,dt \\ 
											& \geq g(0) + g^\prime (0) + \frac{1}{2} \mu \|x - w\|^2 \int_0^1 (1 - t) \,dt \\ 
											& = f(w) + \langle \nabla f(w), x - w \rangle + \frac{\mu}{2} \|x - \mu\|^2
				\end{align}
		\end{enumerate}
	\end{proof}

	One of the best properties of \hyperref[mt-def:convex-function]{convex functions} is that there always exists a unique minimizer. 

	\begin{theorem}[Uniqueness and Distance Bound to Minimizer of Strongly Convex Functions]
	  \label{thm:uniqueness-and-distance-bound-to-minimizer-of}
		Suppose that $f$ is $\mu$-strongly convex. Then, $f$ has a unique minimizer $w^\ast$, and for all $w \in \mathbb{R}^d$, 
		\begin{align}
			\frac{\mu}{2} \|w - w^\ast\|^2 & \leq f(w) - f(w^\ast) & \text{(quadratic growth)} \\ 
			\mu \|w - w^\ast\| & \leq \|\nabla f(w)\| & \text{(error bound)}
		\end{align}
	\end{theorem}
	\begin{proof}
	  To prove existence, we first claim that $f$ is coercive. Indeed, 
		\begin{equation}
			f(x) \geq f(0) + \langle \nabla f(0), x \rangle + \frac{\mu}{2} \|x\|^2
		\end{equation}
		and the quadratic term dominates as $\|x\| \to +\infty$. Therefore, there exists a large enough $c > 0$ s.t. the set 
		\begin{equation}
			A = \{ x \in \mathbb{R}^d \colon f(x) \leq c \}
		\end{equation}
		is nonempty. It is also closed since $f$ is continuous. By coercivity, $A$ is bounded, so by \hyperref[real-thm:heine-borel]{Heine-Borel}, it is compact, and by the \hyperref[real-thm:extreme-value-theorem]{extreme value theorem}, there exists a minimizer $w^\ast$. To prove uniqueness, assume that $w_1, w_2$ are minimizers with $w_1 \neq w_2$. Then, by strong convexity, 
		\begin{equation}
			f(w_1) \geq f(w_2) + \langle \nabla f(w_2), w_1 - w_2 \rangle + \frac{\mu}{2} \|w_1 - w_2\|^2
		\end{equation}
		Since $f$ is differentiable and $w_2$ is a minimizer, $\nabla f(w_2) = 0$, and so $f(w_1) \geq f(w_2) + \frac{\mu}{2} \|w_1 - w_2\|^2 > f(w_2)$, contradicting the fact that $w_1$ is a minimizer. Finally, the claimed inequality follows from strong convexity since 
		\begin{equation}
			f(w) - f(w^\ast) \geq \langle \nabla f(w^\ast), w - w^\ast \rangle + \frac{\mu}{2} \|w - w^\ast\|^2 = \frac{\mu}{2} \|w - w^\ast\|^2
		\end{equation}
		For the error bound, we apply strongly monotone gradient to $w$ and $w^\ast$ and use $\nabla f(w^\ast) = 0$ to get 
		\begin{equation}
			\|\nabla f(w)\| \|w - w^\ast\| \geq \langle \nabla f(w) - \nabla f(w^\ast), w - w^\ast \rangle \geq \mu \|w - w^\ast\|^2
		\end{equation}
		where the first inequality is Cauchy--Schwarz. Dividing by $\|w - w^\ast\|$ gives the claim (trivial if $w = w^\ast$). 
	\end{proof}
	
	\begin{lemma}[$L$-Smooth and $\mu$-Strongly Convex]
	  \label{thm:l-smooth-and-mu-strongly-convex}
	  If $f: \mathbb{R}^d \to \mathbb{R}$ is both $L$-smooth and $\mu$-strongly convex, then 
		\begin{equation}
		  0 \leq \mu \leq L
		\end{equation}
	\end{lemma}
	\begin{proof}
	  $f$ is $L$-smooth, so $\|\nabla f(x) - \nabla f(w) \| \leq L \|x - w\|$. By multiplying $\|x - w\|$ on both sides and applying \hyperref[prob-thm:cauchy-schwarz]{Cauchy-Schwarz}, we get 
		\begin{equation}
		  \langle \nabla f(x) - \nabla f(w), x - w \rangle \leq \| \nabla f(x) - \nabla f(w)\| \|x - w\| \leq L\|x - w\|^2 
		\end{equation}
		Since $f$ is $\mu$-strongly convex, 
		\begin{equation}
		  \langle \nabla f(x) - \nabla f(w), x - w \rangle  \geq \mu \|x - w\|^2 
		\end{equation}
		Therefore, combining the two gives 
		\begin{equation}
		  \mu\|x - w\|^2 \leq L \|x - w\|^2 \implies \mu < L
		\end{equation}
	\end{proof}

	So we are able to put some bounds on the norm of the gradient, and a natural next question is whether we can guarantee something about the distance to the function. For that, we need additional regularity assumptions, such as strong convexity. 

	\begin{theorem}[Linear Convergence of Objective Value on GD Under Strong Convexity]
	  \label{thm:linear-convergence-of-objective-value-on-gd}
	  Suppose that $f$ is $L$-smooth and $\mu$-strongly convex. Then, the iterates of GD satisfy 
		\begin{equation}
			f(w_k) - f(w^\ast) \leq \bigg( 1 - \frac{\mu}{L} \bigg)^k \big( f(w_0) - f(w^\ast) \big) 
		\end{equation}
		Therefore, for any $\epsilon > 0$, we have 
		\begin{equation}
			T \geq \frac{L}{\mu} \log \bigg( \frac{f(w_0) - f(w^\ast)}{\epsilon} \bigg) \implies f(w_k) - f(w^\ast) \leq \epsilon
		\end{equation}

	\begin{corollary}[Linear Convergence of Distance to Minimizer on GD Under Strong Convexity]
	  \label{thm:linear-convergence-of-distance-to-minimizer-on}
	  Suppose that $f$ is $L$-smooth and $\mu$-strongly convex. For any $\epsilon > 0$, we have 
		\begin{equation}
			k \geq \frac{L}{\mu} \log \bigg( \frac{f(w_0) - f(w^\ast)}{\epsilon} \bigg) \implies \|w_T - w^\ast\| \leq \sqrt{\frac{2 \epsilon}{\mu}}
		\end{equation}
	\end{corollary}
	\begin{proof}
		Strong convexity implies quadratic growth $f(w) - f(w^\ast) \geq \frac{\mu}{2}\|w - w^\ast\|^2$. 
	\end{proof}

	Note that for $L$-smooth functions in general, GD converges to criticality at a rather slow rate of $O(\epsilon^{-2})$, compared to $T \gtrsim \log(\epsilon^{-1})$. If $\epsilon = 10^{-6}$, then $\epsilon^{-2} = 10^{12}$---which is a very high number of iterations to get a gradient under $10^{-6}$---while $\log(\frac{1}{\epsilon}) \approx 14$---much cheaper to get the same bound on function suboptimality \textit{and} distance to minimizer. 

	\begin{definition}[Condition Number]
	  \label{def:condition-number}
		Thus, how fast GD converges is dictated by the ratio 
		\begin{equation}
			\kappa = \frac{L}{\mu} \in [1, \infty]
		\end{equation}
		known as the \textbf{condition number}. 
	\end{definition}

	This is pretty fast, but is there a faster algorithm than GD? For a particular function, we may be able to exploit its mathematical properties such find such a minima. For example, we can find the closed form of the minimum of a quadratic function by finding the roots of its gradient, basically making our algorithm $O(1)$. Therefore, we wish to evaluate GD over a \textit{class of functions}. For such a class, an algorithm may do really well on some $f \in \mathcal{F}$ but may do terribly on other functions. Therefore, we wish to find the \hyperref[def:minimax-oracle-error]{minimax oracle error} of GD on some defined $\mathcal{F}$. The following result gives a lower bound on the convergence of GD on strongly convex, $L$-smooth functions. 
	
	\begin{theorem}[Convergence Lower Bound of GD on Strongly Convex Functions]
	  \label{thm:convergence-lower-bound-of-gd-on-strongly}
		Let $L, \mu, \Delta > 0$, $\kappa = \frac{L}{\mu} \geq 1$, and $T \geq 1$. Then, there exists a $\mu$-strongly convex, $L$-smooth function $f: \mathbb{R}^{2T} \to \mathbb{R}$ with $f(w^\ast)= 0$ and $f(0) = \Delta$ such that every method satisfying\footnote{QUESTION: Is this a generalization of GD?}
		\begin{equation}
			x_{k+1} \in \Span\{\nabla f(x_0), \ldots, \nabla f(x_k)\}\footnote{GD automatically satisfies this, so we can think of this as a generalization of GD.}
		\end{equation}
		obeys 
		\begin{equation}
			\frac{f(x_T) - f^\ast}{\Delta} \geq \bigg( \frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1} \bigg)^{2T}
		\end{equation}
		The same holds for the ``best'' iterate. Whenever $\sqrt{\kappa} \geq 4$, then 
		\begin{equation}
			T \leq \min \bigg\{ \frac{d}{2}, \frac{\sqrt{\kappa}}{8} \log\Big( \frac{1}{\epsilon} \Big)\bigg\} \implies \frac{f(w_T) - f^\ast}{\Delta} > \epsilon
		\end{equation}
	\end{theorem}
	\begin{proof}
		The general idea is to first construct the minimizer $w^\ast$ with exponentially decaying entries and a ``chain'' function $f$ for it. We verify that the chain function is indeed in our class of $L$-smooth $\mu$-strongly \hyperref[mt-def:convex-function]{convex functions}, and then show that the chain function only reveals one entry per query. We convert this fact into a lower bound. For simplicity, write 
		\begin{equation}
			p = \frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1} \in (0, 1), \quad a = \sqrt{ \frac{2\Delta (1 + p)^2}{L (1 - p^2)}}
		\end{equation}
		\begin{enumerate}
		  \item \textit{Construct Chain Function}. Let us define our chain function\footnote{We can interpret this as the worst smooth and strongly convex function in the world.} as 
				\begin{equation}
					f: \mathbb{R}^{2T} \to \mathbb{R}, \quad f(w) = \frac{L}{2(1 + p)^2} \bigg[ (1 - p^2) (a - w_1)^2 + \sum_{i=1}^{2T-1} (x_{i+1} - p x_i)^2 \bigg]
				\end{equation}
				with minimizer $w^\ast = (a, ap, ap^2, \ldots, ap^{2T-1})$.\footnote{It is easy to show that this is indeed the minimizer since $f(w^\ast) = 0$ and $f(w) \geq 0$.} We denote 
				\begin{equation}
					f(0) = \frac{L}{2 (1 + p)^2} (1 - p^2) a^2 = \Delta
				\end{equation}

			\item \textit{Verify Smoothness and PL}. We can verify that $f$ is $\mu$-PL by showing that $f$ is $C^2$ and $\mu I \preceq \nabla^2 f(w)$, and that $f$ is $L$-smooth by showing that $f$ is $C^2$ and $\nabla^2 f(w) \preceq L I$. Since $f$ is a quadratic, we can write it as 
				\begin{equation}
					f(w) = \frac{L}{2(1 + p)^2} (w^T B w - (1 - p^2) a e_1), 
				\end{equation}
				where $e_1$ is the 1st standard basis vector of $\mathbb{R}^{2T}$ and $B$ is the tridiagonal matrix 
				\begin{equation}
					B = \begin{pmatrix} 
						1 & -p & & & \\ 
						-p & 1 + p^2 & -p & & \\ 
							 & -p & 1 + p^2 & & \\ 
							 & & & \ddots & -p \\ 
							 & & & -p & 1 + p^2
					\end{pmatrix} 
				\end{equation}
				Expanding, we get 
				\begin{equation}
					w^T B w = w_1^2 + w_{2T}^2 + (1 + p^2) \sum_{i=2}^{2T-1} w_i^2 - 2p \sum_{i=1}^{2T-1} w_i w_{i+1}
				\end{equation}
				Using the fact that $2 |uv| \leq u^2 + v^2$, we get that 
				\begin{align}
					w^T B w & \geq w_1^2 + w_2^2 + (1 + p^2) \sum_{i=2}^{2T-1} w_i^2 - p \bigg( w_1^2 + w_{2T}^2 + 2 \sum_{i=2}^{2T-1} w_i^2 \bigg) \\ 
									& = (1 - p) (w_1^2 + w_{2T}^2) + (1 - p)^2 \sum_{i=1}^{2T-1} w_i^2 \\ 
									& \geq (1 - p)^2 \|w\|^2
				\end{align}
				where for the last inequality, we used the fact that $(1 - p) \geq (1 - p)^2$. A similar argument yields $w^T B w \leq (1 + p)^2 \|w\|^2$. Therefore, 
				\begin{equation}
					\mu I = \frac{L}{\kappa} I = L \bigg( \frac{1 - p}{1 + p} \bigg)^2 I \preceq \underbrace{\frac{L}{(1 + p)^2} B}_{\nabla^2 f(w)} \preceq L I
				\end{equation}

			\item \textit{Each query reveals one coordinate}. We wish to show that the iterates of GD on our function $f$ satisfies $x_{k+1} \in \Span\{\nabla f(x_0), \ldots, \nabla f(x_k)\}$. We do this with induction. Let 
				\begin{equation}
					E_0 = \{0\}, \quad E_j = \Span\{e_1, \ldots, e_j\} 
				\end{equation}
				The gradient of $f$ is 
				\begin{equation}
					\nabla f(w) = \frac{L}{(1 + p)^2} \big( Bw - (1 - p^2) a e_1 \big) 
				\end{equation}
				So, for $w_0 \in E_0$, we have $w_1 \in \Span\{ \nabla f(0)\} = E_1$. This proves the base case. Now suppose that $w_k \in E_k$. Since $B_k$ is tridiagonal, paying attention to the entries of the matrix will show us that $Bw_k \in E_{k+1}$. Hence, $\nabla f(w_k) = \frac{L}{(1 + p)^2} B w_k - \frac{L(1 - p^2)}{(1 + p)^2} a e_1 \in E_{k+1}$. 

			\item \textit{Construct Lower Bound}. For index $T$ (not the transpose), we know that for any vector $w \in E_{T+1}$, we must have that 
				\begin{equation}
					|\underbrace{(w - w^\ast)_{T+1}}_{z_T}| = a p^T
				\end{equation} 
				since $w_{T+1} = 0$. Our goal is to use this as a lower bound for $f(w) - f(w^\ast)$. To do so, notice that since $f(w^\ast) = 0$ and $w_{j+1}^\ast = p w_j^\ast$, we have 
				\begin{equation}
					f(w) - f(w^\ast) = \frac{L}{2 (1 + p)^2} \bigg[ (1 - p^2) z_1 + \sum_{j=1}^{2T - 1} (z_{j+1} - p z_i )^2 \bigg] 
				\end{equation}
				By first expanding $z_{j+1} = p z_j + (z_{j+1} - p z_j)$ and subsequently \hyperref[prob-thm:cauchy-schwarz]{Cauchy-Schwarz}, we have 
				\begin{align}
					a p^T & = p^T z_1 - \sum_{j=1}^T p^{T - j} (z_{j+1} - p z_j) \\ 
								& = \langle \begin{pmatrix} \frac{p^T}{\sqrt{1 - p^2}} \\ p^{T-1} \\ \vdots \\ 1 \end{pmatrix} , \begin{pmatrix} \sqrt{1 - p^2} z_1 \\ z_2 - p z_1 \\ \vdots \\ z_{T+1} - p z_T \end{pmatrix} \\ 
								& \leq \bigg( \frac{p^{2T}}{(1 - p^2)} + \sum_{j=1}^t p^{2 (T - j)}\bigg)^{1/2} \cdot \bigg( (1 - p^2) z_1^2 + \sum_{j=1}^T (z_{j+1} - p z_j)^2 \bigg)^{1/2} 
				\end{align}
				The first term simplifies to 
				\begin{equation}
					\frac{p^{2T}}{1 - p^2} + \frac{1 - p^{2T}}{1 - p^2} = \frac{1}{1 - p^T} 
				\end{equation}
				The second term can be bounded as 
				\begin{equation}
					(1 - p^2) z_1^2 + \sum_{j=1}^T (z_{j+1} - p z_j)^2  \leq (1 - p^2) z_1^2 + \sum_{j=1}^{2T-1} (z_{j+1} - p z_j)^2 = \frac{2(1 + p)^2}{L} \big( f(w) - f(w^\ast) \big) 
				\end{equation}
				Squaring and combining all of this yields 
				\begin{equation}
					a^2 p^{2T} \leq \frac{2 (1 + p)^2}{L (1 - p^2)} \big( f(w) - f(w^\ast) \big) 
				\end{equation}
				Rearranging and simplifying gives 
				\begin{equation}
					f(w) - f(w^\ast) \geq \frac{L(1 - p^2)}{2 (1 + p^2)} a^2 p^{2T} = \Delta \bigg( \frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1} \bigg)^{2T}
				\end{equation}
		\end{enumerate}
	\end{proof}

	So what happens when $\kappa \log(1/\epsilon) \lesssim T \lesssim \sqrt{\kappa} \log(1/\epsilon)$. The answer to this question depends on the family of functions. 

\subsection{Polyak-Lojasiewicz Inequality}

	In recent decades, strong convexity fell out of favor for the PL-inequality, which generalizes the notion of strong convexity. 

	\begin{definition}[Polyak-Lojasiewicz Inequality]
	  \label{def:polyak-lojasiewicz-inequality}
		A differentiable function $f: \mathbb{R}^d \to \mathbb{R}$ satisfies the \textbf{PL-inequality}\footnote{This was published in 1963 and was ignored until 2016.} if there exists a $\mu > 0$ s.t. 
		\begin{equation}
			\mu \big( f(x) - \inf{f} \big) \leq \frac{1}{2} \| \nabla f(x) \|^2 \text{ for all } x \in \mathbb{R}^d
		\end{equation}
	\end{definition}

	The first thing is that we should think of this as a weaker notion of strong convexity. But this is still a relatively strong assumption, and in particular it yields that $\nabla f(x) = 0 \implies f(x) = \inf{f}$

	\begin{lemma}[Strong Convexity Implies PL-Inequality]
	  \label{thm:strong-convexity-implies-pl-inequality}
		Suppose $f: \mathbb{R}^d \to \mathbb{R}$ is differentiable and $\mu$-strongly convex. Then, it satisfies $\mu$-PL. 
	\end{lemma}
	\begin{proof}
	  Since $f$ is strongly convex, it attains the global minimizer $w^\ast$. Therefore, 
		\begin{align}
			f(w^\ast) & \geq f(w) + \langle \nabla f(w), w^\ast - w \rangle + \frac{\mu}{2} \| w - w^\ast \|^2 \\ 
								& = f(w) + \frac{\mu}{2} \bigg[ \frac{\nabla f(w)}{\mu} - (w^\ast - w) - \frac{1}{\mu^2} \|\nabla f(w) \|^2 \bigg] \\ 
								& \geq f(w) - \frac{1}{2 \mu} \|\nabla f(w)\|^2
		\end{align}
		and we rearrange the terms to get PL. 
	\end{proof}

	PL does not require convexity nor even a unique minimizer. 

	\begin{example}[A PL But Nonconvex Function]
	  \label{ex:a-pl-but-nonconvex-function}
		$f(x) = x^2 + 3\sin^2 x$ is nonconvex but satisfies PL. \cite{DBLP:journals/corr/KarimiNS16}
	\end{example}

	\begin{lemma}[Properties of the PL-Inequality]
	  \label{thm:properties-of-the-pl-inequality}
		Let $f: \mathbb{R}^d \to \mathbb{R}$ be differentiable and bounded below, and satisfy $\mu$-PL. 
		\begin{enumerate}
		  \item \textit{Monotonicity}. $f$ satisfies $\mu'$-PL for every $0 < \mu' \leq \mu$. 
			\item \textit{Scaling}. For $a > 0$, $af$ satisfies $(a\mu)$-PL. For any $c \in \mathbb{R}$, $f + c$ satisfies $\mu$-PL. 
			\item \textit{Compatibility with Smoothness}. If $f$ is also $L$-smooth, then $\mu \leq L$ (unless $f$ is constant). 
			\item \textit{Gradient Lower Bound}. $\|\nabla f(x)\| \geq \sqrt{2\mu \big( f(x) - \inf f \big)}$ for all $x$. 
			\item \textit{Affine Precomposition}. If $A \in \mathbb{R}^{m \times d}$ is onto and $b \in \mathbb{R}^m$, then $\varphi(x) = f(Ax + b)$ for $f: \mathbb{R}^m \to \mathbb{R}$ satisfies $(\mu \lambda_{\min}(AA^T))$-PL, where $\lambda_{\min}(AA^T) = \inf_{\|v\| = 1} v^T A A^T v$. 
		\end{enumerate}
	\end{lemma}
	\begin{proof}
		(1) and (4) are just a rewriting of the definition. 

		(2) Since $\inf(af + c) = a \inf f + c$ for $a > 0$, 
		\begin{equation}
		  \tfrac{1}{2}\|\nabla (af)(x)\|^2 = a^2 \cdot \tfrac{1}{2}\|\nabla f(x)\|^2 \geq a^2 \mu \big( f(x) - \inf f \big) = a\mu \big( af(x) - \inf (af) \big)
		\end{equation}
		and adding the constant $c$ changes neither side. 

		(3) By the \hyperref[thm:descent-lemma]{descent lemma}, for any $x$, 
		\begin{equation}
		  \inf f \leq f\big(x - \tfrac{1}{L}\nabla f(x)\big) \leq f(x) - \frac{1}{2L}\|\nabla f(x)\|^2
		\end{equation}
		so $\|\nabla f(x)\|^2 \leq 2L\big(f(x) - \inf f\big)$. Combining with PL, $2\mu \big(f(x) - \inf f\big) \leq 2L \big(f(x) - \inf f\big)$ for all $x$. If $f$ is not constant, then it is not constantly equal to its infimum, so we may pick $x$ with $f(x) > \inf f$ and divide to get $\mu \leq L$. 

		(5) Since $A$ is onto, $x \mapsto Ax + b$ is onto $\mathbb{R}^m$, and so $\inf \varphi = \inf f$. Also $AA^T$ is positive definite so $\lambda_{\min}(AA^T) > 0$, and for any $v$, $\|A^T v\|^2 = v^T A A^T v \geq \lambda_{\min}(AA^T) \|v\|^2$. Therefore, with $z = Ax + b$ and $\nabla \varphi(x) = A^T \nabla f(z)$, 
		\begin{equation}
		  \tfrac{1}{2}\|\nabla \varphi(x)\|^2 \geq \lambda_{\min}(AA^T) \cdot \tfrac{1}{2}\|\nabla f(z)\|^2 \geq \mu \lambda_{\min}(AA^T) \big( f(z) - \inf f \big) = \mu \lambda_{\min}(AA^T) \big( \varphi(x) - \inf \varphi \big)
		\end{equation}
	\end{proof}

	\begin{theorem}[Linear Convergence of Objective Value on GD Under PL Inequality]
		\label{thm:linear_convergence_pl}
		Suppose that $f$ is $L$-smooth and satisfies $\mu$-PL. Then, the iterates of GD $w_{k+1} = w_k - \frac{1}{L} \nabla f(w_k)$ satisfy 
		\begin{equation}
			f(w_k) - \inf(f) \leq \bigg( 1 - \frac{\mu}{L} \bigg)^k \big( f(w_0) - f(w^\ast) \big) 
		\end{equation}
		Therefore, for any $\epsilon > 0$, we have 
		\begin{equation}
			t \geq \frac{L}{\mu} \log \bigg( \frac{f(w_0) - f(w^\ast)}{\epsilon} \bigg) \implies f(w_t) - f(w^\ast) \leq \epsilon
		\end{equation}
		i.e. $t \gtrsim \log(\epsilon^{-1}) \implies f(w_t) - f(w^\ast) < \epsilon$. 
	\end{theorem}
	\begin{proof}
	  Using the \hyperref[thm:descent-lemma]{descent lemma} first and then $\mu$-PL second, we get 
		\begin{align}
			f(w_k) - f(w^\ast) & \leq f(w_{k-1}) - f(w^\ast) - \frac{1}{2L} \|\nabla f(w_{k-1})\|^2 \\ 
												 & \leq f(w_{k-1}) - f(w^\ast) - \frac{\mu}{L} \big( f(w_{k-1}) - f(w^\ast) \big) \\ 
												 & = \big(1 - \frac{\mu}{L} \big) \big( f(w_{k-1}) - f(w^\ast) \big) \\ 
												 & = \big(1 - \frac{\mu}{L} \big)^k \big( f(w_{0}) - f(w^\ast) \big) 
		\end{align}
		where on the last step, we recursed. 
	\end{proof}

	\begin{theorem}[Linear Convergence of GD in Overparameterized Linear Regression]
	  \label{thm:linear-convergence-of-gd-in-overparameterized}
		Consider the linear regression model where we want to optimize
		\begin{equation}
			f_n (w) = \frac{1}{2n} \sum_{i=1}^n (w^T x_i - y_i)^2 = \frac{1}{2n} \|Xw - y\|^2
		\end{equation}
		where $d > n$ (overparamterized) and $\mathrm{rank}(X) = n$ (independent inputs). Define 
		\begin{equation}
			S^\ast = \{w \in \mathbb{R}^d \colon Xw = y\}, \quad f^\ast = \min{f_n (w)} = 0
		\end{equation}
		Suppose we ran GD with stepsize $\eta_k = \frac{1}{\|X\|_{\mathrm{op}}^2}$ on the loss $f_n$. Then, its iterates satisfy 
		\begin{equation}
			\inf_{w^\ast \in S^\ast} \big\{ f(w) - f(w^\ast) \big\} \leq \bigg(1 - \frac{\lambda_{\min} (XX^T)}{\lambda_{\max} (XX^T)} \bigg)^k \, \inf_{w^\ast \in S^\ast} \big\{ f(w_0) - f(w^\ast) \big\}
		\end{equation}
	\end{theorem}
	\begin{proof}
		Let $f(z) = \frac{1}{2n} \|z\|^2$. Then, $\nabla f(z) = \frac{1}{n} z$, and so $f$ is both $\frac{1}{n}$-smooth and $\frac{1}{n}$-PL. Therefore, we can apply our properties of affine precomposition to show that $\frac{1}{2n} \|Xw - y\|^2$ is both $\frac{\|X\|^2}{n}$-smooth---i.e. $\frac{\lambda_{\max}(XX^T)}{n}$-smooth---and $\frac{\lambda_{\min} (XX^T)}{n}$-PL. Therefore, fix some $w^\ast \in S^\ast$, and applying our \hyperref[thm:linear_convergence_pl]{linear convergence result} gives us 
		\begin{equation}
			f(w_k) - f(w^\ast) \leq \bigg( 1 - \frac{\lambda_{\min} (XX^T)}{\lambda_{\max} (XX^T)} \bigg)^k \big(f(w_k) - f(w_0) \big) 
		\end{equation}
		Take the infimum over all $w^\ast \in S^\ast$ and we are done. 
	\end{proof}
	
	Now we try to establish lower bounds on \hyperref[def:function-class]{function classes} of PL functions. Note that the lower bound result for strongly \hyperref[mt-def:convex-function]{convex functions} hold since strong convexity implies PL. 

	\begin{theorem}[Convergence Lower Bound of GD on Strongly Convex Functions]
	  \label{thm:convergence-lower-bound-of-gd-on-strongly-2}
		Let $L, \mu, \Delta > 0$, $\kappa = \frac{L}{\mu} \geq 1$, and $T \geq 1$. Then, there exists a $\mu$-PL, $L$-smooth function $f: \mathbb{R}^{2T} \to \mathbb{R}$ with $f(w^\ast)= 0$ and $f(0) = \Delta$ such that every method satisfying
		\begin{equation}
			x_{k+1} \in \Span\{\nabla f(x_0), \ldots, \nabla f(x_k)\}\footnote{GD automatically satisfies this, so we can think of this as a generalization of GD.}
		\end{equation}
		obeys 
		\begin{equation}
			\frac{f(x_T) - f^\ast}{\Delta} \geq \bigg( \frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1} \bigg)^{2T}
		\end{equation}
		The same holds for the ``best'' iterate. Whenever $\sqrt{\kappa} \geq 4$, then 
		\begin{equation}
			T \leq \min \bigg\{ \frac{d}{2}, \frac{\sqrt{\kappa}}{8} \log\Big( \frac{1}{\epsilon} \Big)\bigg\} \implies \frac{f(w_T) - f^\ast}{\Delta} > \epsilon
		\end{equation}
	\end{theorem}
	\begin{proof}
	  We can always find such a $\mu$-strongly \hyperref[mt-def:convex-function]{convex function}, which is $\mu$-PL. 
	\end{proof}

	For functions satisfying PL, we know that this is a larger family, so perhaps the lower bound might be higher. In fact, this is the case. 

	\begin{theorem}[Convergence Lower Bound of GD on $\mu$-PL Functions\cite{pmlr-v195-yue23a}]
	  \label{thm:convergence-lower-bound-of-gd-on-mu-pl}
		Let $L, \mu, \Delta > 0$ with $\kappa = \frac{L}{\mu} \geq 10^5$ and $T \geq 1$. Then, there exists an $L$-smooth function $f: \mathbb{R}^{2T} \to \mathbb{R}$ with $f(w^\ast) = 0$ and $f(0) = \Delta$ such that every method satisfying 
		\begin{equation}
			x_{k+1} \in \Span\{\nabla f(x_0), \ldots, \nabla f(x_k)\}\footnote{GD automatically satisfies this, so we can think of this as a generalization of GD.}
		\end{equation}
		obeys 
		\begin{equation}
			\frac{f(w_k) - f(w^\ast)}{\Delta} \geq \exp (-10^5 T/\kappa)
		\end{equation}
		The same bound holds for the best iterate. Thus, we get 
		\begin{equation}
			T \leq \min \bigg\{ \frac{d}{2}, \frac{\kappa}{10^5} \log (1/\epsilon) \bigg\} \implies \frac{f(w_T) - f(w^\ast)}{\Delta} \geq \epsilon
		\end{equation}
	\end{theorem}
	\begin{proof}
	  See cited paper. 
	\end{proof}

	Hence, up to constants, we have 
	\begin{align}
		T \lesssim \kappa \log(1/\epsilon) & \implies f(w_T) - f(w^\ast) > \epsilon \\ 
		T \gtrsim \kappa \log(1/\epsilon) & \implies f(w_T) - f(w^\ast) \leq \epsilon
	\end{align}

	\begin{example}[Implicit Regularization]
	  \label{ex:implicit-regularization}
	  Whenever a problem is ill-posed (more than one solution), it is common to add a regularier to promote structure. In overparameterized OLS linear regression, it is natural to use ridge regression. 
		\begin{equation}
			w_\alpha = \argmin_w \frac{1}{2n} \|Xw - y\|^2 + \frac{\alpha}{2} \|w\|^2
		\end{equation}
		Uniqueness of $w_\alpha$ follows by strong convexity. By first-order optimality conditions, 
		\begin{equation}
			\frac{1}{n} X^T (Xw_\alpha - y) + \alpha w_\alpha = 0 \iff w_\alpha = (X^T X + n \alpha I)^{-1} X^T y
		\end{equation}
		This gives us a set of solutions $\{w_\alpha \colon \alpha > 0\}$ that approaches $S^\ast$ as $\alpha \to 0^+$. 
	\end{example}

	\begin{lemma}[Limit of the Gradient Flow Path]
	  \label{thm:limit-of-the-gradient-flow-path}
 
	  We claim 
		\begin{equation}
			w^+ \coloneqq \lim_{\alpha \to 0^+} w_\alpha  = \min_{w \in S^\ast} \|w\|
		\end{equation}
	\end{lemma}
	\begin{proof}
	  Let $X = U \Sigma V^T$ be the SVD of $X$. Then, 
		\begin{align}
			\lim_{\alpha \to 0^+} & = \lim_{\alpha \to 0^+} V \mathrm{diag} \big( \frac{\sigma_i}{\sigma_i^2 + \alpha n} \big) U^T y \\ 
														& = V \Sigma^{-1} U^T y \\ 
														& = X^T (XX^T)^{-1} y = w^+
		\end{align}
		We claim that $w^+ \perp \ker(X)$. Let $z \in \ker(X)$. 
		\begin{equation}
			(w^+)^T z = y^T (X X^T)^{-1} X z = 0
		\end{equation}
		let $w \in S^\ast \setminus \{w^+\}$. Then, there exists a $z \in \ker(X)$ such that $w = w^+ + w$ and so 
		\begin{equation}
			\|w\|^2 = \|w^+\|^2 + \|z\|^2 > \|w^+\|^2
		\end{equation}
		which completes the proof. 
	\end{proof}

	In turn, GD also favors a similar solution.

\subsection{Momentum}

	Recall that \hyperref[ex:quadratics]{badly conditioned quadratics have slow GD convergence}. What we want to do is to making learning progress faster in the low curvature direction. A natural idea due to Polyak is to add momentum, called \textit{Polyak's Heavy-Ball Method}. 

	\begin{definition}[Gradient Descent with Momentum]
		\label{def:gradient_descent_momentum}
		Let $\mathcal{F}$ be the class of differentiable functions $f: S \subset \mathbb{R}^d \to \mathbb{R}$. Fix an initial point $x_0 \in S$ and a sequence of positive step sizes $(\eta_t)$. Then, \textbf{gradient descent with momentum $\beta > 0$} is a deterministic algorithm defined by the first-order oracle
		\begin{equation}
			\mathcal{O}_f : S \to \mathbb{R} \times \mathbb{R}^d, \quad \mathcal{O}_f (x) = \big( f(x), \nabla f(x) \big)
		\end{equation}
		and update maps $\Phi_t : (S \times (\mathbb{R} \times \mathbb{R}^d))^{t+1} \to S$ defined
		\begin{align}
			x_{t+1} & = \Phi_t \big(x_0, (f(x_0), \nabla f(x_0)), \ldots, x_t, (f(x_t), \nabla f(x_t)) \big) \\ 
							& \coloneqq  x_t - \eta_t \nabla f(x_t) + \beta (x_t - x_{t-1})\footnote{where the term $\beta (w_k - w_{k-1})$ ``moves in our previous direction.''}
		\end{align}
		After $k$ iterations, the algorithm outputs $w_{\mathcal{A}_k} (f) = x_k$. 
	\end{definition}

	If the stepsize $\eta$ is constant, then we can unroll the updates and get 
	\begin{equation}
		w_{k+1} - w_k = - \eta \sum_{j=0}^k \beta^{k-j} \nabla f(w_j)
	\end{equation}
	Thus, we are using all previous gradient information, albeit reweighted. Another algorithmic design principle is that we should use previous first-order information when possible. It turns out that for strongly convex quadratics, this is actually optimal. 

	\begin{theorem}[Momentum GD is Optimal for Quadratics \cite{POLYAK19641}]
	  \label{thm:momentum-gd-is-optimal-for-quadratics-cite}
		Let $0 < \mu < L$, $\kappa = L/\mu$, and define
		\begin{equation}
			\mathcal F_{\mu,L} = \left\{ f(x)=\frac12 x^T A x-b^T x+c: A=A^T,\ \mu I\preceq A\preceq LI \right\}.
		\end{equation}
		Consider heavy-ball momentum initialized at $x_{-1}=x_0$,
		\begin{equation}
			x_{k+1} = x_k-\eta\nabla f(x_k) + \beta(x_k-x_{k-1}).
		\end{equation}
		With parameters
		\begin{equation}
			\eta^\ast = \frac{4}{(\sqrt L+\sqrt\mu)^2}, \qquad \beta^\ast = \left( \frac{\sqrt L-\sqrt\mu} {\sqrt L+\sqrt\mu} \right)^2,
		\end{equation}
		for every $f\in\mathcal F_{\mu,L}$ and every $k\geq0$,
		\begin{equation}
			\|x_k-x^\ast\| \leq (1+2k) \left( \frac{\sqrt\kappa-1} {\sqrt\kappa+1} \right)^k \|x_0-x^\ast\|.
		\end{equation}
		Consequently, for every
		$f\in\mathcal F_{\mu,L}$,
		\begin{equation}
			f(x_k)-f^\ast \leq \frac L2(1+2k)^2 \left( \frac{\sqrt\kappa-1} {\sqrt\kappa+1} \right)^{2k} \|x_0-x^\ast\|^2.
		\end{equation}
	\end{theorem}

	\begin{theorem}[Suboptimality of Heavy Ball for General Smooth Functions]
	  \label{thm:suboptimality-of-heavy-ball-for-general-smooth}
		For general smooth functions, and \hyperref[mt-def:convex-function]{convex functions}, it is sub-optimal. (Lessard, Recht, Packard 2016) 
	\end{theorem}

	However, for general smooth functions, momentum is not optimal, so we would like to improve it. We will do so by incorporating Nesterov's strategy. To do this, we can try to incorporate all previous information like we did with the gradients. Recall that \hyperref[def:gradient_descent]{gradient descent} tries to minimize 
	\begin{equation}
		f(w_{t+1}) = \argmin_{w} \bigg\{ \underbrace{f(w_t) + \langle \nabla f(w_t), w - w_t \rangle}_{\text{convex lower bound}} + \frac{1}{\eta_t} \|w - w_t\|^2 \bigg\}
	\end{equation}

	Recall that convexity gives a linear lower bound on $f$. 
	\begin{equation}
		l_y (z) \coloneqq f(y) + \langle \nabla f(y), z - y \rangle \leq f(z) \quad \forall z \label{heart}
	\end{equation}
	GD uses one gradient to build the model it minimizes at each iteration. Just like how momentum does, let's use all of them by taking a \hyperref[lin-def:linear-combination]{linear combination} of such linear approximations, and then adding the quadratic anchor. At the $k$th iteration, we should have information on the $k$ previous gradients. Define the following. 

	\begin{definition}[Nesterov's Auxiliary Estimation Function]
	  \label{def:nesterovs-auxiliary-estimation-function}
		Given $a_j > 0$ and $y_j \in \mathbb{R}^d$, define 
		\begin{equation}
			\Phi_k (z) = \sum_{j=1}^k a_j l_{y_{j-1}} (z) + \frac{1}{2} \|z - w_0\|^2, \quad a_j > 0, y_j \in \mathbb{R}^d
		\end{equation}
		It is not clear what the coefficients $a_j$ (how much ``weight'' we should put to this gradient information) and $y_j$ (the center points where the evaluate the gradient at) should be, but we will decide this later. 
	\end{definition}

	We can then bound $\Phi_k$ to get 
	\begin{equation}
		\Phi_k (z) \leq A_k f(z) + \frac{1}{2} \|z - w_0\|^2, \quad \forall z, \qquad A_k = a_1 + \ldots a_k
	\end{equation}
	Just like how we did for GD, we would like to find $\argmin_z \Phi_K (z)$. Since $\Phi_K(z)$ is differentiable and strongly convex, the \hyperref[real-def:critical-point]{critical point} is the unique minimizer. Evaluating gives 
	\begin{equation}
		0 = \nabla \Phi_k (z) = \sum_{j=1}^k a_j \nabla f(y_{j-1}) + (z - w_0) \implies z = w_0 - \sum_{j=1}^k a_j \nabla f(y_{j-1})
	\end{equation}
	Therefore, at the $k$th iteration, our $z_k$ should be the following, which can be stated recursively 
	\begin{equation}
		z_k = w_0 - \sum_{j=1}^k a_j \nabla f(y_{j-1}) \iff z_{k+1} = z_k - a_{k+1} \nabla f(y_k)
	\end{equation}
	Therefore, at the $k$th iteration, we wish to minimize the isotropic\footnote{Since $\Phi_k$ is quadratic consisting of linear terms and an isotropic quadratic, so $\nabla^2 \Phi_k(z) = I$.} quadratic bowl with minimizer at $z_k$. 
	\begin{equation}
		\Phi_k (z) = \Phi_k (z_k) + \langle \nabla f(z_k), z - z_k \rangle + \frac{1}{2} \|z - z_k\|^2 = \Phi_k (z_k) + \frac{1}{2} \|z - z_k\|^2
	\end{equation}


	To construct a convergence guarantee, we want to show something like 
	\begin{equation}
	 	f(w_k) - f^\ast \leq \frac{\frac{1}{2} \|w_0 - w^\ast\|}{A_k} \iff A_k f(w_k) \leq A_k f^\ast + \frac{1}{2} \|w_0 - w^\ast\|^2
	\end{equation}
	where $A_k$ will increase. But we already have 
	\begin{equation}
		\Phi_k (w^\ast) \leq A_k f^\ast + \frac{1}{2} \|w^\ast - w_0\|^2
	\end{equation}
	Therefore, if we could prove that $A_k f(w_k) \leq \Phi_k(w^\ast)$, then we can indeed show that 
	\begin{equation}
		A_k f(w_k) \leq \Phi_k (w^\ast) \leq A_k f^\ast + \frac{1}{2} \|w^\ast - w_0\|^2
	\end{equation}
	So this is what we will show. However, the RHS of $A_k f(w_k)  \leq \Phi_k (w^\ast)$ depends on the unknown minimizer $w^\ast$. Instead, we impose the stronger condition that 
	\begin{equation}
	  A_k f(w_k) \leq \min_z \Phi_k (z) = \Phi_k (z_k)
	\end{equation}
	If this holds, then we get 
	\begin{equation}
		A_k f(w_k) \leq \Phi_k (z_k) \leq \Phi_k (w^\ast) \leq A_k f^\ast + \frac{1}{2} \|w_0 - w^\ast\|^2 \implies f(w_k) - f^\ast \leq \frac{\|w_0 - w^\ast\|^2}{2 A_k}
	\end{equation}

	So our job is to now choose the evaluation points $y_k$ and weights $a_{k+1}$ such that the inequality holds. Therefore, the general idea is to construct 3 sequences of points in $\mathbb{R}^d$. 
	\begin{enumerate}
	  \item The $w_k$ which are the weight updates. 
		\item The $z_k$ which are the optima of our auxiliary estimate function. 
		\item The $y_k$ which are the evaluation points of our gradients. 
	\end{enumerate}

	So, let's assume we always update 
	\begin{equation}
		z_{k+1} = z_k - a \nabla f(y), \quad w_{k+1} = y - \frac{1}{L} \nabla f(y)
	\end{equation}
	A question is that suppose that $A_k f(w_k) \leq \min_{z} \Phi_k (z) = \Phi(z_k)$ holds. Then, how we do we set $a$ and $y$ to ensure $A_{k+1} f(w_{k+1}) \leq \min_{z} \Phi_{k+1} (z) = \Phi(z_{k+1})$?

	\begin{lemma}[One-Step Estimate]
	  \label{thm:one-step-estimate}
		Suppose $f: \mathbb{R}^d \to \mathbb{R}$ is convex and $L$-smooth,
		and assume that $A_k f(w_k) \leq \Phi_k(z_k)$. Consider the updates
		\begin{equation}
			z_{k+1} = z_k - a_{k+1}\nabla f(y_k), \qquad w_{k+1} = y_k - \frac{1}{L}\nabla f(y_k),
		\end{equation}
		where $A_{k+1} = A_k + a_{k+1}$. Then,
		\begin{align}
			\Phi_{k+1}(z_{k+1}) \geq A_{k+1}f(w_{k+1})
			+ \left\langle \nabla f(y_k), A_k(w_k-y_k) + a_{k+1}(z_k-y_k) \right\rangle + \frac{1}{2} \big(\frac{A_{k+1}}{L}-a_{k+1}^2\big) \|\nabla f(y_k)\|^2. 
		\end{align}
		In particular, 
		\begin{equation}
			y_k = \frac{A_kw_k+a_{k+1}z_k}{A_{k+1}}, a_{k+1}^2 \leq \frac{A_{k+1}}{L} \implies A_{k+1}f(w_{k+1}) \leq \Phi_{k+1}(z_{k+1}).
		\end{equation}
	\end{lemma}
	\begin{proof}
		By definition of $\Phi_{k+1}$ and the inductive hypothesis,
		\begin{align}
			\Phi_{k+1}(z_{k+1})
			&= \Phi_k(z_{k+1}) + a_{k+1}l_{y_k}(z_{k+1}) \\
			&= \Phi_k(z_k) + \frac{1}{2}\|z_{k+1}-z_k\|^2
				+ a_{k+1}\left(
					f(y_k) + \langle \nabla f(y_k),z_{k+1}-y_k\rangle
				\right) \\
			&\geq A_k f(w_k) + \frac{1}{2}\|z_{k+1}-z_k\|^2
				+ a_{k+1}\left(
					f(y_k) + \langle \nabla f(y_k),z_{k+1}-y_k\rangle
				\right).
		\end{align}
		Substituting $z_{k+1}=z_k-a_{k+1}\nabla f(y_k)$ gives
		\begin{align}
			\Phi_{k+1}(z_{k+1})
			&\geq A_k f(w_k) + a_{k+1}f(y_k)
				+ a_{k+1}\langle \nabla f(y_k),z_k-y_k\rangle \\
			&\quad - \frac{a_{k+1}^2}{2}\|\nabla f(y_k)\|^2.
		\end{align}
		By convexity,
		\begin{equation}
			f(w_k) \geq f(y_k)
				+ \langle \nabla f(y_k),w_k-y_k\rangle.
		\end{equation}
		Therefore,
		\begin{align}
			\Phi_{k+1}(z_{k+1})
			&\geq A_{k+1}f(y_k)
				+ \left\langle \nabla f(y_k),
					A_k(w_k-y_k)+a_{k+1}(z_k-y_k)
				\right\rangle \\
			&\quad - \frac{a_{k+1}^2}{2}\|\nabla f(y_k)\|^2.
		\end{align}
		Finally, by the \hyperref[thm:descent-lemma]{descent lemma},
		\begin{equation}
			f(w_{k+1})
			\leq f(y_k)-\frac{1}{2L}\|\nabla f(y_k)\|^2.
		\end{equation}
		Hence,
		\begin{align}
			\Phi_{k+1}(z_{k+1})
			&\geq A_{k+1}f(w_{k+1})
				+ \left\langle \nabla f(y_k),
					A_k(w_k-y_k)+a_{k+1}(z_k-y_k)
				\right\rangle \\
			&\quad + \frac{1}{2}
				\left(\frac{A_{k+1}}{L}-a_{k+1}^2\right)
				\|\nabla f(y_k)\|^2.
		\end{align}
		Choosing
		\begin{equation}
			y_k=\frac{A_kw_k+a_{k+1}z_k}{A_{k+1}}
		\end{equation}
		makes the inner-product term vanish. Moreover, if
		\begin{equation}
			a_{k+1}^2\leq\frac{A_{k+1}}{L},
		\end{equation}
		the remaining term is nonnegative. Consequently,
		\begin{equation}
			A_{k+1}f(w_{k+1})
			\leq\Phi_{k+1}(z_{k+1}),
		\end{equation}
		as desired.
	\end{proof}

	A direct consequence of this is the analysis of vanilla GD. Take 
	\begin{equation}
		a_j = \frac{1}{L}, \quad z_0 = w_0
	\end{equation}
	Then, it isn't too hard to see that 
	\begin{equation}
	  y_k = z_k = w_k \quad \forall k 
	\end{equation}

	Note that if we just took $a_k = \frac{1}{L}$, then we would automatically have 
	\begin{equation}
		a_{k+1}^2 = \frac{1}{L^2} \leq \frac{k+1}{L^2} = \frac{A_{k+1}}{L}
	\end{equation}
	but this leaves a lot of slack. Instead, we can take $a_k = \sqrt{\frac{A_k}{L}}$. Then, 
	\begin{equation}
		\sqrt{A_{k+1}} - \sqrt{A_k} = \frac{a_{k+1}}{\sqrt{A_{k+1}} - \sqrt{A_k}} \geq \frac{a_{k+1}}{2 \sqrt{A_{k+1}}} = \frac{1}{2 \sqrt{L}}
	\end{equation}
	which implies that 
	\begin{equation}
		\sqrt{A_{k+1}} - \sqrt{A_0} \geq \frac{k}{2\sqrt{L}} \implies A_{k+1} \geq \frac{k^2}{2L}
	\end{equation}
	This allows the $A_{k+1}$ to increase quadratically, but the price to pay is that then the coefficients of the gradients might become too big, so $z_k$ might diverge if we don't couple it with a much more conservative $w_k$. To see why this is more conservative, note that 
	\begin{align}
		w_{k+1} & = y_k - \frac{1}{L} \nabla f(y_k) \\ 
						& = \frac{A_k w_k + a_{k+1} z_k}{A_{k+1}} - \frac{a_{k+1}^2}{A_{k+1}} \nabla f(y_k) \\ 
						& = \frac{A_k w_k + a_k z_{k+1}}{A_{k+1}}
	\end{align}
	This is precisely the method that Nesterov proposed in 83. 

	\begin{definition}[Nesterov Accelerated Gradient Descent]
	  \label{def:nesterov-accelerated-gradient-descent}
		We set $a_{k+1}$ recursively as 
		\begin{equation}
			L a_{k+1}^2 = A_k + a_{k+1}
		\end{equation}
		and iteratively update 
		\begin{align}
			y_k & = \frac{A_k w_k + a_{k+1} z_k}{A_{k+1}} \\ 
			w_{k+1} & = y_k - \frac{1}{L} \nabla f(y_k) \\ 
			z_{k+1} & = z_k - a_{k+1} \nabla f(y_k)
		\end{align}
	\end{definition}

	\begin{theorem}[Convergence]
	  \label{thm:convergence}
		Suppose that $f: \mathbb{R}^d \to \mathbb{R}$ is convex, $L$-smooth function. Then, the iterates of AGD satisfy 
		\begin{equation}
			f(w_k) - f^\ast \leq \frac{2L \|w_0 - w^\ast\|^2}{k^2} \quad \forall k
		\end{equation}
	\end{theorem}
	\begin{proof}
	  We first prove that $A_k f(w_k) \leq \min_z \Phi_k(z) = \Phi_k (z_k)$ holds for all $k$. For $k = 0$, this is trivial since $A_0 = 0 = \min \Phi_0$.  Then, the one step lemma proves it for all $k$. Then, the fact that 
		\begin{equation}
			\sqrt{A_{k+1}} - \sqrt{A_0} \geq \frac{k}{2\sqrt{L}} \implies A_{k+1} \geq \frac{k^2}{2L}
		\end{equation}
		and 
		\begin{equation}
			f(w_k) - f^\ast \leq \frac{\|w_0 - w^\ast\|^2}{2 A_k}
		\end{equation}
		gives the result. 
	\end{proof}

	Remarkably, we can write this method as a modified version of Polyak's heavy ball. Define 
	\begin{equation}
		\lambda_0 = 0, \quad \lambda_{k+1} = \frac{1 + \sqrt{1 + 4 \lambda_k^2}}{2}
	\end{equation}
	Further, 
	\begin{equation}
		w_{k+1} = y_k - \frac{1}{L} \nabla f(y_k), \quad y_k = w_k + \beta_k (w_{k+1} - w_k)
	\end{equation}
	with $\beta_k = \frac{\lambda_{k+1} - 1}{\lambda_{k+2}}$. 

	\begin{exercise}[Equivalence of the Update with AGD]
	  Show that the iterates generated by this update are the same as those generated by AGD. 
	\end{exercise}

\subsection{Preconditioning}

	Another way to speed up algorithms is by simply changing the model we minimize at each iteration. The idea is pretty simple. Recall that in GD, we simply iterate as 
	\begin{equation}
		w_{k+1} \gets \argmin_z \bigg\{ \langle \nabla f(w_k), z - w_k \rangle + \frac{1}{\eta_k} \|z - w_k\|_2^2 \bigg\}
	\end{equation}

	Perhaps using the Euclidean norm may not be best choice, especially since for badly conditioned quadratics, this leads to very slow convergence. 

	\begin{definition}[Mahalanobis Norm]
	  \label{def:mahalanobis-norm}
	  Given a symmetric positive definite matrix $\Sigma \succ 0$, define the norm 
		\begin{equation}
			\|w\|_\Sigma = \sqrt{w^T \Sigma w}
		\end{equation}
	\end{definition}

	Thus, giving a sequence of matrices $P_1, P_2, \ldots$, setting the ``right'' geometry at iteration $k$, we can define the following. 

	\begin{definition}[Preconditioned Gradient Descent]
	  \label{def:preconditioned-gradient-descent}
		\begin{equation}
			w_{k+1} \gets \argmin_z \bigg\{ \langle \nabla f(w_k), z - w_k \rangle + \frac{1}{\eta_k} \|z - w_k\|_\Sigma^2 \bigg\} = w_k - \eta_k P_k^{-1} \nabla f(w_k)
		\end{equation}
	\end{definition}

	Recall that by the spectral theorem, we can diagonalize PSD \hyperref[lin-def:symmetric-matrices]{symmetric matrices} as $P = U D U^T$. The preconditioning rescales certain directions via $D^{-1}$, hopefully making conditioning better. There are several natrual questions. First, does preconditioning actually help? How do we design good preconditioners? Are they efficient? Let's establish some definitions first. 

	\begin{definition}[$L_P$-Smooth]
	  \label{def:lp-smooth}
		We say that differentiable $f: \mathbb{R}^d \to \mathbb{R}$ is $L_P$-smooth with respect to $P \succ 0$ if 
		\begin{equation}
			\|\nabla f(w_1) - \nabla f(w_2) \|_{P^{-1}} \leq L_P \|w_1 - w_2\|_P \quad \forall w_1, w_2 \in \mathbb{R}^d
		\end{equation}
	\end{definition}

	\begin{definition}[$\mu_P$-Strongly Convex]
	  \label{def:mup-strongly-convex}
		$f: \mathbb{R}^d \to \mathbb{R}$ is $\mu_P$-strongly convex w.r.t. $P \succ 0$ if 
		 \begin{equation}
			 \frac{\mu_P}{2} \|w_1 - w_2 \|_P^2 + \langle \nabla f(w_1), w_2 - w_1 \rangle + f(w_1) \leq f(w_2) \quad \forall w_1, w_2
		 \end{equation}
	\end{definition}

	\begin{definition}[$\mu_P$-PL]
	  \label{def:mup-pl}
		$f: \mathbb{R}^d \to \mathbb{R}$ satisfies $\mu_P$-PL w.r.t. $P \succ 0$ if 
		\begin{equation}
			\mu_P (f(w) - f^\ast) \leq \frac{1}{2} \|\nabla f(w)\|_{P^{-1}}^2
		\end{equation}
	\end{definition}

	If we fix $P_k = P$, then there is a very slick interpretation. 

	\begin{lemma}[Scale Change]
	  \label{thm:scale-change}
 
		Let $P$ be a symmetric and positive-definite. $f: \mathbb{R}^d \to \mathbb{R}$ be differentiable, and $\varphi(z) \coloneqq f(P^{-1/2} z)$. Initialize $w_0$ and $z_0 = P^{1/2} w_0$. 
		Then, consider the following. 
		\begin{enumerate}
		  \item \textit{Iterates of PGD}. Take the iterates of PGD of $f$ from $w_0$. 
				\begin{equation}
					w_{k+1} = w_k - \eta_k P^{-1} \nabla f(w_k)
				\end{equation}

			\item \textit{Iterates of GD}. Take the iterates of GD of $\varphi$ from $z_0$. 
				\begin{equation}
					z_{k+1} = z - \eta_k \nabla \varphi(z_k)
				\end{equation}
		\end{enumerate}
		Then, 
		\begin{equation}
			w_k = P^{-1/2} z_k, \quad f(w_k) = \varphi(z_k)
		\end{equation}
		This can be seen as the commuting diagram. 
		\begin{equation}
			\begin{tikzcd}[column sep=large, row sep=large]
			z_0 \arrow[r, "P^{-1/2}"] \arrow[d, "\text{(GD) on } \varphi"'] 
				& w_0 \arrow[d, "\text{(PGD) on } f"] \\
			z_K 
				& w_K \arrow[l, "P^{1/2}"]
			\end{tikzcd}
		\end{equation}
	\end{lemma}
	\begin{proof}
	  This is trivial using induction. 
	\end{proof}

	\begin{example}[Quadratic]
	  \label{ex:quadratic}
	  Performing PGD with preconditioning matrix $P$ on the quadratic
		\begin{equation}
			f(z) = z^T A z + b^T z + c, \quad A \succ 0, b \in \mathbb{R}^d, c \in \mathbb{R}
		\end{equation}
		is equivalent to performing GD on the function 
		\begin{equation}
			\varphi(w) = f(P^{-1/2} w) = w^T (P^{-1/2} A P^{-1/2}) w + (b^T P^{-1/2}) z + c
		\end{equation}
		which we know is $\mu$-PL and $L$-smooth. So,

\subsection{Saddle Points} 

	We started with GD, and we said that if we only have smoothness, we can make sure that the norm of the gradient goes to $0$. If we have more structure like SC or PL, then we will have fast convergence. We then talked about when is it that we have local convergence. But we have only covered nonconvex problems where the \hyperref[real-def:critical-point]{critical points} are local minima. This is what PL tells us. So when do first order methods converge to local/global minimizers only? In neural networks, we often just initialize at random and it seems to work. So we want to figure out why. 

	Today, we will ``show'' that GD avoids certain critical points, which are called \textit{strict saddles}. 

	\begin{definition}[Strict Saddle]
	  \label{def:strict-saddle}
		A point $w_{\#}$ is a \textbf{strict saddle point} if 
		\begin{enumerate}
			\item $\nabla f(w_{\#}) = 0$  and $\lambda_{\min}(\nabla^2 f(w^{\#}) < 0$.\footnote{Note that a maximizer could be a saddle. We just require some direction to point down.} 
			\item 
		\end{enumerate}
	\end{definition}
	
	\begin{example}[]
	  \label{ex:function-simulation-see-when-initialize}
	  Consider the function 
		\begin{equation}
		  f(w, z) = -(\alpha/2) w^2 + (\beta/2)z^2, \quad \alpha, \beta > 0
		\end{equation}
		From the simulation, we can see that when we initialize the optimizer to the form $(w, 0)$, then it gets stuck at this saddle point. Fortunately, this set is \hyperref[mt-def:almost_everywhere]{measure $0$}. 
	\end{example}

	\hyperref[mt-thm:fubinis-theorem]{Fubini's theorem} from measure theory implies the Fubini's theorem in the notes. Now let's talk about escaping strict saddles. 

	\begin{theorem}[Escaping Strict Saddles]
	  \label{thm:escaping-strict-saddles}
		Let $f: \mathbb{R}^d \to \mathbb{R}$ be a $C^2$ function with an $L$-Lipschitz gradient. Suppose we ran GD with $\eta \in (0, \frac{1}{L})$ and $w_0$ is picked at random w.r.t. a continuous density. Then, for any \hyperref[def:strict-saddle]{strict saddle} $w_{\#}$, we have 
		\begin{equation}
			\mathbb{P}(\lim_{k \to +\infty} w_k = w_{\#}) = 0
		\end{equation}
		Note that this limit may not converge at all. 
	\end{theorem}
	\begin{proof}
		The idea is to take each \hyperref[real-def:critical-point]{critical point}, and think of the set of points that will converge onto that point. WLOG, assume $w_{\#} = 0$ (since we can translate) and the \hyperref[mv-def:hessian]{Hessian} is diagonal. 
		\begin{equation}
			\nabla^2 f(0) = \begin{pmatrix} 
				D_{-} & 0 \\ 0 & D_+ 
			\end{pmatrix}, \quad D_- \prec 0, D_+ \succeq 0
		\end{equation}
		Define the GD map as $T(w) = w - \eta \nabla f(w)$. Define 
		\begin{equation}
			\gamma = \lambda_{\min} (-D_-) \succ 0
		\end{equation}
		By continuity of the Hessian, there is a radius $r > 0$ s.t. 
		\begin{equation}
			\|\nabla^2 f(w) - \nabla^2 f(0)\|_{\mathrm{op}}  \leq \frac{\gamma}{4}, \quad w \in \underbrace{\overline{B_r}(0)}_{K}
		\end{equation}
		which is equivalent to 
		\begin{equation}
			\| \nabla T(w) - \nabla T(0) \|_{\mathrm{op}} \leq \frac{\eta \gamma}{4}
		\end{equation}
		The idea is to divide up the measure $0$ ``line'' into a closer set and a further set. Set
		\begin{equation}
			S = \{w \in K \colon T^k (w) \in K \forall k \} = \bigcap_{k=0}^\infty (T^k)^{-1} (K)
		\end{equation}
		Finally, let 
		\begin{equation}
			\mathcal{C}(w_{\#}) = \{w_0 \mid T^k (w_0) \to w_{\#} \}
		\end{equation}
		Our goal is to show that this set has measure $0$. Any trajectory that converges to $w_{\#}$ eventually stays in $S$. Thus, 
		\begin{equation}
			\mathcal{C}(w_{\#}) \subset \bigcup_{j=0}^\infty (T^j)^{-1} (S)
		\end{equation}
		and we wish to show that the RHS has measure $0$. We do this in multiple steps. 
		\begin{enumerate}
			\item \textit{$S$ has measure $0$}. The idea is that given any two points in the ball $K$, the points must get further and further away, and so at most 1 point can stay in the ball. Since the Hessian is of the diagonal form above, we must have 
				\begin{equation}
					\nabla T(0) = \begin{pmatrix} I - \eta D_- & \\ & I - \eta D_+ \end{pmatrix} 
				\end{equation}
				where the first and second diagonal blocks are $m \times m$ and $(d -m) \times (d - m)$ matrices. In particular, for $u \in \mathbb{R}^m, v \in \mathbb{R}^{d - m}$, 
				\begin{equation}
				  \|(I - \eta D_-) u\| \geq (1 + \eta \gamma) \|u\|
				\end{equation}
				where $\gamma = \lambda_{\min} (-D_-)$ is the minimum eigenvalue of $\nabla T(0)$. We also know that 
				\begin{equation}
				  \|(I - \eta D_-)v\| \leq \|v\|
				\end{equation}
				Consider four points 
				\begin{equation}

	\begin{lemma}[A Three-Point Identity for Smooth Functions]
	  \label{thm:a-three-point-identity-for-smooth-functions}
 
		Another lemma is that for any pair $w, y \in \mathbb{R}^d$, 
		\begin{equation}
		  \| T(w) - T(y)\| \geq (1 - \eta L) \| w - y\|
		\end{equation}
		This is pretty counterintuitive, but remember that $\eta \in (0, \frac{1}{L})$. Hence, $T$ is injective and its inverse $T^{-1}$ is $(1 - \eta L)^{-1}$-Lipschitz. 
	\end{lemma}
	\begin{proof}
	  Expanding, 
		\begin{align}
			\|T(w) - T(y) \| & = \|w - y - \eta (\nabla f(w) - \nabla f(y))\| \\ 
											 & \geq \|w - y\| - \eta \|\nabla f(w) - \nabla f(y)\| \\ 
											 & \geq (1 - \eta L) \|w - y\|
		\end{align}
		where the first inequality comes from reverse triangle inequality, and the second comes from the fact that $f$ is $L$-smooth. 
	\end{proof}

	\begin{lemma}[Lipschitz Maps Preserve Null Sets]
	  \label{thm:lipschitz-maps-preserve-null-sets}
 
		Suppose $F: \mathbb{R}^d \to \mathbb{R}^d$ is $L$-lipschitz, and $Q$ has measure $0$. Then, $F(Q)$ has measure $0$. 
	\end{lemma}
